1"""Plain-words summaries of momwire's solver advisories (AK#1981). 

2 

3momwire composes its advisories for the people who maintain it: issue 

4numbers, solver internals (`n_qp_pair`, `n_per_edge_per_wire`), and its own 

50-based wire indices, which name no wire in the user's deck. Jeff G8HUL met 

6`CoarseCrossingNode` on a vertical with a ground stake and called it "totally 

7incomprehensible" (groups.io antenna-research #1605). He was right about the 

8wording, and the advisory was right in substance. 

9 

10So each KNOWN category gets one plain sentence or three, led by what the 

11advisory is about in the deck's own terms, roughly what it is worth, and 

12what to do. momwire's text is kept untouched as the detail behind it. 

13 

14Read, never rewritten: the numbers come out of momwire's own sentence, so a 

15summary can only say what the advisory said. A sentence this module cannot 

16read gets no summary, and the UI shows momwire's text in full as before. 

17The tests build each advisory with momwire's own composer, so a rewording on 

18the momwire side fails here rather than going quietly unsummarised. 

19 

20The quadrature lever momwire's crossing advisory names is deliberately NOT 

21repeated. On Jeff's deck (buried, so bs2 already runs at order 32) forcing 

22the order anywhere from 8 to 128 moves X by 0.002 ohm while the mesh moves 

23it 3.5 ohm (AK#1981); the #760 figure it quotes is the soil-A fan at order 8. 

24Adding segments is the advice that holds. 

25"""

49def _length(mm: float) -> str: 

50 """A segment length as a modeller reads it: millimetres when short.""" 

51 return f"{mm:.0f} mm" if mm < 100 else f"{mm / 1000:.2f} m"

54def _coarse_crossing(text: str, locate) -> str | None: 

55 m = _CROSSING.search(text) 

56 if m is None: 

57 return None 

58 wire, end, side, h = int(m.group(1)), m.group(2), m.group(3), float(m.group(5)) 

59 point = locate(wire, end) if locate is not None else None 

60 where = ( 

61 f"at x = {point[0]:.3g} m, y = {point[1]:.3g} m" 

62 if point is not None 

63 else "at a joint" 

64 ) 

65 floor = "" 

66 f = _CROSSING_FLOOR.search(text) 

67 if f is not None: 

68 floor = ( 

69 f" This wire meets the ground at a slope, so no segment there can " 

70 f"be shorter than {_length(float(f.group(1)))}." 

71 ) 

72 return ( 

73 f"A wire meets the ground {where}, and the segments of the wire " 

74 f"{side} the ground next to that " 

75 f"joint are {_length(h)} long: too long for the impedance there to " 

76 f"have settled. Expect the reactance to move by up to a few ohms as " 

77 f"you refine. Add segments to the wires at the joint (or to the whole " 

78 f"model) until the readout stops changing.{floor}" 

79 )

82def _surface_height(text: str, _locate) -> str | None: 

83 m = _SURFACE.search(text) 

84 if m is None: 

85 return None 

86 n, h, ratio = int(m.group(1)), float(m.group(2)), m.group(3) 

87 wires = "A wire lies" if n == 1 else f"{n} wires lie" 

88 radii = "" if ratio == "inf" else f" ({float(ratio):.3g} wire radii)" 

89 return ( 

90 f"{wires} only {h:.3g} mm above the ground{radii}. That close, the " 

91 f"impedance depends strongly on the exact height, so model the " 

92 f"height the wire really has: the same wire lying deeper in the grass " 

93 f"is a measurably different antenna." 

94 )

97def _far_mesh(text: str, _locate) -> str | None: 

98 m = _FAR_MESH.search(text) 

99 if m is None: 

100 return None 

101 return ( 

102 f"razor-2p's answer keeps moving as segments are added: on typical " 

103 f"models it sits a few percent from converged (median {m.group(1)} %). " 

104 f"Doubling the segments roughly halves that, and bs2 is already " 

105 f"converged at the same segment counts." 

106 )

109def _ek_above_ground(text: str, _locate) -> str | None: 

110 """B-spline (momwire 0.78, momwire#1170) and razor-2p (momwire 0.82, 

111 momwire#1614) serve the extended kernel on a deck with buried wires in 

112 part: the pairs above the ground take it, the buried wires keep the 

113 standard kernel. One summary for both ways of getting there, an explicit 

114 request and a deck whose dialect asks.""" 

115 m = _EK_ABOVE.search(text) 

116 if m is None: 

117 return None 

118 n, total = int(m.group(1)), int(m.group(2)) 

119 buried = "The 1 wire" if n == 1 else f"The {n} wires" 

120 return ( 

121 f"The extended (fat-wire) kernel was used on the wires above the " 

122 f"ground. {buried} below the ground (of {total}) used the standard " 

123 f"thin-wire kernel, because the extended kernel is not available " 

124 f"there. The difference matters for fat wires on short segments, so " 

125 f"on ordinary buried radials it is small." 

126 )

137def summary(category: str, text: str, locate=None) -> str | None: 

138 """The plain-words summary of one momwire advisory, or None. 

139 

140 ``locate(wire, end)`` maps momwire's 0-based wire index and ``"start"`` / 

141 ``"end"`` to that end's (x, y, z) in metres, or None when it cannot; it 

142 lets the crossing summary name the joint by where it is rather than by an 

143 index the deck never shows. None for an unknown category, and for a known 

144 one whose sentence this module cannot read. 

145 """ 

146 fn = _SUMMARISERS.get(category) 

147 return fn(text, locate) if fn is not None else None

150# -------------------------------------------------------------------------- 

151# AK's own "ExtendedKernel" advisory (AK#1996) 

152# -------------------------------------------------------------------------- 

153# The extended kernel as a DEFAULT that this solve could not take. Unlike the 

154# advisories above this one is composed here, from what the engine knows, and 

155# never from momwire's refusal sentence: that sentence is written for 

156# momwire's maintainers (issue numbers, a Python argument) and blamed nobody 

157# in particular, where the summary used to blame the deck (G8HUL, 

158# antenna-research #1620: no EK card, read as NEC-4). 

159# 

160# Four pieces, each its own function so one can change without the others: 

161# WHY the kernel was on, WHAT refused it, WHAT WAS AFFECTED, and what the 

162# user can do. The B-spline basis (since momwire 0.78) and razor-2p (since 

163# momwire 0.82) serve a buried deck in part (extended kernel above the 

164# ground, standard kernel on the buried wires), so on those two nothing falls 

165# back and this advisory is not composed; momwire's own 

166# `ExtendedKernelAboveGround` says what happened, summarised by 

167# `_ek_above_ground` above. 

168 

169_DIALECT_WORD = {"nec4": "NEC-4", "nec5": "NEC-5"} 

170 

171# Why the extended kernel could not be used, completing "It is ...". 

172EK_CAUSE_WORDS = { 

173 "basis": "This solver does not have it", 

174 "enrichment": "This solver does not have it together with singular enrichment", 

175 "buried": "It is not available for wires below the ground", 

176 "stepped": "It is not available where wires of different radii meet", 

177} 

178 

179EK_SOURCES = ("dialect", "ez_engine", "ez_reading", "card", "default")

182def _ek_why_on(source: str, dialect: str | None) -> str: 

183 """Why this solve was to use the extended kernel. ``source`` is what the 

184 engine could establish: the deck's reading ("dialect"), an EZNEC model's 

185 engine setting ("ez_engine") or the reading chosen for it ("ez_reading"), 

186 a NEC-2 deck's own EK card ("card"), or none of these ("default": a 

187 caller turned the default on for a design that does not say why).""" 

188 nec = _DIALECT_WORD.get(dialect) 

189 solved = "solved with the extended (fat-wire) kernel" 

190 if source == "dialect" and nec: 

191 return f"This deck is read as {nec}, which is {solved}." 

192 if source == "ez_reading" and nec: 

193 return f"This EZNEC model is read as {nec}, which is {solved}." 

194 if source == "ez_engine" and nec: 

195 return f"This EZNEC model is set to the {nec} engine, so it is {solved}." 

196 if source == "card": 

197 return "This deck's EK card asks for the extended (fat-wire) kernel." 

198 return "This design is solved with the extended (fat-wire) kernel by default."

201def _ek_affected(cause: str) -> tuple[str, str]: 

202 """What the fallback touched, as (the summary's clause, the details' 

203 sentence). Today it is always the whole solve, whatever the cause.""" 

204 also = { 

205 "buried": ", not only the wires below the ground", 

206 "stepped": ", not only the wires at that joint", 

207 }.get(cause, "") 

208 return ( 

209 "this whole solve used the standard thin-wire kernel", 

210 "What was affected: the whole solve. Every wire used the standard " 

211 f"thin-wire kernel{also}.", 

212 )

219def _ek_caused_by(cause: str, facts: dict) -> str: 

220 """Which part of the model or the setup the refusal is about, with the 

221 deck's own numbers where the engine supplied them in ``facts``.""" 

222 if cause == "buried": 

223 depth = facts.get("depth_m") 

224 down = f", down to {_metres(depth)} deep" if depth else "" 

225 return f"What caused it: this model has wire below the ground{down}." 

226 if cause == "stepped": 

227 joints = facts.get("joints") or () 

228 if not joints: 

229 return "What caused it: wires of different radii meet at a joint." 

230 x, y, z = joints[0] 

231 more = len(joints) - 1 

232 others = ( 

233 "" 

234 if more == 0 

235 else " and at 1 other joint" 

236 if more == 1 

237 else f" and at {more} other joints" 

238 ) 

239 return ( 

240 "What caused it: wires of different radii meet at " 

241 f"x = {x:.3g} m, y = {y:.3g} m, z = {z:.3g} m{others}." 

242 ) 

243 if cause == "enrichment": 

244 return ( 

245 "What caused it: singular enrichment is switched on for this " 

246 "solver, and the two cannot be used together." 

247 ) 

248 return "What caused it: the solver chosen for this slot has no extended kernel."

252 "How much it matters: the two kernels differ where a wire is fat for its " 

253 "segments. Expect a fraction of a percent where the segments are more " 

254 "than about 10 wire radii long, and several percent where they are " 

255 "shorter than about 3. A thin wire on ordinary segments is hardly " 

256 "affected." 

257)

264 ), 

265 "enrichment": (

290def extended_kernel_fallback( 

291 cause: str, 

292 source: str = "default", 

293 dialect: str | None = None, 

294 facts: dict | None = None, 

295) -> tuple[str, str]: 

296 """The "ExtendedKernel" advisory's ``(summary, details)`` in plain words. 

297 

298 ``cause`` is a key of `EK_CAUSE_WORDS`; ``source`` one of `EK_SOURCES` 

299 with ``dialect`` ("nec4" / "nec5") for every source that names one; ``facts`` 

300 the deck's own numbers for the cause: ``depth_m`` (how far the deepest 

301 wire goes below the ground) and ``joints`` (the (x, y, z) of each joint 

302 where wires of different radii meet).""" 

303 facts = facts or {} 

304 clause, affected = _ek_affected(cause) 

305 summary = ( 

306 f"{_ek_why_on(source, dialect)} {EK_CAUSE_WORDS[cause]}, so {clause}. " 

307 "The difference matters for fat wires on short segments." 

308 ) 

309 details = " ".join( 

310 [affected, _ek_caused_by(cause, facts), _EK_HOW_MUCH, _EK_WHAT_TO_DO[cause]] 

311 ) 

312 return summary, details
