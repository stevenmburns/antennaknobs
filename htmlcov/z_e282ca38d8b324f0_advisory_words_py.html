1"""Plain-words summaries of momwire's solver advisories (AK#1981). 

2 

3momwire composes its advisories for the people who maintain it: issue 

4numbers, solver internals (`n_qp_pair`, `n_per_edge_per_wire`), and its own 

50-based wire indices, which name no wire in the user's deck. Jeff G8HUL met 

6`CoarseCrossingNode` on a vertical with a ground stake and called it "totally 

7incomprehensible" (groups.io antenna-research #1605). He was right about the 

8wording, and the advisory was right in substance. 

9 

10So each KNOWN category gets one plain sentence or three, led by what the 

11advisory is about in the deck's own terms, roughly what it is worth, and 

12what to do. momwire's text is kept untouched as the detail behind it. 

13 

14Read, never rewritten: the numbers come out of momwire's own sentence, so a 

15summary can only say what the advisory said. A sentence this module cannot 

16read gets no summary, and the UI shows momwire's text in full as before. 

17The tests build each advisory with momwire's own composer, so a rewording on 

18the momwire side fails here rather than going quietly unsummarised. 

19 

20The quadrature lever momwire's crossing advisory names is deliberately NOT 

21repeated. On Jeff's deck (buried, so bs2 already runs at order 32) forcing 

22the order anywhere from 8 to 128 moves X by 0.002 ohm while the mesh moves 

23it 3.5 ohm (AK#1981); the #760 figure it quotes is the soil-A fan at order 8. 

24Adding segments is the advice that holds. 

25""" 

26 

27from __future__ import annotations 

28 

29import re 

30from collections.abc import Callable 

31 

32_NUM = r"([0-9]+(?:\.[0-9]+)?)" 

33 

34_CROSSING = re.compile( 

35 r"crossing node: wire (\d+)'s (start|end) is the (above|below) member, and the " 

36 rf"finest mesh within {_NUM} mm of the node is {_NUM} mm" 

37) 

38_CROSSING_FLOOR = re.compile(rf"from L_min = {_NUM} mm at the node") 

39 

40_SURFACE = re.compile( 

41 rf"(\d+) near-ground wire\(s\) at h = {_NUM} mm, h/a = ({_NUM[1:-1]}|inf)" 

42) 

43 

44_FAR_MESH = re.compile(rf"a median {_NUM} % from converged, with a tail to {_NUM} %") 

45 

46 

47def _length(mm: float) -> str: 

48 """A segment length as a modeller reads it: millimetres when short.""" 

49 return f"{mm:.0f} mm" if mm < 100 else f"{mm / 1000:.2f} m" 

50 

51 

52def _coarse_crossing(text: str, locate) -> str | None: 

53 m = _CROSSING.search(text) 

54 if m is None: 

55 return None 

56 wire, end, side, h = int(m.group(1)), m.group(2), m.group(3), float(m.group(5)) 

57 point = locate(wire, end) if locate is not None else None 

58 where = ( 

59 f"at x = {point[0]:.3g} m, y = {point[1]:.3g} m" 

60 if point is not None 

61 else "at a joint" 

62 ) 

63 floor = "" 

64 f = _CROSSING_FLOOR.search(text) 

65 if f is not None: 

66 floor = ( 

67 f" This wire meets the ground at a slope, so no segment there can " 

68 f"be shorter than {_length(float(f.group(1)))}." 

69 ) 

70 return ( 

71 f"A wire meets the ground {where}, and the segments of the wire " 

72 f"{side} the ground next to that " 

73 f"joint are {_length(h)} long: too long for the impedance there to " 

74 f"have settled. Expect the reactance to move by up to a few ohms as " 

75 f"you refine. Add segments to the wires at the joint (or to the whole " 

76 f"model) until the readout stops changing.{floor}" 

77 ) 

78 

79 

80def _surface_height(text: str, _locate) -> str | None: 

81 m = _SURFACE.search(text) 

82 if m is None: 

83 return None 

84 n, h, ratio = int(m.group(1)), float(m.group(2)), m.group(3) 

85 wires = "A wire lies" if n == 1 else f"{n} wires lie" 

86 radii = "" if ratio == "inf" else f" ({float(ratio):.3g} wire radii)" 

87 return ( 

88 f"{wires} only {h:.3g} mm above the ground{radii}. That close, the " 

89 f"impedance depends strongly on the exact height, so model the " 

90 f"height the wire really has: the same wire lying deeper in the grass " 

91 f"is a measurably different antenna." 

92 ) 

93 

94 

95def _far_mesh(text: str, _locate) -> str | None: 

96 m = _FAR_MESH.search(text) 

97 if m is None: 

98 return None 

99 return ( 

100 f"razor-2p's answer keeps moving as segments are added: on typical " 

101 f"models it sits a few percent from converged (median {m.group(1)} %). " 

102 f"Doubling the segments roughly halves that, and bs2 is already " 

103 f"converged at the same segment counts." 

104 ) 

105 

106 

107_SUMMARISERS: dict[str, Callable] = { 

108 "CoarseCrossingNode": _coarse_crossing, 

109 "SurfaceRadialHeight": _surface_height, 

110 "RazorFarMeshClass": _far_mesh, 

111} 

112 

113 

114def summary(category: str, text: str, locate=None) -> str | None: 

115 """The plain-words summary of one momwire advisory, or None. 

116 

117 ``locate(wire, end)`` maps momwire's 0-based wire index and ``"start"`` / 

118 ``"end"`` to that end's (x, y, z) in metres, or None when it cannot; it 

119 lets the crossing summary name the joint by where it is rather than by an 

120 index the deck never shows. None for an unknown category, and for a known 

121 one whose sentence this module cannot read. 

122 """ 

123 fn = _SUMMARISERS.get(category) 

124 return fn(text, locate) if fn is not None else None
