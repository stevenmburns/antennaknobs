1"""1x2 stacked bowtie array with a balanced-line corporate feed. 

2 

3A worked example of feeding a phased array through `BalancedLine` rather than 

4the framework's per-element delta-gap phasing (`Array2x2Builder`). Two bowtie 

5elements are stacked half a wavelength apart and combined by a matched 

6balanced-line corporate feed: 

7 

8 element (~100 Ω) --100 Ω BalancedLine half--> center tap <--half-- element 

9 | 

10 50 Ω coax (single-ended) 

11 

12The impedance ladder is the intent: two ~100 Ω-class elements combined by 

13100 Ω half-lines that parallel to 50 Ω at the center tap — a 1:1 match to 50 Ω 

14coax (SWR ~1.01 at the design frequency), +5.9 dBi bidirectional broadside 

15(φ = 0/180) with deep endfire nulls. In practice the geometry is tuned so the 

16*tap* presents 50 Ω directly, NOT by matching each element to a transparent 

17100 Ω line: an element's impedance in an array is feed-dependent (the current 

18ratio the balanced network sets differs from an isolated or delta-gap drive), 

19so there is no feed-independent "100 Ω element" — the tuned element-plus-line 

20network as a whole realizes the 50 Ω tap. (Setting each element to exactly 

21100+0j under a delta-gap drive gives ~43+23j at the tap, SWR 1.7, not 50 Ω — 

22which is why the tune targets the tap.) 

23 

24Against ideal equal-current feeding of the same geometry the balanced feed is 

25pattern-identical (within ~0.2 dB everywhere): it is lossless and matched, so 

26it faithfully delivers the broadside excitation — the corporate feed costs 

27nothing in gain or pattern here. 

28 

29Freq-based geometry (unlike `specialty.bowtie`, whose dimensions are hand-tuned 

30absolute metres): every length is a wavelength factor, so the whole antenna and 

31its feed scale with `freq`. 

32 

33Construction (issues #575/#576/#579): 

34 - Each bowtie is authored WITHOUT its center feed-bridge wire, so the two 

35 feed-gap ends are free conductor ENDS. `PortAtEnd` grabs each end (four 

36 junction ports total), and one `BalancedLine` per element pairs them to the 

37 shared center-tap nodes. 

38 - The center tap is a `PortVirtual` node pair. The 50 Ω coax is modelled as a 

39 single-ended feed there: `Driven` on one tap conductor, the other bonded to 

40 the common datum (the coax shield) — i.e. coax straight onto the balanced 

41 tap, no balun. That single-ended grounding also pins the feed network's 

42 common mode, so `zcomm` is not needed (the lines are pure feeders, not 

43 loop conductors — contrast `wire.sterba_bl`, whose risers close radiating 

44 loops and REQUIRE `zcomm`). `zcomm` is exposed for study but defaults off. 

45 

46Engine support: **momwire only** — `PortAtEnd` resolves to a junction-node port 

47(momwire#172) that NEC-2 has no card for, so `PyNECEngine` rejects the design. 

48""" 

49 

50import math 

51from types import MappingProxyType 

52 

53from antennaknobs import AntennaBuilder 

54from antennaknobs.network import ( 

55 BalancedLine, 

56 Driven, 

57 Network, 

58 PortAtEnd, 

59 PortVirtual, 

60 Shunt, 

61 Wire, 

62) 

63 

64 

65class Builder(AntennaBuilder): 

66 default_params = MappingProxyType( 

67 { 

68 "freq": 28.47, 

69 # design_freq scales the geometry AND anchors auto_mesh; hidden. 

70 "design_freq": 28.47, 

71 # Bowtie flare half-angle and length (as a wavelength factor), 

72 # tuned so the center TAP presents 50 Ohm to the coax (SWR ~1.01). 

73 "angle_deg": 44.0, 

74 "length_factor": 0.560, 

75 # Element stacking distance / wavelength (sets the half-line length). 

76 "del_z_factor": 0.5, 

77 # Corporate-feed line impedance (matched to the element). 

78 "zdiff": 100.0, 

79 # Feeders are common-mode-open (0 -> None): the single-ended tap 

80 # pins the CM. Exposed for the modelling study only. 

81 "zcomm": 0.0, 

82 "ui_params": MappingProxyType( 

83 { 

84 "design_freq": {"hidden": True}, 

85 "target_z0": 50.0, 

86 "angle_deg": {"min": 30.0, "max": 60.0, "step": 0.5}, 

87 "length_factor": {"min": 0.45, "max": 0.65, "step": 0.005}, 

88 "del_z_factor": {"min": 0.35, "max": 0.75, "step": 0.05}, 

89 "zdiff": {"min": 50.0, "max": 300.0, "step": 10.0}, 

90 "zcomm": {"min": 0.0, "max": 400.0, "step": 25.0}, 

91 } 

92 ), 

93 } 

94 ) 

95 

96 # ---- geometry ------------------------------------------------------- 

97 def _element_geo(self): 

98 """(wavelength, half-width y, half-height z) of one bowtie.""" 

99 wl = self.design_wavelength 

100 length = self.length_factor * wl 

101 theta = math.radians(self.angle_deg) 

102 half = 0.5 * length / (1.0 + math.sin(theta)) 

103 return wl, half * math.cos(theta), half * math.sin(theta) 

104 

105 def _element_wires(self, idx, zoff): 

106 """One bowtie at vertical offset `zoff`, feed-bridge omitted; the two 

107 feed-gap ends are named ``feedL{idx}`` / ``feedR{idx}`` for PortAtEnd.""" 

108 _, y, z = self._element_geo() 

109 eps = 0.05 # feed-gap half-width (m) 

110 nameL, nameR = f"feedL{idx}", f"feedR{idx}" 

111 # (y, z) pairs in the element's own plane (x = 0); name only the two 

112 # feed-leg wires whose inner end is a feed terminal (#578: a named wire 

113 # must be referenced — both are, by PortAtEnd). 

114 specs = [ 

115 ((-y, 0.0), (-y, z), None), 

116 ((-y, z), (-eps, eps), None), 

117 ((-eps, eps), (eps, eps), None), # top bridge (solid) 

118 ((eps, eps), (y, z), None), 

119 ((y, z), (y, 0.0), None), 

120 ((-y, 0.0), (-y, -z), None), 

121 ((-y, -z), (-eps, -eps), nameL), # left feed leg: end p1 = (-eps,-eps) 

122 ((eps, -eps), (y, -z), nameR), # right feed leg: end p0 = (eps,-eps) 

123 ((y, -z), (y, 0.0), None), 

124 ] 

125 return [ 

126 Wire((0.0, y0, z0 + zoff), (0.0, y1, z1 + zoff), name=nm) 

127 for ((y0, z0), (y1, z1), nm) in specs 

128 ] 

129 

130 def build_wires(self): 

131 wl, _, _ = self._element_geo() 

132 dz = self.del_z_factor * wl 

133 # Two elements straddling z = 0, so the center tap sits at mid-height. 

134 return self._element_wires(0, -0.5 * dz) + self._element_wires(1, +0.5 * dz) 

135 

136 # ---- feed network --------------------------------------------------- 

137 def build_network(self): 

138 wl, _, _ = self._element_geo() 

139 dz = self.del_z_factor * wl 

140 line_len = 0.5 * dz # element feed -> center tap 

141 zc = self.zcomm or None 

142 

143 ports = {"JC1": PortVirtual("JC1"), "JC2": PortVirtual("JC2")} 

144 branches = [] 

145 for idx in range(2): 

146 pL, pR = f"eL{idx}", f"eR{idx}" 

147 ports[pL] = PortAtEnd(f"feedL{idx}", end="p1") 

148 ports[pR] = PortAtEnd(f"feedR{idx}", end="p0") 

149 # port A = element pair, port B = shared center tap; same polarity 

150 # on both elements -> in-phase (broadside). 

151 branches.append( 

152 BalancedLine( 

153 a1=pL, 

154 a2=pR, 

155 b1="JC1", 

156 b2="JC2", 

157 zdiff=self.zdiff, 

158 length=line_len, 

159 zcomm=zc, 

160 ) 

161 ) 

162 # 50 Ohm coax at the tap: JC1 = center conductor (driven), JC2 = shield 

163 # (bonded to the common datum). The ground also pins the feed CM. 

164 branches.append(Shunt(port="JC2", l=0.0)) 

165 return Network( 

166 ports=ports, 

167 branches=branches, 

168 sources=[Driven(port="JC1", voltage=1 + 0j)], 

169 )
