1"""NEC-5 drop-in backend for the web UI (issue #825 follow-up). 

2 

3Mirrors the response shape of `web.server`'s momwire solver paths so the 

4frontend can swap a slot to NEC-5 via the `solver` request field, exactly 

5like the PyNEC backend. 

6 

7NEC-5 is licensed, user-supplied software: the backend exists in the 

8roster only when the serving machine resolves ``$NEC5_EXE`` (see 

9`engines.nec5.find_nec5`). The hosted simulator never defines it, so the 

10slot chooser can never offer NEC-5 hosted — running a local web instance 

11against a personally licensed binary is the supported use, and network 

12exposure of that instance is the operator's responsibility. Availability 

13is a runtime binary probe, not an import probe, so it is re-checked per 

14request (`have_nec5()`), unlike PyNEC's import-time ``HAVE_PYNEC``. 

15""" 

16 

17from __future__ import annotations 

18 

19from ..engines.nec5 import find_nec5 

20from .examples import REGISTRY as EXAMPLES 

21 

22 

23def have_nec5() -> bool: 

24 """True when a licensed NEC-5 binary is resolvable right now.""" 

25 return find_nec5() is not None 

26 

27 

28def solve(req: dict) -> dict: 

29 geometry = req.get("geometry", next(iter(EXAMPLES))) 

30 ex = EXAMPLES.get(geometry) or next(iter(EXAMPLES.values())) 

31 if ex.nec5_solve is None: 

32 raise ValueError(f"NEC-5 solve not implemented for geometry {ex.name!r}") 

33 return ex.nec5_solve(req) 

34 

35 

36def pattern(req: dict) -> dict: 

37 geometry = req.get("geometry", next(iter(EXAMPLES))) 

38 ex = EXAMPLES.get(geometry) or next(iter(EXAMPLES.values())) 

39 if ex.nec5_pattern is None: 

40 raise ValueError(f"NEC-5 pattern not implemented for geometry {ex.name!r}") 

41 return ex.nec5_pattern(req) 

42 

43 

44def _sweep_at(req: dict, freq_mhz: float) -> complex: 

45 """Single-frequency Z, one binary run per point (the streamed sweep 

46 endpoints call per-point; NEC5Engine's batched FR stepping is a future 

47 optimisation for uniform grids).""" 

48 req2 = dict(req) 

49 req2["measurement_freq_mhz"] = freq_mhz 

50 res = solve(req2) 

51 return complex(res["z_in_re"], res["z_in_im"]) 

52 

53 

54def _sweep_at_multifeed(req: dict, freq_mhz: float): 

55 """(primary_z, per-feed z list) at one frequency — the multi-feed 

56 NDJSON contract, same as the PyNEC twin.""" 

57 req2 = dict(req) 

58 req2["measurement_freq_mhz"] = freq_mhz 

59 res = solve(req2) 

60 primary = complex(res["z_in_re"], res["z_in_im"]) 

61 feeds_z = [complex(f["z_re"], f["z_im"]) for f in res.get("feeds", [])] 

62 return primary, feeds_z
