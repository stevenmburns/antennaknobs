1"""The frequency sweep's default range: ONE precedence (AK#1757, step 4). 

2 

3`design_range` is the rule every tool reads when a frequency sweep names no 

4range of its own: ``antennaknobs sweep --swr``, ``antennaknobs analyze`` (a 

5`analyses.FREQUENCY` sweep with no ``lo``/``hi``), the workbench's 

6``/analyses``, and ``/examples`` (whose ``sweep_range`` is `declared`, the 

7rungs a design states outright). First match wins: 

8 

91. ``"file"``: a file design's own sweep, ``ui_params["sweep_range"]`` with 

10 source "file" (a ``.nec`` FR card, a ``.ssn`` Generator sweep). 

112. ``"design"``: a design's ``ui_params["sweep_range"]``, else its 

12 ``meas_freq_range`` (the dial span it has always declared). 

133. ``"policy"``: its ``sweep_policy``: the band holding the anchor when 

14 band-locked, else the policy's factors around it. 

154. ``"default"``: ×1/1.25–×1.25 of the measurement frequency, the CLI's 

16 historical window. 

17 

18Rungs 1–2 are the design's own and absolute, so every tool sweeps the same 

19span. Rungs 3–4 are relative to an anchor. Here the anchor is the design's 

20own frequency; the workbench's frequency is session state (the band tab, the 

21dial lock), so its copy of rungs 3–4 stays in ``lib/sweep.ts``, where that 

22state lives, and ``/analyses`` serves no range for them. 

23 

24A density or spacing the design does not state is each tool's own: 

25``spacing`` / ``step`` / ``points`` None here. The CLI's own is 21 linear 

26points; the workbench's is its adaptive log grid. 

27"""

64@dataclass(frozen=True) 

65class SweepPolicy: 

66 """Where to centre the freq sweep and how wide to make it. 

67 

68 `anchor` picks which scalar the sweep range is anchored to: 

69 - "design_freq": sweep around the antenna's design frequency 

70 (the wider out-of-band picture; default for single-band antennas). 

71 - "meas_freq": sweep around the current measurement frequency 

72 (multi-band antennas like fan_dipole — keeps the trace focused 

73 on the band the user is currently tuning, since the design 

74 frequency stays pinned to band 0). 

75 

76 `lo_factor` / `hi_factor` are multiplicative bounds applied to the 

77 anchor. Defaults give a broad resonance/out-of-band view; multi-band 

78 antennas narrow to ±5% so the trace doesn't cross into neighbouring 

79 bands the user isn't tuning. 

80 """ 

81 

82 anchor: str = "design_freq" # "design_freq" | "meas_freq" 

83 lo_factor: float = 0.8 

84 hi_factor: float = 1.25 

85 # When True, the frontend snaps the sweep range to the [min_mhz, 

86 # max_mhz] of the band whose range contains the current anchor 

87 # frequency (typically measFreq for multi-band antennas). Falls 

88 # back to lo_factor/hi_factor multiplicatively if the anchor sits 

89 # outside every band. Used by fandipole so the sweep trace stays 

90 # inside the amateur band the user is currently tuning instead of 

91 # bleeding into the adjacent bands. 

92 band_locked: bool = False 

93 

94

120def sweep_policy(ui: Mapping) -> SweepPolicy: 

121 """Build a SweepPolicy from a `ui_params` dict's `sweep_policy` entry. 

122 

123 Accepts the positional 3-tuple `(anchor, lo_factor, hi_factor)` form or the 

124 mapping form (which can opt into named fields like `band_locked` without 

125 supplying every positional; missing fields fall back to the dataclass 

126 defaults). The mapping form is any Mapping: a design that freezes its 

127 ui_params spells it as a MappingProxyType. No entry yields the default 

128 policy; any other spelling — another type, or a key SweepPolicy does not 

129 have — raises. A dict-only check once dropped Dominator's and Challenger's 

130 band lock without a word (2026-07-12 to 2026-09-23), and a misspelt key 

131 would do the same. Takes any ui dict, so the same derivation runs for the 

132 default's ui_params and for each variant's deep-merged ui_params (see 

133 `variant_ui` in `_make_example`).""" 

134 raw = ui.get("sweep_policy") 

135 if raw is None: 

136 return DEFAULT_SWEEP_POLICY 

137 if isinstance(raw, (tuple, list)) and len(raw) == 3: 

138 return SweepPolicy( 

139 anchor=str(raw[0]), 

140 lo_factor=float(raw[1]), 

141 hi_factor=float(raw[2]), 

142 ) 

143 if isinstance(raw, Mapping): 

144 unknown = set(raw) - {"anchor", "lo_factor", "hi_factor", "band_locked"} 

145 if unknown: 

146 raise ValueError( 

147 f"sweep_policy has unknown keys {sorted(unknown)}; " 

148 "expected anchor, lo_factor, hi_factor, band_locked" 

149 ) 

150 d = DEFAULT_SWEEP_POLICY 

151 return SweepPolicy( 

152 anchor=str(raw.get("anchor", d.anchor)), 

153 lo_factor=float(raw.get("lo_factor", d.lo_factor)), 

154 hi_factor=float(raw.get("hi_factor", d.hi_factor)), 

155 band_locked=bool(raw.get("band_locked", d.band_locked)), 

156 ) 

157 raise ValueError( 

158 f"sweep_policy must be a mapping or an (anchor, lo_factor, hi_factor) " 

159 f"triple, got {type(raw).__name__}: {raw!r}" 

160 )

166@dataclass(frozen=True) 

167class FrequencyRange: 

168 """A frequency sweep's span in MHz, the precedence rung it came from, and 

169 the density the design states: ``step`` (MHz, lin) or ``points`` (the 

170 total count), at most one; ``spacing`` "lin" / "log", or None when the 

171 design states none.""" 

172 

173 lo: float 

174 hi: float 

175 level: str 

176 spacing: str | None = None 

177 step: float | None = None 

178 points: int | None = None 

179 

180 def count(self, default: int = CLI_POINTS) -> int: 

181 """How many points the range asks for: its own density, else 

182 ``default``. A step that does not divide the span still closes on 

183 ``hi`` (``lib/sweep.ts``'s ``sweepPointCount``, the same rule).""" 

184 if self.step is not None: 

185 span = self.hi - self.lo 

186 n = math.floor(span / self.step + 1e-9) + 1 

187 return n + 1 if self.lo + (n - 1) * self.step < self.hi - span * 1e-9 else n 

188 return self.points if self.points is not None else default 

189 

190 def grid(self, points: int | None = None) -> np.ndarray: 

191 """The frequencies swept. ``points`` given replaces the range's own 

192 density (``sweep --npoints``); else its own, else `CLI_POINTS`. An 

193 unstated spacing is the CLI's own, linear. A step that divides the 

194 span is ``linspace`` over it, so the grid equals the one 

195 ``--range lo hi --npoints n`` sweeps, bit for bit.""" 

196 if points is None and self.step is not None: 

197 n = self.count() 

198 if not math.isclose(self.lo + (n - 1) * self.step, self.hi, rel_tol=1e-9): 

199 return np.array( 

200 [*(self.lo + i * self.step for i in range(n - 1)), self.hi] 

201 ) 

202 return np.linspace(self.lo, self.hi, n) 

203 n = points if points is not None else self.count() 

204 if self.spacing == "log": 

205 return np.geomspace(self.lo, self.hi, n) 

206 return np.linspace(self.lo, self.hi, n) 

207 

208 def as_spec(self) -> dict: 

209 """The wire shape (``/examples`` ``sweep_range``, ``/analyses``): 

210 ``{lo, hi, spacing, source}`` plus ``step`` or ``points``. An unstated 

211 spacing is served as "log", the workbench's own.""" 

212 out: dict = { 

213 "lo": self.lo, 

214 "hi": self.hi, 

215 "spacing": self.spacing or "log", 

216 "source": "file" if self.level == "file" else "design", 

217 } 

218 if self.step is not None: 

219 out["step"] = self.step 

220 elif self.points is not None: 

221 out["points"] = self.points 

222 return out

233def normalise_sweep_range(raw) -> FrequencyRange | None: 

234 """``ui_params["sweep_range"]`` as a range, or None when absent or 

235 malformed (AK#1682). One density survives, in the order step > 

236 points_per_decade > points; a ``step`` on a log range becomes 

237 ``points``, and ``points_per_decade`` (accepted for backward 

238 compatibility) always does: round(ppd · log10(hi/lo)) + 1, at least 2.""" 

239 if not isinstance(raw, Mapping): 

240 return None 

241 try: 

242 lo, hi = float(raw["lo"]), float(raw["hi"]) 

243 except (KeyError, TypeError, ValueError): 

244 return None 

245 if not (0.0 < lo < hi and math.isfinite(hi)): 

246 return None 

247 spacing = str(raw.get("spacing", "lin")).lower() 

248 if spacing not in ("lin", "log"): 

249 return None 

250 level = "file" if raw.get("source") == "file" else "design" 

251 step, ppd, points = ( 

252 _positive(raw, "step"), 

253 _positive(raw, "points_per_decade"), 

254 _positive(raw, "points"), 

255 ) 

256 kw: dict = {} 

257 if step is not None: 

258 if spacing == "lin": 

259 kw["step"] = step 

260 else: 

261 kw["points"] = int(math.floor((hi - lo) / step + 1e-9)) + 1 

262 elif ppd is not None: 

263 kw["points"] = max(2, round(ppd * math.log10(hi / lo)) + 1) 

264 elif points is not None and points >= 2: 

265 kw["points"] = int(points) 

266 return FrequencyRange(lo, hi, level, spacing, **kw)

270 ui = params.get("ui_params")

274def declared(params: Mapping) -> FrequencyRange | None: 

275 """Rungs 1–2: the range a design states outright, or None.""" 

276 ui = _ui(params) 

277 own = normalise_sweep_range(ui.get("sweep_range")) 

278 if own is not None: 

279 return own 

280 dial = ui.get("meas_freq_range") 

281 if isinstance(dial, (tuple, list)) and len(dial) == 2: 

282 try: 

283 lo, hi = float(dial[0]), float(dial[1]) 

284 except (TypeError, ValueError): 

285 return None 

286 if 0.0 < lo < hi and math.isfinite(hi): 

287 return FrequencyRange(lo, hi, "design") 

288 return None

291def _bands(ui: Mapping) -> tuple[BandSpec, ...]: 

292 # A Mapping under "bands" is a repeat group's config, not a band table

300def design_range(params: Mapping) -> FrequencyRange: 

301 """The frequency sweep's range on a design with ``params`` (a builder's 

302 ``_params``: its knobs and ``ui_params``), rungs 1–4 (module doc).""" 

303 own = declared(params) 

304 if own is not None: 

305 return own 

306 freq = float(params["freq"]) 

307 ui = _ui(params) 

308 policy = sweep_policy(ui) 

309 if policy != DEFAULT_SWEEP_POLICY: 

310 anchor = ( 

311 float(params.get("design_freq", freq)) 

312 if policy.anchor == "design_freq" 

313 else freq 

314 ) 

315 if policy.band_locked: 

316 for band in _bands(ui): 

317 if band.min_mhz <= anchor <= band.max_mhz: 

318 return FrequencyRange(band.min_mhz, band.max_mhz, "policy") 

319 # Off every band, a lock falls back to its factors around ITS anchor 

320 # (`SweepPolicy`), which need not be ``freq``. 

321 return FrequencyRange( 

322 anchor * policy.lo_factor, anchor * policy.hi_factor, "policy" 

323 ) 

324 return FrequencyRange(freq / _DEFAULT_FRACTION, freq * _DEFAULT_FRACTION, "default")
