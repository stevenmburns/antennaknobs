1"""The multiport-Y route, shared by the card-deck engines (AK#1280, AK#1678). 

2 

3A network no card expresses (a line, a transformer, a virtual driver, a 

4self-tuning tuner, ...) is solved outside the field solve: the engine measures 

5the antenna's short-circuit admittance Y at its real ports, one run per port, 

6and the shared `NetworkReducer` stamps the network on it. PyNEC has done this 

7since #575, NEC-5 since #1280, NEC-2 since #1678. 

8 

9What differs between NEC-5 and NEC-2 is only how a port is ADDRESSED (NEC-5's 

10EX sits on a knot, NEC-2's on a segment centre), how a deck is written and how 

11a printout is read. Everything downstream of Y is the same arithmetic, and it 

12lives here so the two cannot drift: the port index map and the reducer, the 

13reduced impedance and sweep, the network-resolved drive with the power its 

14sources deliver, the budget with the engine's conductor loss folded in, the 

15per-source-watt gain factor, and the reciprocity tripwire. 

16 

17Every function takes the ENGINE and reads ``eng._compute_y_matrix`` and 

18``eng._reducer`` at call time, never a bound method captured earlier: the gates 

19replace an engine's ``_compute_y_matrix`` on the instance, and a captured one 

20would run the binary behind their back. 

21"""

34def make_port_reducer(eng, network, real_names): 

35 """The engine's reducer over `network`, with its real ports first (in 

36 `real_names` order, which is Y's) and the virtual ports after them. 

37 

38 A self-tuning tuner (AK#1646, #1661) is tuned from this engine's OWN port 

39 admittance, so the reducer's ``y_at`` calls back into the engine.""" 

40 port_to_idx = {n: i for i, n in enumerate(real_names)} 

41 next_idx = len(real_names) 

42 for name, port in network.ports.items(): 

43 if isinstance(port, PortVirtual): 

44 port_to_idx[name] = next_idx 

45 next_idx += 1 

46 return make_reducer( 

47 network, 

48 port_to_idx, 

49 next_idx, 

50 y_at=lambda wl: eng._compute_y_matrix(wl), 

51 design_freq_mhz=design_freq_mhz(eng.builder), 

52 )

55def reduced_impedance(eng, freq_mhz): 

56 """The driven impedances at `freq_mhz`, one per network source.""" 

57 wl = C_LIGHT / (float(freq_mhz) * 1e6) 

58 return np.atleast_1d(eng._reducer.driven_impedance(eng._compute_y_matrix(wl), wl))

61def reduced_impedance_sweep(eng, freqs): 

62 """(n_freqs, n_driven) driven impedances, one reduction per frequency: the 

63 antenna Y and every branch the reducer stamps are frequency-dependent, so a 

64 single multi-frequency printout cannot serve this route.""" 

65 freqs = np.asarray(freqs, dtype=float) 

66 if freqs.ndim != 1 or freqs.size == 0: 

67 raise ValueError("freqs must be a 1-D non-empty array") 

68 out = [np.atleast_1d(reduced_impedance(eng, f)) for f in freqs] 

69 return np.array(out).reshape(freqs.size, -1)

84def reduced_state(eng, freq_mhz, *, impedances=False) -> ReducedState: 

85 """Y once, then everything a single-deck reading needs from it: the voltage 

86 the network resolves at every real port (which is what drives the one 

87 structure deck), the source power and the network's own losses — and, with 

88 ``impedances``, the driven impedances too.""" 

89 wl = C_LIGHT / (float(freq_mhz) * 1e6) 

90 Y = eng._compute_y_matrix(wl) 

91 red = eng._reducer 

92 zs = list(np.atleast_1d(red.driven_impedance(Y, wl))) if impedances else None 

93 V = red.resolve_voltages(red.apply_branches(Y, wl)) 

94 _v, efficiency, p_in, budget = red.excited_state(Y, wl) 

95 return ReducedState(Y, wl, V, zs, efficiency, p_in, budget)

98def fold_wire_loss(efficiency, p_in, net_budget, p_wire): 

99 """``(efficiency, budget rows)`` with the engine's conductor loss folded in. 

100 

101 The network's input power and losses come from the reducer, which the 

102 structure deck cannot see; the wire loss comes from the deck, which the 

103 reducer cannot see. LOSSES ONLY in the rows (issue #1354).""" 

104 if p_wire > 0.0 and p_in > 0.0: 

105 efficiency = max(0.0, min(1.0, efficiency - p_wire / p_in)) 

106 return efficiency, list(net_budget) + [("Wire loss", p_wire)]

109def source_gain_factor(p_struct, p_source, error_cls): 

110 """P_structure / P_source, the factor turning a NEC gain per STRUCTURE watt 

111 into gain per SOURCE watt (AK#1637); 1.0 when `p_source` is None (the 

112 native route, where the structure's input power already is the source's). 

113 

114 A NEC normalises gain by the power into the structure, the sum over its EX 

115 cards. On this route the network sits between the sources and those cards, 

116 and a lossy one burns its share first.""" 

117 if p_source is None: 

118 return 1.0 

119 if p_source <= 0.0 or p_struct <= 0.0: 

120 raise error_cls( 

121 f"cannot normalise the pattern per source watt: structure " 

122 f"input {p_struct} W, source power {p_source} W" 

123 ) 

124 return p_struct / p_source

127def check_reciprocity(Y, names, rtol, error_cls, engine): 

128 """The worst |Y[i,j] - Y[j,i]| relative to sqrt(|Y[i,i]| |Y[j,j]|), the 

129 ports' own scale; raises `error_cls` past `rtol`. 

130 

131 Y[i, j] and Y[j, i] come from DIFFERENT runs, so nothing about the 

132 arithmetic makes them agree by construction, and a reading put into the 

133 wrong port breaks the symmetry. Agreement is not evidence the entries are 

134 right, though: an error that is itself symmetric passes (AK#1629).""" 

135 n = len(names) 

136 worst = (0.0, None) 

137 for i in range(n): 

138 for j in range(i + 1, n): 

139 scale = np.sqrt(abs(Y[i, i]) * abs(Y[j, j])) 

140 if scale <= 0: 

141 continue 

142 rel = abs(Y[i, j] - Y[j, i]) / scale 

143 if rel > worst[0]: 

144 worst = (rel, (names[i], names[j])) 

145 if worst[0] > rtol: 

146 a, b = worst[1] 

147 raise error_cls( 

148 f"the multiport Y is not reciprocal: ports {a!r} and {b!r} " 

149 f"disagree by {worst[0]:.3e} of the ports' own admittance, over " 

150 f"the {rtol:g} this route allows. Y[i,j] and Y[j,i] come from " 

151 f"different runs and every entry is {engine}'s own reported " 

152 "current, so this is the port-to-row bookkeeping in " 

153 "`_compute_y_matrix`, not the network." 

154 ) 

155 return worst[0]
