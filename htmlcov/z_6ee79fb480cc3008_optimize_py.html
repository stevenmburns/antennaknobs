1"""Knob optimisation: vary a chosen subset of params within user-set bounds to 

2optimise an electrical objective (impedance match / SWR, resonance, …). 

3 

4Deliberately free of any web framework. The objective is evaluated through an 

5injected ``solve_fn(req) -> response`` callback, so the same code runs: 

6 - under the FastAPI ``/optimize`` endpoint, wired to a registry example's 

7 cheap impedance-only ``momwire_solve`` (no far-field — we only read Z), and 

8 - under unit tests, wired to a builder/example solve directly or to a stub. 

9 

10The optimiser is a bounded Nelder–Mead (derivative-free — each objective eval is 

11a full MoM solve, so finite-difference gradients would be wasteful) started from 

12the params' current values. It's a *local* search: it refines the operating 

13point the user is already near, which matches the "nudge these knobs to tune 

14this" workflow. A global pass (differential_evolution) can be layered on later. 

15 

16Request shape (the bits this module reads), in addition to a normal solve req: 

17 optimize = { 

18 "free": [{"name": "length_factor", "min": 0.9, "max": 1.1}, ...], 

19 "objective": "swr" | "resonance" | "match_z0", 

20 "max_evals": <int, optional cap>, 

21 } 

22"""

36def _swr(z_re: float, z_im: float, z0: float) -> float: 

37 """Voltage SWR of impedance Z against a real reference Z0. 1.0 = perfect 

38 match; clamped just below the open-circuit singularity so a totally 

39 mismatched candidate returns a large-but-finite penalty the optimiser can 

40 still descend from.""" 

41 z = complex(z_re, z_im) 

42 denom = z + z0 

43 gamma = 1.0 if denom == 0 else abs((z - z0) / denom) 

44 gamma = min(gamma, 1.0 - 1e-9) 

45 return (1.0 + gamma) / (1.0 - gamma)

48def _feed_zs(out: dict) -> list[complex]: 

49 """The impedances a solve response puts up for scoring, one per feed. 

50 

51 Bare multi-feed designs (several independently driven ports, no common 

52 network) carry a per-feed table in ``feeds`` — every port needs a match, 

53 so every entry is scored. Everything else carries only ``z_in_re``/ 

54 ``z_in_im``: single-feed designs trivially, and networked designs whose 

55 one source drives the elements through a splitter/harness — there the 

56 meaningful match is the network's driven plane, which is exactly what 

57 ``z_in`` is referenced to (issues #652/#785).""" 

58 feeds = out.get("feeds") 

59 if feeds: 

60 return [complex(float(f["z_re"]), float(f["z_im"])) for f in feeds] 

61 return [complex(float(out["z_in_re"]), float(out["z_in_im"]))]

64def _objective_value(out: dict, key: str) -> float: 

65 """The scalar to minimise. Multi-feed designs score their WORST feed 

66 (minimax, issue #785): a sum lets one badly mismatched element hide behind 

67 several good ones, and the guarantee a driven array actually wants is "no 

68 element is badly matched". The CLI optimiser aggregates the same way; the 

69 two still differ in *form* (CLI default: |Z−Z0| distance, web default: 

70 SWR) — deliberate, each matches what its surface displays.""" 

71 z0 = float(out.get("z0_ohms", 50.0)) 

72 zs = _feed_zs(out) 

73 if key == "resonance": 

74 return max(abs(z.imag) for z in zs) # cancel reactance 

75 if key == "match_z0": 

76 return max(abs(z - z0) for z in zs) # complex distance to Z0 

77 return max(_swr(z.real, z.imag, z0) for z in zs) # default: minimise SWR

80# Root-finding tolerances. `_ROOT_XTOL` is deliberately the SAME 1e-4 that the 

81# Nelder-Mead call below passes as `xatol`, so switching methods does not 

82# silently change how tightly a knob is resolved; `_ROOT_FTOL` is the residual 

83# at which a root is called found. Neither is the study's 1 ohm reporting 

84# tolerance (#1202) -- that is how the gates *score* a run, not where a run 

85# stops. A secant costs one solve per step and converges superlinearly, so 

86# tightening it past the reporting tolerance is nearly free. 

87_ROOT_XTOL = 1e-4 # knob units 

88_ROOT_FTOL = 1e-3 # ohm

91def _signed_reactance(out: dict) -> float | None: 

92 """The SIGNED X a scalar root-finder needs, or None if there isn't one. 

93 

94 `_objective_value(.., "resonance")` is `max(abs(X))` over the feeds: an 

95 absolute value, and on a multi-feed design a minimax. Neither is a function 

96 with a sign change to bracket -- |X| touches zero without crossing it, and 

97 a minimax of several feeds generally has no common root at all. So the 

98 root-finding path is single-feed only, and this returns None to say so. 

99 """ 

100 zs = _feed_zs(out) 

101 if len(zs) != 1: 

102 return None 

103 return zs[0].imag

106def _secant_root(probe, x0, lo, hi, budget, *, span_frac=0.05): 

107 """Secant on a scalar residual over [lo, hi]. Returns (x, ok, reason). 

108 

109 Costs ONE new solve per step: the previous iterate is reused, where a 

110 Newton finite-difference derivative would pay two. Measured on #1202: 3-4 

111 solves against Nelder-Mead's 10-16. 

112 

113 Bails out (ok=False) when the step leaves the box or the residual grows on 

114 two consecutive steps -- the two ways a secant goes wrong on a curve that 

115 is not locally monotonic. The caller brackets instead. 

116 """ 

117 span = max((hi - lo) * span_frac, _ROOT_XTOL) 

118 a = min(max(float(x0), lo), hi) 

119 fa = probe(a) 

120 if fa is None: 

121 return a, False, "multi-feed" 

122 if abs(fa) <= _ROOT_FTOL: 

123 return a, True, "already-at-root" 

124 # One bracketing step, taken DOWNHILL in |f| where the box allows it. 

125 b = a + span if a + span <= hi else a - span 

126 b = min(max(b, lo), hi) 

127 fb = probe(b) 

128 if fb is None: 

129 return a, False, "multi-feed" 

130 grew = 0 

131 best = (abs(fa), a) if abs(fa) < abs(fb) else (abs(fb), b) 

132 while budget > 0: 

133 budget -= 1 

134 if fb == fa: 

135 return best[1], False, "flat" 

136 c = b - fb * (b - a) / (fb - fa) 

137 if not (lo - 1e-12 <= c <= hi + 1e-12): 

138 return best[1], False, "left-the-box" 

139 c = min(max(c, lo), hi) 

140 if abs(c - b) <= _ROOT_XTOL: 

141 return c, True, "xtol" 

142 fc = probe(c) 

143 if fc is None: 

144 return best[1], False, "multi-feed" 

145 if abs(fc) < best[0]: 

146 best = (abs(fc), c) 

147 grew = grew + 1 if abs(fc) >= abs(fb) else 0 

148 if grew >= 2: 

149 return best[1], False, "residual-grew" 

150 a, fa, b, fb = b, fb, c, fc 

151 if abs(fc) <= _ROOT_FTOL: 

152 return c, True, "ftol" 

153 return best[1], False, "budget"

156def _surrogate_root_start(probe, box, n, rng, z0): 

157 """Sample the box, fit Z, and return where the ROOT is predicted to be, 

158 plus the sampled points ranked by how close they actually came. 

159 

160 Deliberately NOT `_surrogate_seed`, and the difference is the whole point. 

161 That one returns its best SOLVED sample, which is right for a Nelder-Mead 

162 finisher: NM needs a good incumbent and cannot use a prediction. 

163 

164 A Newton finisher needs something else -- the right BASIN. Measured on the 

165 buried vertical (#1202): R along the X = 0 contour rises to 50.7 at 

166 radial_factor 0.31, falls to 45.1 at 0.54, then saturates at 48.5. The 

167 best-scoring SAMPLE landed at 0.65, past that ridge, where R can never 

168 reach 50 and Newton walks into the upper bound. The fitted surface's 

169 predicted crossing landed at 0.43, on the correct side, and Newton 

170 converged from it in 8 solves. 

171 

172 So the prediction is a legitimate START (it gets solved on the first step), 

173 while the ANSWER stays a solved point -- the abort rule is untouched. 

174 """ 

175 d = len(box) 

176 lo = np.array([b[0] for b in box], dtype=float) 

177 span = np.array([b[1] - b[0] for b in box], dtype=float) 

178 us, zs = [], [] 

179 for u in _seed_points(d, n, rng): 

180 z = probe(lo + np.clip(u, 0.0, 1.0) * span) 

181 if z is None: 

182 return None, [] 

183 us.append(np.asarray(u, dtype=float)) 

184 zs.append(z) 

185 ranked = [ 

186 list(lo + u * span) 

187 for u, _ in sorted(zip(us, zs, strict=True), key=lambda t: abs(t[1] - z0)) 

188 ] 

189 A = _quad_features(np.asarray(us)) 

190 if A.shape[0] < A.shape[1]: 

191 return (ranked[0] if ranked else None), ranked 

192 re = np.linalg.lstsq(A, np.array([z.real for z in zs]), rcond=None)[0] 

193 im = np.linalg.lstsq(A, np.array([z.imag for z in zs]), rcond=None)[0] 

194 cand = rng.random((4096, d)) 

195 F = _quad_features(cand) 

196 pred = np.hypot(F @ re - z0, F @ im) 

197 u_pred = cand[int(np.argmin(pred))] 

198 return list(lo + u_pred * span), ranked

201def _newton_root2(probe, x0, box, budget, *, broyden_between=True): 

202 """Newton on the two-component residual F = (R - R0, X). Returns (x, ok, why). 

203 

204 Three things here are load-bearing, and each was measured in #1202: 

205 

206 **Box-aware finite differences.** A plain forward difference is DEGENERATE 

207 on the boundary: at a box corner every perturbation clips back onto the 

208 corner, J comes out identically zero and the linear solve dies singular. 

209 Measured on moxon from the far start -- Newton stopped after 4 solves with 

210 a residual of 35 ohm. Stepping inward instead turns that into 7 solves. 

211 

212 **A fresh Jacobian costs 2 solves, not 3.** The base point is already 

213 solved and the #1176 memo answers it for free, so only the two 

214 perturbations are new. That is why refreshing is affordable at all. 

215 

216 **The stall detector.** On the buried vertical, R along the X = 0 contour 

217 rises to 50.7 at radial_factor 0.31, falls to 45.1 at 0.54, then saturates 

218 at 48.5 -- so a start past that ridge sends Newton chasing R = 50 into the 

219 upper bound, where the residual CANNOT be zeroed. Unguarded it burned all 

220 80 solves thrashing there. Two non-improving steps buy one Jacobian 

221 refresh; a third hands back to the caller, which falls back to Nelder-Mead 

222 from the best SOLVED point. 

223 """ 

224 n = len(x0) 

225 lo = np.array([b[0] for b in box], dtype=float) 

226 hi = np.array([b[1] for b in box], dtype=float) 

227 h = np.maximum((hi - lo) * 0.002, 1e-12) 

228 used = 0 

229 

230 def clip(x): 

231 return np.minimum(np.maximum(x, lo), hi) 

232 

233 def call(x): 

234 nonlocal used 

235 used += 1 

236 return probe(x) 

237 

238 def jac(x, F): 

239 """Forward differences, flipped INWARD wherever the box says so.""" 

240 J = np.zeros((n, n)) 

241 for j in range(n): 

242 step = h[j] if x[j] + h[j] <= hi[j] else -h[j] 

243 if x[j] + step < lo[j]: 

244 step = h[j] 

245 xp = x.copy() 

246 xp[j] = min(max(xp[j] + step, lo[j]), hi[j]) 

247 dh = xp[j] - x[j] 

248 Fp = call(xp) 

249 if Fp is None: 

250 return None 

251 J[:, j] = (Fp - F) / (dh if dh else h[j]) 

252 return J 

253 

254 x = clip(np.asarray(x0, dtype=float)) 

255 F = call(x) 

256 if F is None: 

257 return list(x), False, "multi-feed" 

258 best = (float(np.linalg.norm(F)), x.copy()) 

259 if best[0] <= _ROOT_FTOL: 

260 return list(x), True, "already-at-root" 

261 J = jac(x, F) 

262 if J is None: 

263 return list(best[1]), False, "multi-feed" 

264 stall = 0 

265 refreshed = False 

266 while used < budget: 

267 try: 

268 step = np.linalg.solve(J, -F) 

269 except np.linalg.LinAlgError: 

270 return list(best[1]), False, "singular" 

271 t = 1.0 

272 xn, Fn = None, None 

273 while used < budget: 

274 xt = clip(x + t * step) 

275 Ft = call(xt) 

276 if Ft is None: 

277 return list(best[1]), False, "multi-feed" 

278 if np.linalg.norm(Ft) < np.linalg.norm(F) or t < 0.06: 

279 xn, Fn = xt, Ft 

280 break 

281 t *= 0.5 

282 if xn is None: 

283 return list(best[1]), False, "budget" 

284 prev = float(np.linalg.norm(F)) 

285 nrm = float(np.linalg.norm(Fn)) 

286 if nrm < best[0]: 

287 best = (nrm, xn.copy()) 

288 moved = float(np.linalg.norm(xn - x)) 

289 dx, dF = xn - x, Fn - F 

290 x, F = xn, Fn 

291 if nrm <= _ROOT_FTOL: 

292 return list(x), True, "ftol" 

293 if moved <= _ROOT_XTOL: 

294 # The step went to zero. If the residual is still above tolerance 

295 # this is a stationary point of |F| that is NOT a root -- on a real 

296 # deck, the two contours not crossing anywhere reachable. Name it 

297 # so, rather than reporting the convergence test that caught it. 

298 done = best[0] <= _ROOT_FTOL 

299 return list(best[1]), done, "xtol" if done else "no-crossing" 

300 # A step that does not cut the residual by at least 1 % is not 

301 # progress. Two of those in a row is the stall. 

302 stall = 0 if nrm < 0.99 * prev else stall + 1 

303 if stall >= 2: 

304 if refreshed: 

305 # Two more non-improving steps AFTER a refresh: the residual 

306 # cannot be zeroed from here (typically a bound). Hand back. 

307 return list(best[1]), False, "stalled" 

308 refreshed = True 

309 stall = 0 

310 Jn = jac(x, F) 

311 if Jn is None: 

312 return list(best[1]), False, "multi-feed" 

313 J = Jn 

314 continue 

315 if broyden_between and np.dot(dx, dx) > 0: 

316 J = J + np.outer(dF - J @ dx, dx) / np.dot(dx, dx) 

317 else: 

318 Jn = jac(x, F) 

319 if Jn is None: 

320 return list(best[1]), False, "multi-feed" 

321 J = Jn 

322 return list(best[1]), best[0] <= _ROOT_FTOL, "budget"

325def _bracket_brent(probe, lo, hi, budget, *, n_scan=5): 

326 """Scan for a sign change, then secant-with-bisection-fallback inside it. 

327 

328 This is the path for a start the secant could not use. If the scan finds NO 

329 sign change, the residual has no root in the box -- reported as such 

330 (ok=False, "no-sign-change") rather than parked silently on a bound. 

331 """ 

332 if budget < n_scan: 

333 n_scan = max(2, budget) 

334 vs = list(np.linspace(lo, hi, n_scan)) 

335 fs = [] 

336 for v in vs: 

337 if budget <= 0: 

338 return v, False, "budget" 

339 budget -= 1 

340 f = probe(v) 

341 if f is None: 

342 return v, False, "multi-feed" 

343 fs.append(f) 

344 best = min(zip(vs, fs, strict=True), key=lambda t: abs(t[1])) 

345 br = None 

346 for i in range(len(vs) - 1): 

347 if fs[i] == 0.0: 

348 return vs[i], True, "scan-hit" 

349 if fs[i] * fs[i + 1] < 0: 

350 br = (vs[i], fs[i], vs[i + 1], fs[i + 1]) 

351 break 

352 if br is None: 

353 return best[0], False, "no-sign-change" 

354 a, fa, b, fb = br 

355 while budget > 0: 

356 budget -= 1 

357 c = b - fb * (b - a) / (fb - fa) if fb != fa else 0.5 * (a + b) 

358 if not (min(a, b) < c < max(a, b)): 

359 c = 0.5 * (a + b) 

360 fc = probe(c) 

361 if fc is None: 

362 return best[0], False, "multi-feed" 

363 if abs(fc) < abs(best[1]): 

364 best = (c, fc) 

365 if abs(fc) <= _ROOT_FTOL or abs(b - a) <= _ROOT_XTOL: 

366 return c, True, "ftol" 

367 if fa * fc < 0: 

368 b, fb = c, fc 

369 else: 

370 a, fa = c, fc 

371 return best[0], False, "budget"

374def _residual(out: dict, key: str) -> float | None: 

375 """What a ROOT-finder is driving to zero, or None when the objective is not 

376 a root problem. Distinct from `_objective_value`: the residual falls 

377 monotonically under Newton/secant, which is exactly why the readout shows 

378 it instead of a simplex's best-so-far (#1202).""" 

379 zs = _feed_zs(out) 

380 if len(zs) != 1: 

381 return None 

382 z = zs[0] 

383 if key == "resonance": 

384 return abs(z.imag) 

385 if key == "match_z0": 

386 return abs(z - float(out.get("z0_ohms", 50.0) or 50.0)) 

387 return None

401def _metrics(out: dict) -> dict: 

402 """The handful of numbers the UI shows before/after, derived from one solve. 

403 

404 On a multi-feed design ``swr`` is the WORST feed's — the number the 

405 objective drives (#785) — while ``z_in_re``/``z_in_im`` stay feed 0 to 

406 match the primary readout; ``worst_feed``/``n_feeds`` say which and how 

407 many. Single-feed responses keep the exact four-key shape. 

408 

409 ``feeds`` carries every port's Z (#789) so a live Smith chart can draw the 

410 whole array mid-run. Without it the chart plots ``z_in`` — feed 0 — while 

411 the minimax objective chases some other feed, so the ring can sit still 

412 while the impedance actually being optimised walks off screen. Z only: 

413 the settled solve's feed rows carry position and drive voltage too, and 

414 neither changes during a run.""" 

415 z0 = float(out.get("z0_ohms", 50.0)) 

416 z_re = float(out["z_in_re"]) 

417 z_im = float(out["z_in_im"]) 

418 m = { 

419 "z_in_re": z_re, 

420 "z_in_im": z_im, 

421 "z0_ohms": z0, 

422 "swr": _swr(z_re, z_im, z0), 

423 } 

424 zs = _feed_zs(out) 

425 if len(zs) > 1: 

426 swrs = [_swr(z.real, z.imag, z0) for z in zs] 

427 worst = max(range(len(swrs)), key=swrs.__getitem__) 

428 m["swr"] = swrs[worst] 

429 m["worst_feed"] = worst 

430 m["n_feeds"] = len(zs) 

431 m["feeds"] = [{"z_re": z.real, "z_im": z.imag} for z in zs] 

432 return m

435# --- surrogate seeding (issue #1176) -------------------------------------- 

436# 

437# Nelder-Mead starts from the user's current point and crawls. On two knobs it 

438# does not converge at all inside its own budget — measured 81/82 solves 

439# against a `maxfev` of 80 — so its answer is "as good as 80 evals got". 

440# 

441# The seed fits a cheap response surface to the solves it has already made, 

442# reads a promising point off it, and hands that to Nelder-Mead as the start. 

443# Measured (#1176 design study), against NM alone on the same decks: 

444# 

445# invvee, 2 knobs NM 81 solves -> SWR 1.0000 seeded 44 -> 1.0010 

446# moxon, 2 knobs NM 82 solves -> SWR 1.0000 seeded 44 -> 1.0038 

447# 

448# 1.004 against 1.000 is far below anything an antenna measurement resolves, 

449# at 54 % of the solves. 

450# 

451# FIT THE IMPEDANCE, NOT THE OBJECTIVE. This is the load-bearing decision and 

452# it is the opposite of the obvious implementation. SWR has a kink at the 

453# match and a cliff beyond it, and the catalog's objectives are strongly 

454# ASYMMETRIC about their optimum (moxon's `t0_factor` runs 1.43 -> 2.19 -> 

455# 4.40 -> 6.52 across three steps to the right of its minimum, a second 

456# difference of 32 % of the range). A quadratic fitted to that is biased 

457# toward the gentle side: the first prototype converged in fewer solves to a 

458# DISPLACED optimum, and a local refinement stage did not fix it. Z is smooth 

459# through the same region. Fitting Re(Z) and Im(Z) separately and forming the 

460# objective from the prediction: 

461# 

462# deck fit SWR fit Z 

463# invvee 1.0192 1.0159 

464# moxon 1.6121 1.1106 <- the asymmetric one 

465# 

466# THE ANSWER IS ALWAYS A SOLVED POINT. The surface proposes; it never decides. 

467# Everything returned from here was measured, so an abort mid-seed hands back 

468# the best point actually solved rather than the surface's argmin — which was 

469# measured up to 0.5 SWR away from what that point turned out to be worth. 

470

472def _quad_features(u): 

473 """Columns of the quadratic basis on the unit cube: 1, x_i, x_i x_j.""" 

474 n, d = u.shape 

475 cols = [np.ones(n)] 

476 cols.extend(u[:, i] for i in range(d)) 

477 for i in range(d): 

478 cols.extend(u[:, i] * u[:, j] for j in range(i, d)) 

479 return np.vstack(cols).T

482def _seed_points(d, n, rng): 

483 """A space-filling start: Latin hypercube, one sample per stratum. 

484 

485 Not `rng.random((n, d))`: an unstratified draw leaves gaps and clusters at 

486 these tiny sample counts, and the whole point of the seed is to see the 

487 box before fitting a surface to it. 

488 """ 

489 cuts = (np.arange(n)[:, None] + rng.random((n, d))) / n 

490 for i in range(d): 

491 rng.shuffle(cuts[:, i]) 

492 return cuts

495def _surrogate_seed(probe, d, n_seed, budget, rng): 

496 """Seed a search by fitting Z over the unit cube; return the best SOLVED 

497 point as (u, value). 

498 

499 `probe(u)` solves at unit-cube coordinates and returns 

500 `(objective, z_complex, z0)`. Everything this returns was measured. 

501 """ 

502 us, objs, zs = [], [], [] 

503 

504 def take(u): 

505 obj, z, z0 = probe(u) 

506 us.append(np.asarray(u, dtype=float)) 

507 objs.append(float(obj)) 

508 zs.append((complex(z), float(z0))) 

509 return obj 

510 

511 for u in _seed_points(d, n_seed, rng): 

512 take(u) 

513 

514 while len(us) < budget: 

515 U = np.asarray(us) 

516 A = _quad_features(U) 

517 if A.shape[0] < A.shape[1]: 

518 break 

519 re = np.linalg.lstsq(A, np.array([z.real for z, _ in zs]), rcond=None)[0] 

520 im = np.linalg.lstsq(A, np.array([z.imag for z, _ in zs]), rcond=None)[0] 

521 cand = rng.random((2048, d)) 

522 F = _quad_features(cand) 

523 z0 = zs[0][1] 

524 pred = np.array([_swr(r, i, z0) for r, i in zip(F @ re, F @ im, strict=True)]) 

525 # nearest-neighbour guard: proposing a point we have already solved 

526 # spends an eval to learn nothing (the memo would answer it, but the 

527 # surface would then never move) 

528 for idx in np.argsort(pred): 

529 u = cand[idx] 

530 if all(np.linalg.norm(u - prev) > 1e-3 for prev in us): 

531 take(u) 

532 break 

533 else: 

534 break 

535 

536 best = int(np.argmin(objs)) 

537 return us[best], objs[best]

540def optimize( 

541 base_req: dict, 

542 free: list[dict], 

543 objective: str = "swr", 

544 *, 

545 solve_fn: Callable[[dict], dict], 

546 max_evals: int | None = None, 

547 on_progress: Callable[[dict], None] | None = None, 

548 seed_surrogate: bool = False, 

549 seed_state: int = 0, 

550) -> dict: 

551 """Optimise ``objective`` over the ``free`` params within their bounds. 

552 

553 ``free`` is a list of ``{"name", "min", "max"}``. ``solve_fn`` takes a solve 

554 request and returns a response carrying ``z_in_re``/``z_in_im``/``z0_ohms``. 

555 Returns the best params found plus before/after objective + metrics. 

556 

557 ``on_progress``, if given, is called once per solve (see ``_solve_at``) with 

558 ``{"n_evals", "params", "objective", "metrics"}``. ``None`` (the default) 

559 leaves behaviour identical to no callback support at all. 

560 """ 

561 if not free: 

562 raise ValueError("no free params selected to optimise") 

563 if objective not in OBJECTIVES: 

564 objective = "swr" 

565 

566 names = [f["name"] for f in free] 

567 lo = [float(f["min"]) for f in free] 

568 hi = [float(f["max"]) for f in free] 

569 

570 # Start from each param's current value, clipped into its bound. 

571 x0 = [] 

572 for name, lob, hib in zip(names, lo, hi, strict=True): 

573 cur = float(base_req.get(name, 0.5 * (lob + hib))) 

574 x0.append(min(max(cur, lob), hib)) 

575 

576 n_evals = 0 

577 n_solves = 0 

578 # Which stage is running, for the readout (#1202). A root-finder's residual 

579 # falls monotonically and a simplex's does not, so the readout needs to say 

580 # which it is looking at rather than leaving the user to infer it. 

581 phase = "search" 

582 # Where the run is, for the readout (#1176). `seed_total` is 0 whenever 

583 # the seed is not running, so "am I seeding" is one comparison on the 

584 # client and not a phase machine. 

585 seed_index = 0 

586 seed_total = 0 

587 # One run's solved points, keyed on the EXACT parameter tuple (issue 

588 # #1176). Nelder-Mead keeps only its current simplex, so it re-probes 

589 # points it has already paid for: measured 21-35 % of the solves on a 

590 # one-knob run (invvee 21 %, a vertical over Sommerfeld 35 %, the buried 

591 # screen 25-29 %) and 2-4 % on a two-knob run, where the simplex crawls 

592 # through a tight region without landing twice on the same floats. 

593 # 

594 # THE KEY IS EXACT, deliberately. A tolerance-keyed cache would answer a 

595 # DIFFERENT point with a previous point's solve — faster, and a lie about 

596 # the objective, which would change the search's trajectory and the answer 

597 # it lands on. With exact keys the run is bit-for-bit the run it was, only 

598 # cheaper; the "within xatol" repeats (up to 57 % on two knobs) are 

599 # deliberately left on the table for the surrogate seeding to address, not 

600 # a rounding rule here. 

601 # 

602 # PER RUN, never across requests. The optimizer's evals bypass the server's 

603 # `_SOLVE_CACHE` on purpose — an unbounded budget is a sustained-CPU lever 

604 # (#346) — and this does not reopen that: it is bounded by `max_evals`, 

605 # lives on the stack of one `optimize()` call, and makes a run strictly 

606 # cheaper than the run it replaces. 

607 memo: dict[tuple[float, ...], dict] = {} 

608 

609 def _solve_at(x) -> dict: 

610 nonlocal n_evals, n_solves 

611 req = dict(base_req) 

612 params = {} 

613 for name, v in zip(names, x, strict=True): 

614 val = float(v) 

615 req[name] = val 

616 params[name] = val 

617 n_evals += 1 

618 key = tuple(params[name] for name in names) 

619 out = memo.get(key) 

620 if out is None: 

621 out = solve_fn(req) 

622 n_solves += 1 

623 memo[key] = out 

624 # Emitted here, not via scipy's minimize(callback=...): that callback 

625 # fires once per Nelder-Mead ITERATION (a reflection/expansion/contraction 

626 # that itself costs 1-2 evals), not once per eval — e.g. ~30 callbacks for 

627 # 60 evals — and it doesn't fire at all until the initial simplex (N+1 

628 # evals) is built, leaving a dead zone at the start of every run. Every 

629 # solve, in every phase (baseline, simplex, confirmation), goes through 

630 # this one choke point, so hooking here is the only way to get gapless 

631 # per-eval granularity that also covers the initial-simplex evals. 

632 if on_progress is not None: 

633 on_progress( 

634 { 

635 "n_evals": n_evals, 

636 "params": params, 

637 "objective": _objective_value(out, objective), 

638 "metrics": _metrics(out), 

639 "seed_index": seed_index, 

640 "seed_total": seed_total, 

641 "phase": phase, 

642 "residual": _residual(out, objective), 

643 } 

644 ) 

645 return out 

646 

647 def f(x) -> float: 

648 return _objective_value(_solve_at(x), objective) 

649 

650 out0 = _solve_at(x0) 

651 

652 # --- scalar root path: one knob, X = 0 (#1202) ---------------------- 

653 # `resonance` on a single knob is a SCALAR ROOT, not a minimum, and 

654 # Nelder-Mead never uses that. Measured: 3-4 solves against NM's 10-16 on 

655 # both study decks from both starts. 

656 # 

657 # Guarded on the response being SINGLE-FEED. Multi-feed `resonance` is a 

658 # minimax of |X| over ports, which has no sign change to bracket and 

659 # generally no common root -- `_signed_reactance` returns None and this 

660 # whole branch is skipped, leaving NM exactly as it was. 

661 method = "nelder-mead" 

662 root_reason = None 

663 x_root = None 

664 # Two knobs to Z0 is a two-component root (R - R0, X). Same single-feed 

665 # guard as the scalar path, and the same reason for it. 

666 newton_path = ( 

667 objective == "match_z0" and len(free) == 2 and _residual_vec(out0) is not None 

668 ) 

669 if ( 

670 objective == "resonance" 

671 and len(free) == 1 

672 and _signed_reactance(out0) is not None 

673 ): 

674 lo0, hi0 = lo[0], hi[0] 

675 budget0 = (int(max_evals) if max_evals else min(200, 40 * len(free))) - n_evals 

676 

677 def _probe_x(v): 

678 return _signed_reactance(_solve_at([v])) 

679 

680 phase = "secant" 

681 x_root, ok, root_reason = _secant_root( 

682 _probe_x, x0[0], lo0, hi0, max(budget0, 2) 

683 ) 

684 if not ok and root_reason != "multi-feed": 

685 # The secant's two failure modes (step left the box, residual grew 

686 # twice) both mean "this start is not usable", not "there is no 

687 # root". Scan for a bracket before giving up on the method. 

688 phase = "bracket" 

689 x_root, ok, root_reason = _bracket_brent( 

690 _probe_x, lo0, hi0, max(budget0 - n_evals + 1, 2) 

691 ) 

692 if ok: 

693 x_root = [x_root] 

694 method = "secant" if phase == "secant" else "bracket-brent" 

695 else: 

696 # No root reachable in the box (or the budget ran out). Fall through 

697 # to Nelder-Mead from the best SOLVED point, and say why in the 

698 # result -- never park silently on a bound. 

699 phase = "fallback" 

700 method = f"nelder-mead (root: {root_reason})" 

701 if x_root is not None: 

702 x0 = [min(max(float(x_root), lo0), hi0)] 

703 x_root = None 

704 

705 # --- surrogate seeding (#1176) ------------------------------------- 

706 # Runs BEFORE Nelder-Mead and hands it a start, rather than replacing it. 

707 # Framing it as a peer method would invite picking the one that does not 

708 # finish: measured, the surface alone reaches 1.016 (invvee) / 1.111 

709 # (moxon) where NM reaches 1.0000, and the two together reach 1.001 / 

710 # 1.004 in 54 % of the solves. It is a seeding stage, and the name says so. 

711 # 

712 # One knob is excluded: a 1-D quadratic needs 3 points and NM converges in 

713 # 24 solves there anyway (measured), so the seed would spend an eighth of 

714 # the run to save nothing. The saving is on 2+ knobs, which is also where 

715 # NM runs out of budget rather than converging. 

716 n_seed = 0 

717 if seed_surrogate and len(free) >= 2: 

718 d = len(free) 

719 n_coef = 1 + d + d * (d + 1) // 2 

720 n_seed = n_coef 

721 # 2x the coefficient count: 12 points for two knobs, which is what 

722 # the study measured as the knee — 18 was no better (moxon 1.0080 

723 # against 1.0038), so the seed stays small and leaves the budget to 

724 # the finisher. Never more than a third of the run. 

725 total_budget = int(max_evals) if max_evals else min(200, 40 * len(free)) 

726 seed_budget = max(n_coef, min(2 * n_coef, total_budget // 3)) 

727 phase = "seeding" 

728 lo_a = np.asarray(lo, dtype=float) 

729 span = np.asarray(hi, dtype=float) - lo_a 

730 

731 seed_total = seed_budget 

732 

733 def _probe(u): 

734 nonlocal seed_index 

735 seed_index += 1 

736 out = _solve_at(lo_a + np.clip(np.asarray(u, float), 0.0, 1.0) * span) 

737 return ( 

738 _objective_value(out, objective), 

739 complex(out.get("z_in_re", 0.0), out.get("z_in_im", 0.0)), 

740 float(out.get("z0_ohms", 50.0) or 50.0), 

741 ) 

742 

743 u_best, _obj_best = _surrogate_seed( 

744 _probe, d, n_seed, seed_budget, np.random.default_rng(seed_state) 

745 ) 

746 # The incumbent is a SOLVED point, never the surface's argmin. 

747 x0 = list(lo_a + u_best * span) 

748 seed_index = 0 

749 seed_total = 0 

750 

751 if newton_path: 

752 box2 = list(zip(lo, hi, strict=True)) 

753 total2 = int(max_evals) if max_evals else min(200, 40 * len(free)) 

754 rng2 = np.random.default_rng(seed_state) 

755 

756 def _probe_z(x): 

757 # Drives the seeding readout the same way #1176's does: the survey 

758 # samples the whole box, so its residual jumps around and a bare 

759 # eval number reads as the run going backwards. 

760 nonlocal seed_index 

761 seed_index += 1 

762 zs = _feed_zs(_solve_at(list(x))) 

763 return zs[0] if len(zs) == 1 else None 

764 

765 def _probe_f(x): 

766 return _residual_vec(_solve_at(list(x))) 

767 

768 # The seed runs unconditionally on this path: the study measured that 

769 # bare Newton FAILS on a real deck from the catalogue start, so the 

770 # global sample is part of the method here rather than a toggle. The 

771 # `seed_surrogate` toggle keeps its meaning for the Nelder-Mead path, 

772 # which is the only place it ever applied. 

773 phase = "seeding" 

774 n_coef2 = 1 + 2 + 3 

775 seed_index, seed_total = 0, n_coef2 

776 x_pred, ranked = _surrogate_root_start( 

777 _probe_z, box2, n_coef2, rng2, float(out0.get("z0_ohms", 50.0) or 50.0) 

778 ) 

779 n_seed = n_coef2 

780 seed_index, seed_total = 0, 0 

781 

782 # Try the predicted crossing first, then the best solved samples. A 

783 # restart costs only its own steps -- the samples are already paid for, 

784 # and the memo answers them free. 

785 starts = [x for x in ([x_pred] if x_pred else []) + ranked[:2] if x] 

786 phase = "newton" 

787 xr, ok, root_reason = None, False, "no-start" 

788 for st in starts: 

789 budget_n = total2 - n_evals 

790 if budget_n < 4: 

791 root_reason = "budget" 

792 break 

793 xr, ok, root_reason = _newton_root2(_probe_f, st, box2, budget_n) 

794 if ok: 

795 break 

796 if ok: 

797 x_root, method = xr, "seed + newton" 

798 else: 

799 # Stalled, singular, or out of budget. Hand the best SOLVED point 

800 # to Nelder-Mead and say why -- never present a bound as a root. 

801 phase = "fallback" 

802 method = f"nelder-mead (root: {root_reason})" 

803 if xr is not None: 

804 x0 = [ 

805 min(max(float(v), lob), hib) 

806 for v, lob, hib in zip(xr, lo, hi, strict=True) 

807 ] 

808 

809 # Cap the work: each eval is a full solve. ~40 per free param, hard-capped so 

810 # a wide search can't run away. The UI can override via max_evals. The 

811 # seed's own evals come out of this budget, so a seeded run never costs 

812 # more solves than an unseeded one. 

813 if x_root is not None: 

814 # The root path converged; Nelder-Mead has nothing to add and would 

815 # only spend solves confirming it. 

816 x_best = [ 

817 min(max(float(v), lob), hib) 

818 for v, lob, hib in zip(x_root, lo, hi, strict=True) 

819 ] 

820 else: 

821 phase = "nelder-mead" if phase == "search" else phase 

822 maxfev = max( 

823 1, (int(max_evals) if max_evals else min(200, 40 * len(free))) - n_evals + 1 

824 ) 

825 res = minimize( 

826 f, 

827 x0, 

828 method="Nelder-Mead", 

829 bounds=list(zip(lo, hi, strict=True)), 

830 options={"maxfev": maxfev, "xatol": 1e-4, "fatol": 1e-5}, 

831 ) 

832 

833 # res.x can sit a hair outside bounds after the final reflection; clip. 

834 x_best = [ 

835 min(max(float(v), lob), hib) 

836 for v, lob, hib in zip(res.x, lo, hi, strict=True) 

837 ] 

838 out1 = _solve_at(x_best) 

839 

840 before = _objective_value(out0, objective) 

841 after = _objective_value(out1, objective) 

842 # Only claim the optimum if it actually didn't get worse (Nelder–Mead can 

843 # report success while terminating at the start point for a flat objective). 

844 best_params = ( 

845 dict(zip(names, x_best, strict=True)) 

846 if after <= before 

847 else dict(zip(names, x0, strict=True)) 

848 ) 

849 

850 return { 

851 "objective": objective, 

852 "params": {k: float(v) for k, v in best_params.items()}, 

853 "objective_before": before, 

854 "objective_after": min(before, after), 

855 "metrics_before": _metrics(out0), 

856 "metrics_after": _metrics(out1 if after <= before else out0), 

857 "n_evals": n_evals, 

858 # Objective evaluations vs solver calls: they differ by the memo's 

859 # hits (#1176). `n_evals` keeps its meaning — one per `_solve_at`, so 

860 # the progress stream stays gapless — and this is what the run 

861 # actually cost. 

862 "n_solves": n_solves, 

863 # How many of the evals went to the surrogate seed (#1176); 0 when it 

864 # did not run. The readout distinguishes the phases with this. 

865 "n_seed": n_seed, 

866 # Which path actually ran, and what the root-finder was driving to zero 

867 # (#1202). `residual_after` is None for objectives that are not roots. 

868 "method": method, 

869 "residual_before": _residual(out0, objective), 

870 "residual_after": _residual(out1 if after <= before else out0, objective), 

871 "improved": after < before, 

872 }
