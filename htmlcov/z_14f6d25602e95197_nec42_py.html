1"""NEC-4.2 as an external engine over a text coupling — issue #1603. 

2 

3NEC-4.2 (Burke & Poggio, LLNL, LLNL-CODE-491368) is licensed software: each 

4user holds their own licence and supplies their own console binary, exactly as 

5NEC-5 users do (#825). antennaknobs ships nothing of it — no binary in the 

6workbench bundle, the Docker image, a wheel or the hosted app — and reaches it 

7the way it reaches NEC-2 and NEC-5: a deck written to a fresh temp dir, the 

8binary run there, its printout parsed. 

9 

10A thin subclass of `NEC2Engine`, because NEC-4.2 reads NEC-2's cards and prints 

11NEC-2's blocks. What differs, and where each difference lives: 

12 

13* **Invocation.** ``nec42cl <deck> <printout>``, both names as ARGUMENTS, run 

14 from the deck's directory (`run_deck`). No probing between forms: the 

15 binary this lane targets takes exactly one. 

16* **Ground cards** (`nec_export`'s ``dialect="nec42"``). The Sommerfeld card 

17 caches its tables to files in the cwd unless it ends ``NOFILE``, so every 

18 ``GN 2`` (or 4.2's newer ``GN 3``) is written ``... NOFILE`` — and the run 

19 happens in a temp dir that is removed afterwards regardless, never the 

20 user's cwd. The MININEC-type pair is refused: measured, NEC-4.2 accepts 

21 NEC-2's ``GN 1`` + ``GD`` and prints the perfect-ground pattern to the digit. 

22* **Buried wires are served.** ``GE -1`` is NEC-4's flag for a ground with 

23 wires below it, so a design `refuse_nec2_geometry` turns away runs here over 

24 the Sommerfeld ground (`refuse_nec42_geometry` holds what is still refused). 

25* **The power budget** prints ``WIRE LOSS`` where NEC-2 prints ``STRUCTURE 

26 LOSS``, and its ``NETWORK LOSS`` line appears only when the deck carries a 

27 network (`NEC42Engine._parse_power_budget`). 

28 

29Everything else — feeds at segment centres (odd parity), the gyrator idiom for 

30a current source, the multiport-Y route, every other parser — is `NEC2Engine`'s, 

31verified against real ``nec42cl`` printouts. NEC-4.2's native ``EX 6`` current 

32source, graded meshes and a writer of its own are antennaknobs#1803. 

33"""

71def run_deck(exe: str, deck: str, *, timeout: float) -> str: 

72 """One deck through the binary, returning the printout text. 

73 

74 Runs in a FRESH temp dir per call, removed on the way out: NEC-4.2 writes 

75 what it writes into its cwd (the Sommerfeld table files, when a card lacks 

76 NOFILE), and the user's working directory is never that place. Exit codes 

77 are not trusted — the health signal is a printout, as on NEC-2 and NEC-5. 

78 """ 

79 with tempfile.TemporaryDirectory(prefix="nec42_") as td: 

80 tdp = Path(td) 

81 (tdp / "model.nec").write_text(deck) 

82 try: 

83 proc = run_exe( 

84 [exe, "model.nec", "model.out"], stdin_text="", cwd=td, timeout=timeout 

85 ) 

86 except subprocess.TimeoutExpired as e: 

87 raise NEC42Error(f"NEC-4.2 timed out after {timeout:.0f}s") from e 

88 except OSError as e: 

89 raise NEC42Error(f"{exe} could not be executed: {e}") from e 

90 out = tdp / "model.out" 

91 text = out.read_text(errors="replace") if out.is_file() else "" 

92 if not text.strip(): 

93 tail = ((proc.stdout or "") + (proc.stderr or ""))[-300:] 

94 raise NEC42Error( 

95 f"NEC-4.2 produced no printout from {exe} (exit {proc.returncode}); " 

96 f"output tail: {tail!r}" 

97 ) 

98 return text

105def probe_nec42(explicit: str | None = None, *, timeout: float = 20.0) -> str | None: 

106 """The resolved NEC-4.2 executable, or None with the reason logged. 

107 

108 `probe_nec2`'s discipline (#1339): a path that resolves is not evidence the 

109 file is a NEC, so this RUNS NEC-2's one-wire probe deck once per (path, 

110 mtime, size) and requires an ANTENNA INPUT PARAMETERS block. 

111 """ 

112 exe = find_nec42(explicit) 

113 if exe is None: 

114 return None 

115 ident = _identity(exe) 

116 if ident is None: 

117 return None 

118 cached = _PROBE_CACHE.get(exe) 

119 if cached is not None and cached[:2] == ident: 

120 return exe if cached[2] else None 

121 reason = "" 

122 try: 

123 text = run_deck(exe, _PROBE_DECK, timeout=timeout) 

124 ok = _AIP_HEADER in text 

125 if not ok: 

126 reason = ( 

127 f"{exe} ran but produced no {_AIP_HEADER} block; " 

128 f"printout head: {text[:300]!r}" 

129 ) 

130 except NEC2Error as e: 

131 ok, reason = False, f"{exe} is not a working NEC-4.2 binary: {e}" 

132 _PROBE_CACHE[exe] = (ident[0], ident[1], ok, reason) 

133 if not ok: 

134 _log.warning("NEC-4.2 unavailable: %s", reason) 

135 return exe if ok else None

138def refuse_nec42_geometry(tups, ground) -> None: 

139 """Refuse geometry a NEC-4.2 deck from this writer cannot carry, by name. 

140 

141 Buried wires are SERVED — the difference from `refuse_nec2_geometry` — over 

142 the Sommerfeld ground, under ``GE -1``. What stays refused, each measured or 

143 documented rather than guessed: 

144 

145 * a wire crossing z=0 MID-SPAN. Measured on a 6 m vertical from z=-1 m: at 

146 30 segments (a segment end on the interface) the crossing wire and the 

147 same wire split at z=0 print the same impedance to the digit; at 29, one 

148 segment straddles the interface and the binary prints 38-621j ohm for 

149 the 167-125j ohm answer, without a word. Splitting at z=0 is the fix; 

150 * a wire lying IN the plane (both ends z=0), as on NEC-2; 

151 * buried wires under any ground but the Sommerfeld one: PEC and the 

152 reflection-coefficient ground have no below-ground medium; 

153 * a conductor that ENDS on the plane in a deck that also has buried wires. 

154 ``GE -1`` leaves the current expansion alone at z=0, so that end has no 

155 basis function there and the conductor reads as open-circuited — NEC-5's 

156 flag, with NEC-5's measured consequence (`NEC5Engine`, #1025). A 

157 conductor that continues below the plane (a rise from a buried hub) is 

158 served. 

159 """ 

160 if ground is None or ground == "free": 

161 return 

162 wires = [as_wire(t) for t in tups] 

163 buried = False 

164 contacts = [] 

165 for i, w in enumerate(wires): 

166 z0, z1 = float(w.p0[2]), float(w.p1[2]) 

167 if z0 == 0.0 and z1 == 0.0: 

168 raise ValueError( 

169 f"wire {i + 1} lies in the ground plane (z=0), where its own " 

170 "image coincides with it — NEC-4.2's ground forbids it" 

171 ) 

172 if (z0 < 0.0 < z1) or (z1 < 0.0 < z0): 

173 raise NotImplementedError( 

174 f"wire {i + 1} crosses the ground plane mid-span: split it AT " 

175 "z=0. NEC-4.2 serves buried wires, but a segment straddling the " 

176 "interface is solved wrong without a warning (measured), so the " 

177 "wrapper refuses it rather than report that number" 

178 ) 

179 if z0 < 0.0 or z1 < 0.0: 

180 buried = True 

181 if not (isinstance(ground, tuple) and ground[0] == "finite"): 

182 raise NotImplementedError( 

183 f"wire {i + 1} dips below z=0: NEC-4.2 serves buried wires " 

184 "only over its Sommerfeld ground — choose the Sommerfeld " 

185 "(finite) ground, not PEC, the reflection-coefficient or " 

186 "the MININEC-type ground, none of which has a medium below " 

187 "the plane" 

188 ) 

189 for p in (w.p0, w.p1): 

190 if float(p[2]) == 0.0: 

191 contacts.append(tuple(float(c) for c in p)) 

192 if not buried: 

193 return 

194 for node in contacts: 

195 continues_below = any( 

196 (tuple(float(c) for c in end) == node and float(other[2]) < 0.0) 

197 for w in wires 

198 for end, other in ((w.p0, w.p1), (w.p1, w.p0)) 

199 ) 

200 if not continues_below: 

201 x, y, _z = node 

202 raise NotImplementedError( 

203 f"a conductor ends ON the ground plane at ({x:g}, {y:g}, 0) while " 

204 "this design also has buried wires: the ground flag buried wires " 

205 "need (GE -1) gives a wire ending at z=0 no basis function there, " 

206 "so NEC-4.2 would read the conductor as open-circuited. Continue " 

207 "the conductor BELOW the interface (a rise from the buried hub up " 

208 "to this node is the served spelling), or lift it clear of the " 

209 "plane" 

210 )

213class NEC42Engine(NEC2Engine): 

214 """A user-supplied, licensed NEC-4.2 console binary, driven over text. 

215 

216 `NEC2Engine` with NEC-4.2's invocation, ground cards, geometry rules and 

217 power-budget labels (see the module docstring). ``sommerfeld`` picks the 

218 Sommerfeld card for a "finite" ground: 2 (the default, the evaluation 

219 NEC-2 also has) or 3, NEC-4.2's newer one. 

220 """ 

221 

222 label = "NEC-4.2" 

223 _capture_name = "nec42" 

224 

225 def __init__( 

226 self, 

227 builder, 

228 *, 

229 ground=_GROUND_DEFAULT, 

230 nec42_exe: str | None = None, 

231 timeout: float = 120.0, 

232 capture_dir=None, 

233 wire_radius: float | None = None, 

234 sommerfeld: int = 2, 

235 ): 

236 if sommerfeld not in (2, 3): 

237 raise ValueError(f"sommerfeld must be 2 or 3, not {sommerfeld!r}") 

238 self.sommerfeld = int(sommerfeld) 

239 self._nec42_exe = nec42_exe 

240 if isinstance(ground, tuple) and ground and ground[0] == "mininec": 

241 from ..nec_export import MININEC_NEC42_REFUSAL 

242 

243 raise NotImplementedError(MININEC_NEC42_REFUSAL) 

244 super().__init__( 

245 builder, 

246 ground=ground, 

247 timeout=timeout, 

248 capture_dir=capture_dir, 

249 wire_radius=wire_radius, 

250 ) 

251 

252 def _resolve_exe(self, explicit: str | None) -> str: 

253 exe = find_nec42(self._nec42_exe) 

254 if exe is None: 

255 raise NEC42Error( 

256 "no NEC-4.2 binary: set $NEC42_EXE (or [engines] nec42_exe in " 

257 "settings.toml, or pass nec42_exe=) to your licensed NEC-4.2 " 

258 "console executable" 

259 ) 

260 return exe

262 def _check_geometry_against_ground(self) -> None: 

263 refuse_nec42_geometry(self.tups, self.ground) 

264 self._has_buried_wires = self.ground not in (None, "free") and any( 

265 float(as_wire(t).p0[2]) < 0.0 or float(as_wire(t).p1[2]) < 0.0 

266 for t in self.tups 

267 ) 

268 

269 def _deck_dialect(self) -> dict: 

270 return {"dialect": "nec42", "sommerfeld": self.sommerfeld} 

271 

272 def _run_binary(self, deck: str) -> str: 

273 return run_deck(self.exe, deck, timeout=self.timeout)

275 def _ground_bonds_z0(self) -> bool: 

276 # Under GE -1 (buried wires present) a z=0 node is not bonded: it is an 

277 # ordinary junction between the wires meeting there. 

278 return super()._ground_bonds_z0() and not self._has_buried_wires

281 def _parse_power_budget(text: str) -> dict: 

282 """The POWER BUDGET block, in `NEC2Engine._parse_power_budget`'s keys. 

283 

284 NEC-4.2's labels, from its own printouts:: 

285 

286 INPUT POWER = 3.1574E-03 WATTS 

287 RADIATED POWER= 3.1574E-03 WATTS 

288 WIRE LOSS = 0.0000E+00 WATTS 

289 NETWORK LOSS = 1.4211E-14 WATTS (only with a network) 

290 EFFICIENCY = 100.00 PERCENT 

291 

292 ``WIRE LOSS`` where NEC-2 says ``STRUCTURE LOSS``. The ``NETWORK LOSS`` 

293 line is printed only when the deck carries a network (measured: absent 

294 on a plain dipole, present on the gyrator deck), so its absence reads 

295 as 0 W; every other line is required, and a missing one raises, for 

296 NEC-2's reason — a partial budget is how 100 % gets shipped as measured. 

297 """ 

298 try: 

299 chunk = text.split("POWER BUDGET", 1)[1] 

300 except IndexError: 

301 raise NEC42Error("no POWER BUDGET in NEC-4.2 printout") from None 

302 keys = { 

303 "INPUT POWER": "input_w", 

304 "RADIATED POWER": "radiated_w", 

305 "WIRE LOSS": "wire_loss_w", 

306 "STRUCTURE LOSS": "wire_loss_w", 

307 "NETWORK LOSS": "network_loss_w", 

308 "EFFICIENCY": "efficiency_pct", 

309 } 

310 out: dict = {} 

311 for line in chunk.splitlines()[:12]: 

312 for label, key in keys.items(): 

313 if label in line and "=" in line: 

314 out[key] = float(line.split("=", 1)[1].split()[0]) 

315 out.setdefault("network_loss_w", 0.0) 

316 missing = set(keys.values()) - set(out) 

317 if missing: 

318 raise NEC42Error( 

319 f"incomplete POWER BUDGET section: missing {sorted(missing)}, " 

320 f"parsed {out}" 

321 ) 

322 return out
