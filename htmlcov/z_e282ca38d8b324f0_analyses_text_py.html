1"""Reading analyses back from their text, without running it (AK#2045). 

2 

3A shared link carries the analyses a page built as the same stylized Python 

4a design's ``build_analyses()`` returns and `analyses.to_code` prints (``an. 

5Analysis(...)``, ``an.Sweep(...)``, ...), compressed the way an opened deck 

6is (deflate-raw, base64url: `pack` / `unpack`, the browser's 

7``CompressionStream("deflate-raw")``). 

8 

9A link is a stranger's input, and on the hosted app ``exec`` of it would be 

10remote code execution, so this module NEVER executes the text. `read` parses 

11it with `ast` and walks the tree itself, constructing values only through a 

12whitelist of this package's own spec constructors (`CONSTRUCTORS`), from 

13keyword arguments, tuples and lists, dict literals with string keys (a group 

14knob's entries in an ``an.State``) and literal constants (numbers, a unary 

15minus on a number, strings, booleans, None). Every other node is refused BY 

16NAME (``refused: Name 'os'``, ``refused: Lambda``), before anything is 

17built. The constructors run their own checks, so a value the text could not 

18have been written for by hand is refused the same way it would be in a 

19design file. 

20 

21What is not here, by design: 

22 

23- ``an.Metric``: a callable metric names a Python function, and a function 

24 comes from a design or study file, never from a link (the rule 

25 `analyses.from_data` keeps for the workbench's data). 

26- any name but ``an``, any attribute but a whitelisted ``an.<Name>``, any 

27 call but a whitelisted constructor's, and any positional argument beyond 

28 the ones the printed form itself writes (`POSITIONAL`). 

29 

30Bounds come first, so a hostile link costs no more than its own length: 

31`MAX_PACKED_CHARS` on the link parameter, `MAX_TEXT_CHARS` on the text it 

32decompresses to (the decompressor stops at that many bytes, so a small 

33payload cannot expand into a large one), `MAX_DEPTH` on bracket nesting 

34(counted on the token stream before `ast` sees the text) and `MAX_NODES` on 

35the tree. 

36"""

48#: The longest link parameter read: the deck cap's own size (8000 characters 

49#: in ``lib/decks.ts``) with room to spare. Over it is refused unread. 

50MAX_PACKED_CHARS = 16_000

52#: The longest text read, in bytes once decompressed. The three analyses of 

53#: ``dipoles.two_wire_doublet`` print to about 900; this is ~70 times that. 

54MAX_TEXT_CHARS = 64_000

56#: The deepest bracket nesting read. The printed form nests about six deep 

57#: (list, Analysis, Cross, Cell, State, a group's dict). 

58MAX_DEPTH = 24

144def pack(text: str) -> str: 

145 """``text`` as the link carries it: UTF-8, deflate-raw, base64url with 

146 no padding (what ``lib/decks.ts`` writes for a deck).""" 

147 c = zlib.compressobj(9, zlib.DEFLATED, -15) 

148 raw = c.compress(text.encode("utf-8")) + c.flush() 

149 return base64.urlsafe_b64encode(raw).decode("ascii").rstrip("=") 

150

152def unpack(z: str) -> str: 

153 """The text `pack` wrote, refused by name when it is over a bound, does 

154 not decode, or decompresses past `MAX_TEXT_CHARS` (the decompressor is 

155 told to stop there, so a bomb never expands).""" 

156 if not isinstance(z, str) or not z: 

157 raise AnalysesTextError("the link carries no analyses") 

158 if len(z) > MAX_PACKED_CHARS: 

159 raise AnalysesTextError( 

160 f"refused: the analyses parameter is {len(z)} characters, over " 

161 f"the {MAX_PACKED_CHARS} limit" 

162 ) 

163 try: 

164 packed = base64.urlsafe_b64decode(z + "=" * (-len(z) % 4)) 

165 d = zlib.decompressobj(-15) 

166 data = d.decompress(packed, MAX_TEXT_CHARS + 1) 

167 if len(data) > MAX_TEXT_CHARS or d.unconsumed_tail: 

168 raise _too_long() 

169 data += d.flush() 

170 if not d.eof: 

171 raise AnalysesTextError("the link's analyses are cut short") 

172 except AnalysesTextError: 

173 raise 

174 except (ValueError, zlib.error) as exc: 

175 raise AnalysesTextError(f"the link's analyses do not decode ({exc})") from None 

176 if len(data) > MAX_TEXT_CHARS: 

177 raise _too_long() 

178 try: 

179 return data.decode("utf-8") 

180 except UnicodeDecodeError as exc: 

181 raise AnalysesTextError(f"the link's analyses are not UTF-8 ({exc})") from None

193def write(analyses) -> str: 

194 """The text of one or more analyses: a list of their `to_code`, which 

195 `read` reads back to equal values.""" 

196 one = isinstance(analyses, an.Analysis) or an.is_optimize(analyses) 

197 items = (analyses,) if one else tuple(analyses) 

198 body = "".join(f" {an.to_code(a, 4)},\n" for a in items) 

199 return f"[\n{body}]\n"

202def read(text: str) -> tuple: 

203 """The analyses ``text`` writes, without executing it (module docstring): 

204 one ``an.<...>(...)`` expression, or a list or tuple of them, each an 

205 ``an.Analysis`` (or a library call that makes one) or a kept multi-band 

206 run, ``an.Optimize`` (AK#2045: a shared link may carry one, and the 

207 workbench jumps to it as to a kept study). Anything else is an 

208 `AnalysesTextError` naming what was refused.""" 

209 if not isinstance(text, str): 

210 raise AnalysesTextError("the analyses text must be a string") 

211 if len(text) > MAX_TEXT_CHARS: 

212 raise _too_long() 

213 _check_depth(text) 

214 try: 

215 tree = ast.parse(text, mode="eval") 

216 except (SyntaxError, ValueError, RecursionError, MemoryError) as exc: 

217 raise AnalysesTextError(f"the analyses text does not parse ({exc})") from None 

218 n = sum(1 for _ in ast.walk(tree)) 

219 if n > MAX_NODES: 

220 raise _refuse(f"{n} tree nodes, over the {MAX_NODES} limit") 

221 # Walk the whole tree for refusals before constructing anything. 

222 _Check().visit(tree.body) 

223 body = tree.body 

224 nodes = body.elts if isinstance(body, (ast.List, ast.Tuple)) else [body] 

225 out = [] 

226 for node in nodes: 

227 value = _build(node) 

228 if not (isinstance(value, an.Analysis) or an.is_optimize(value)): 

229 raise _refuse( 

230 f"{type(value).__name__} at the top level; the text lists " 

231 "an.Analysis (or an.Optimize) values" 

232 ) 

233 out.append(value) 

234 return tuple(out)

238 """`read` of the text a link's ``analyses=`` parameter carries.""" 

239 return read(unpack(z))

242def _check_depth(text: str) -> None: 

243 """Refuse bracket nesting past `MAX_DEPTH` on the token stream, so the 

244 parser never recurses on a hostile depth (brackets inside a string are 

245 one STRING token and do not count).""" 

246 depth = 0 

247 try: 

248 for tok in tokenize.generate_tokens(io.StringIO(text).readline): 

249 if tok.type != tokenize.OP: 

250 continue 

251 if tok.string in "([{": 

252 depth += 1 

253 if depth > MAX_DEPTH: 

254 raise _refuse(f"nesting deeper than {MAX_DEPTH}") 

255 elif tok.string in ")]}": 

256 depth -= 1 

257 except tokenize.TokenError as exc: 

258 raise AnalysesTextError(f"the analyses text does not parse ({exc})") from None 

259 except (SyntaxError, IndentationError) as exc: 

260 raise AnalysesTextError(f"the analyses text does not parse ({exc})") from None

272 # The dotted text as written (unparsing is printing, not running),

275 return text if len(text) <= 60 else f"{text[:57]}..."

278class _Check(ast.NodeVisitor): 

279 """Every node the text may hold, and a refusal by name for any other. 

280 It visits only what it allows, so nothing below a refused node is 

281 reached.""" 

282 

283 def generic_visit(self, node): 

284 raise _refuse(type(node).__name__) 

285 

286 def visit_Constant(self, node: ast.Constant): 

287 v = node.value 

288 if v is None or isinstance(v, (bool, int, float, str)): 

289 return 

290 raise _refuse(f"Constant of type {type(v).__name__}") 

291 

292 def visit_UnaryOp(self, node: ast.UnaryOp): 

293 if isinstance(node.op, ast.USub) and _number(node.operand): 

294 return 

295 raise _refuse( 

296 f"UnaryOp {type(node.op).__name__}; only a minus on a literal number" 

297 ) 

298 

299 def visit_Tuple(self, node: ast.Tuple): 

300 for e in node.elts: 

301 self.visit(e) 

302 

303 visit_List = visit_Tuple 

304 

305 def visit_Dict(self, node: ast.Dict): 

306 for k, v in zip(node.keys, node.values, strict=True): 

307 if k is None: 

308 raise _refuse("** in a dict") 

309 if not (isinstance(k, ast.Constant) and isinstance(k.value, str)): 

310 raise _refuse(f"dict key {type(k).__name__}; keys are strings") 

311 self.visit(v) 

312 

313 def visit_Name(self, node: ast.Name): 

314 raise _refuse(f"Name {node.id!r}") 

315 

316 def visit_Attribute(self, node: ast.Attribute): 

317 if not (isinstance(node.value, ast.Name) and node.value.id == "an"): 

318 raise _refuse(f"Attribute {_attr_name(node)!r}") 

319 if node.attr not in CONSTANTS: 

320 raise _refuse(f"Attribute {_attr_name(node)!r}") 

321 

322 def visit_Call(self, node: ast.Call): 

323 f = node.func 

324 if not ( 

325 isinstance(f, ast.Attribute) 

326 and isinstance(f.value, ast.Name) 

327 and f.value.id == "an" 

328 ): 

329 what = ( 

330 f"Name {f.id!r}" 

331 if isinstance(f, ast.Name) 

332 else f"Attribute {_attr_name(f)!r}" 

333 if isinstance(f, ast.Attribute) 

334 else type(f).__name__ 

335 ) 

336 raise _refuse(f"call of {what}") 

337 if f.attr in _NOT_FROM_TEXT: 

338 raise _refuse(f"an.{f.attr}: {_NOT_FROM_TEXT[f.attr]}") 

339 if f.attr not in CONSTRUCTORS: 

340 raise _refuse(f"call of Attribute {_attr_name(f)!r}") 

341 allowed = POSITIONAL[f.attr] 

342 for a in node.args: 

343 if isinstance(a, ast.Starred): 

344 raise _refuse(f"Starred argument to an.{f.attr}") 

345 if len(node.args) > allowed: 

346 raise _refuse( 

347 f"positional argument {allowed + 1} to an.{f.attr}, which takes " 

348 f"{allowed} positionally; pass it by keyword" 

349 ) 

350 for kw in node.keywords: 

351 if kw.arg is None: 

352 raise _refuse(f"** argument to an.{f.attr}") 

353 if kw.arg.startswith("_"): 

354 raise _refuse(f"keyword {kw.arg!r} to an.{f.attr}") 

355 for a in node.args: 

356 self.visit(a) 

357 for kw in node.keywords: 

358 self.visit(kw.value) 

359

361def _build(node): 

362 """The value of a node `_Check` passed.""" 

363 if isinstance(node, ast.Constant): 

364 return node.value 

365 if isinstance(node, ast.UnaryOp): 

366 return -node.operand.value 

367 if isinstance(node, (ast.Tuple, ast.List)): 

368 return tuple(_build(e) for e in node.elts) 

369 if isinstance(node, ast.Dict): 

370 return {k.value: _build(v) for k, v in zip(node.keys, node.values, strict=True)} 

371 if isinstance(node, ast.Attribute): 

372 return CONSTANTS[node.attr] 

373 if isinstance(node, ast.Call): 

374 fn, _ = CONSTRUCTORS[node.func.attr] 

375 args = [_build(a) for a in node.args] 

376 kwargs = {} 

377 for kw in node.keywords: 

378 if kw.arg in kwargs: 

379 raise _refuse(f"keyword {kw.arg!r} given twice to an.{node.func.attr}") 

380 kwargs[kw.arg] = _build(kw.value) 

381 try: 

382 return fn(*args, **kwargs) 

383 except (TypeError, ValueError) as exc: 

384 raise AnalysesTextError(f"an.{node.func.attr}: {exc}") from None 

385 raise _refuse(type(node).__name__) # unreachable after _Check
