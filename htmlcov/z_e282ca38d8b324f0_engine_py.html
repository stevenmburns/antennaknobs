39# --------------------------------------------------------------------------- 

40# The duck-typed power-budget protocol every engine stamps, and its ONE rule 

41# --------------------------------------------------------------------------- 

42# 

43# Three attributes, read by `cli.py`'s pattern/schematic commands and by the web 

44# adapter's `_budget_rows`, on whichever engine ran: 

45# 

46# _excited_p_in input power, watts 

47# _excited_efficiency structural efficiency, a fraction 

48# _excited_power_budget [(label, watts), ...] 

49# 

50# **THE ROWS ARE LOSSES.** Nothing else. Every consumer derives the remainder 

51# that reaches the antenna as `p_in - sum(watts)` — `cli.py` prints it as 

52# "antenna (accepted)" and `SolveReadout.tsx` renders the same subtraction — so a 

53# row that is not a loss is subtracted from the input as if it were. 

54# 

55# This was learned the expensive way (issue #1354). The NEC-5 wrapper stamped 

56# ("Radiated", P_rad) beside ("Wire loss", P_loss) because its printout's POWER 

57# BUDGET block lists both, and the NEC-2 wrapper copied it. On a lossless design 

58# that makes P_rad == P_in, so both consumers printed **"antenna (accepted): 

59# 0 mW (0.0%)"** where momwire printed 100 %. The web NEC-5 tab had been showing 

60# that zero on screen; the CLI only started showing it when #1397 first reached 

61# these rows with the subtraction. 

62# 

63# Radiated power is NOT a budget row. It is available as `_excited_efficiency` 

64# (times `_excited_p_in`) and, on the wrappers that parse it from a printout, as 

65# `_excited_p_radiated`. The frontend's own "radiated (incl. ground)" row comes 

66# from the norm check, which is the honest third ledger — a loss list was never 

67# the channel for it.

175# A value within this of a half counts as the half. A piece's length in 

176# segments comes out of floating point: (1 - 0.77) * 150 is 34.49999999999999, 

177# where its mirror image, 0.23 * 150, may be 34.5. Without the tolerance a wire 

178# and its mirror image could get different counts. 

179_HALF_TOL = 1e-9 

180 

181 

182def _nearest_count(segments): 

183 """The count, at least 1, nearest `segments`. A half rounds up, 46.5 -> 47, 

184 to within `_HALF_TOL`, the same convention as `_nearest_odd_count`.""" 

185 return max(1, math.floor(segments + 0.5 + _HALF_TOL)) 

186 

187 

188def _nearest_odd_count(segments): 

189 """The odd count, at least 1, nearest `segments`: its middle is a segment 

190 centre. A tie, an even `segments` exactly between two odd counts (to within 

191 `_HALF_TOL`), takes the larger one, so the rule never depends on which odd 

192 count sits at an even half-index: 2 -> 3, 4 -> 5, 6 -> 7.""" 

193 return max(1, 2 * math.floor((segments - 1) / 2 + 0.5 + _HALF_TOL) + 1)

228def split_spans(n_seg, positions, parity): 

229 """The pieces a wire of `n_seg` segments is cut into so that every port at 

230 `positions` (fractions in [0, 1], distinct) is fed exactly (AK#1511). The 

231 deterministic rule decided on the issue; with h = 1/n_seg: 

232 

233 A port AT a wire end (0.0 or 1.0) is PLACED there, never cut to (AK#1619). 

234 It is already on a site of both families — that is `wire_catalog.on_site`'s 

235 rule — and no cut can put a segment centre at a wire end, so it takes no 

236 part in the geometry below: the cuts are decided by the INTERIOR ports 

237 alone and the end port is recorded on the piece that bounds it, as that 

238 piece's `p0_port` or `p1_port`. Before AK#1619 an end port was fed like any 

239 other, which centred it in a piece running out to the end — 1/16 of the way 

240 up a base-fed vertical on a six-segment wire — whenever a second port on 

241 the wire forced a split at all. 

242 

243 A knot engine (even parity) cuts at every port. Each piece takes the count 

244 nearest length/h, at least 1, a half rounding up (`_nearest_count`), and 

245 port i is fed at the `hi` knot of piece i, which it shares with piece i + 1. 

246 

247 A segment-centre engine (odd parity) gives port i a short piece 

248 [u_i - x_i, u_i + x_i] centred on it, with the odd count nearest 2 x_i / h 

249 (at least 1), so its middle segment's centre is the port. x_i is the 

250 smallest of its limits: 

251 

252 - g/4 for each neighbouring port at gap g; 

253 - for an end port, b/3 for its wire end at distance b. The guard: with 

254 OTHER the smallest of the port's remaining limits (its neighbour's g/4 

255 and, for a lone port, the other end's b'/3), an end with b <= h and 

256 b <= OTHER has the limit b instead, so the piece runs to that end with no 

257 filler. The two ends of a lone port cannot both fire (b <= b'/3 and 

258 b' <= b/3 together need b = 0). 

259 

260 Plain fillers take the gaps, each the count nearest length/h, at least 1, a 

261 half rounding up, and the zero-length filler where the guard fired is 

262 omitted. So a filler between 

263 two ports is at least g/2, an end filler at least 2b/3, and every piece has 

264 a lower bound set by the local geometry: no slivers.""" 

265 n = max(int(n_seg), 1) 

266 u = sorted(float(a) for a in positions) 

267 # The end ports, by their index in `u`, and the interior ports that alone 

268 # decide the cuts (AK#1619). `u` is sorted and its entries are distinct, so 

269 # an end port can only be the first or the last of them. 

270 at_p0 = 0 if u and u[0] == 0.0 else None 

271 at_p1 = len(u) - 1 if u and u[-1] == 1.0 else None 

272 inner = [i for i, x in enumerate(u) if x not in (0.0, 1.0)] 

273 v = [u[i] for i in inner] 

274 k = len(v) 

275 if parity == "even": 

276 # A knot engine cuts at every interior port. An end port needs no cut 

277 # either way — it is already on the wire's own end knot, and cutting 

278 # there duplicates 0.0 or 1.0 in `cuts` and emits a zero-length piece, 

279 # which reaches the geometry layer as a degenerate edge (AK#1605, the 

280 # ground-contact feed). 

281 cuts = [0.0, *v, 1.0] 

282 spans = [ 

283 SplitSpan( 

284 lo, hi, _nearest_count((hi - lo) * n), inner[j] if j < k else None 

285 ) 

286 for j, (lo, hi) in enumerate(itertools.pairwise(cuts)) 

287 ] 

288 return SplitPlan( 

289 spans=_end_ports(spans, at_p0, at_p1), half=(), guard=(False, False) 

290 ) 

291 h = 1.0 / n 

292 half = [] 

293 guard = [False, False] 

294 for i, ui in enumerate(v): 

295 near = [] 

296 if i > 0: 

297 near.append((ui - v[i - 1]) / 4) 

298 if i < k - 1: 

299 near.append((v[i + 1] - ui) / 4) 

300 ends = [] 

301 if i == 0: 

302 ends.append((0, ui)) 

303 if i == k - 1: 

304 ends.append((1, 1.0 - ui)) 

305 limits = list(near) 

306 for side, b in ends: 

307 other = min(near + [b2 / 3 for s2, b2 in ends if s2 != side]) 

308 if b <= h and b <= other: 

309 guard[side] = True 

310 # `b` is the room between the port and that end, reserved so 

311 # the span does not overrun it. A port exactly AT the end no 

312 # longer reaches here — `inner` filtered it out (AK#1619) — but 

313 # one a hair off it still can, and a zero would cap the 

314 # half-width on BOTH sides, since one `half` serves 

315 # `lo = ui - xi` and `hi = ui + xi`, collapsing the span to zero 

316 # width and reaching the geometry layer as a degenerate edge 

317 # (AK#1605). Every b > 0 keeps its old limit, so no existing 

318 # split moves. 

319 if b: 

320 limits.append(b) 

321 else: 

322 limits.append(b / 3) 

323 half.append(min(limits)) 

324 spans = [] 

325 edge = 0.0 

326 for i, (ui, xi) in enumerate(zip(v, half, strict=True)): 

327 lo = 0.0 if i == 0 and guard[0] else ui - xi 

328 hi = 1.0 if i == k - 1 and guard[1] else ui + xi 

329 if not (i == 0 and guard[0]): 

330 spans.append(SplitSpan(edge, lo, _nearest_count((lo - edge) * n), None)) 

331 spans.append(SplitSpan(lo, hi, _nearest_odd_count(2 * xi * n), inner[i])) 

332 edge = hi 

333 if not guard[1]: 

334 spans.append(SplitSpan(edge, 1.0, _nearest_count((1.0 - edge) * n), None)) 

335 # With no interior port at all the loop emitted nothing and the wire is one 

336 # uncut piece, which is the whole plan for a wire whose only ports are at 

337 # its ends. 

338 return SplitPlan( 

339 spans=_end_ports(spans, at_p0, at_p1), 

340 half=tuple(half), 

341 guard=tuple(guard), 

342 )

406def fed_records(wires, network, parity, owners=None): 

407 """`SimulationEngine.fed_segments` over an engine's COERCED wire list and 

408 its network AS MESHED (AK#1456): the legacy ``ex`` wires, then every gap 

409 or end port, each on the wire it names. 

410 

411 A port NEC-5 feeds through a wire split at its ports (AK#1510, AK#1511) 

412 arrives as a `PortAtVertex` at the p1 end of the piece named for it 

413 (``<wire>@<port>``), and `owners` (the authored entry each wire came from) 

414 says the next piece is the same wire. That source is the knot the two 

415 pieces share, so its record is a knot whose two segments are that piece's 

416 last and the next piece's first. A segment-centre engine's split port 

417 arrives as a gap at the middle of its own piece, and reports that piece.""" 

418 from .network import PortAtEnd 

419 

420 wires = [as_wire(t) for t in wires] 

421 site = "knot" if parity == "even" else "centre" 

422 out = [ 

423 _fed_record(None, i, w, site) 

424 for i, w in enumerate(wires) 

425 if w.ex is not None and not isinstance(w.n_seg, GradedSegments) 

426 ] 

427 if network is None: 

428 return out 

429 by_name = {w.name: i for i, w in enumerate(wires) if w.name is not None} 

430 for name, port in network.ports.items(): 

431 if isinstance(port, PortOnWire): 

432 index, where = by_name.get(port_wire(port)), site 

433 elif isinstance(port, (PortAtVertex, PortAtEnd)):

435 else: 

436 continue 

437 if index is None or isinstance(wires[index].n_seg, GradedSegments): 

438 continue

440 if ( 

441 where == "end" 

442 and getattr(port, "end", None) == "p1" 

443 and owners is not None 

444 and len(owners) == len(wires) 

445 and nxt < len(wires) 

446 and owners[nxt] == owners[index] 

447 ): 

448 rec = _fed_record(name, index, wires[index], "knot") 

449 rec["length_after_m"] = _fed_record(None, nxt, wires[nxt], "knot")[ 

450 "length_m" 

451 ] 

452 else: 

453 rec = _fed_record(name, index, wires[index], where) 

454 out.append(rec) 

455 return out

458def _port_positions(builder): 

459 """``({wire name: [(port name, at), ...]}, referenced)`` for the gap ports 

460 of the builder's network (AK#1469): every wire carrying a gap port with an 

461 explicit position, and the set of wire names any port references. 

462 

463 A wire whose ports all sit at the middle is absent: the parity rule already 

464 serves it, and the old counts must not move. No builder (a stub borrowing 

465 the coercion) means no positions.""" 

466 build = getattr(builder, "build_network", None) 

467 net = build() if callable(build) else None 

468 if net is None: 

469 return {}, set() 

470 by_wire: dict = {} 

471 referenced = set() 

472 for name, port in net.ports.items(): 

473 if isinstance(port, PortOnWire): 

474 referenced.add(port_wire(port)) 

475 if not getattr(port, "distributed", False): 

476 by_wire.setdefault(port_wire(port), []).append((name, port_at(port))) 

477 elif isinstance(getattr(port, "wire", None), str): 

478 referenced.add(port.wire) 

479 positions = { 

480 w: ports 

481 for w, ports in by_wire.items() 

482 if any(at is not None for _name, at in ports) 

483 } 

484 return positions, referenced

487def vertex_only_names(network): 

488 """The wire names whose only attachment in `network` is a `PortAtVertex` 

489 (issue #898), as a frozenset: empty for no network. 

490 

491 A vertex source sits at a wire's END knot, which every count provides, so 

492 parity coercion, which exists to put a site at the MIDDLE, has nothing to 

493 do for such a wire and would only change the author's mesh. Any other port 

494 touching a wire (a gap port by the wire it names, an end port by its 

495 `.wire`) keeps the wire under the normal rule. A `PortVirtual`'s name is a 

496 circuit node, not a wire; a coincidental match only costs an unnecessary 

497 parity bump, never a wrong mesh.""" 

498 if network is None: 

499 return frozenset() 

500 vertex = {p.wire for p in network.ports.values() if isinstance(p, PortAtVertex)} 

501 other = set() 

502 for p in network.ports.values(): 

503 if isinstance(p, PortAtVertex): 

504 continue 

505 name = ( 

506 port_wire(p) 

507 if isinstance(p, PortOnWire) 

508 else getattr(p, "name", None) or getattr(p, "wire", None) 

509 ) 

510 if name: 

511 other.add(name) 

512 return frozenset(vertex - other)

515class SimulationEngine(ABC): 

516 supports_far_field: ClassVar[bool] = False 

517 # Engines that demand a specific basis parity override this. The 

518 # geometry loader bumps a FED or NAMED wire's n_seg up to the next 

519 # valid value so the attachment lands on a middle segment (we never 

520 # bump down — n=0 is invalid); unmarked wires keep their exact count 

521 # (issue #450). "any" disables coercion. 

522 segment_parity: ClassVar[SegmentParity] = "any" 

523 # An engine whose gap can only sit on a site of its own grid sets this: a 

524 # wire carrying positioned ports that are not all sites of its own count, 

525 # one port or several, is split so every port sits exactly on one (AK#1511; 

526 # `split_spans`): at the middle of a short piece of its own for a 

527 # segment-centre engine, at the knot two pieces share for a knot engine. 

528 # No engine changes the count to reach a site. The momwire engine sets it

552 @staticmethod 

553 def coerce_n_seg(n_seg: int, parity: SegmentParity) -> int: 

554 # Floor below the parity step. n_seg=0 is invalid for every engine 

555 # (momwire divides edge length by it), and even-parity engines need 

556 # at least 2 segments to host a feed straddling a midpoint. 

557 if parity == "even": 

558 n_seg = max(2, n_seg) 

559 return n_seg + 1 if n_seg % 2 == 1 else n_seg 

560 if parity == "odd": 

561 n_seg = max(1, n_seg) 

562 return n_seg + 1 if n_seg % 2 == 0 else n_seg 

563 return max(1, n_seg)

580 def _network_as_meshed(self, net): 

581 """`net` as this engine meshes it (AK#1510, AK#1511), so every reader 

582 of a split port's position (segment, knot, card, drive point, reducer 

583 attach) feeds the site the split made, never `at` re-applied to a 

584 piece. A shallow copy: the builder's network is untouched. 

585 

586 Each port on a split wire maps to the piece named for it 

587 (`_piece_name`). On a segment-centre engine the port sits at the MIDDLE 

588 of that piece. On a knot engine it is the series source at the piece's 

589 p1 end, the knot it shares with the next piece: the `PortAtVertex` 

590 NEC-5 serves natively (issue #898), whose source and loads sit at that 

591 knot as they would at an interior one. 

592 

593 A port at the WIRE's own end is the exception, and the one AK#1619 

594 fixed: it keeps that end — arclength 0.0 or 1.0 of the piece that 

595 bounds it, which both families place exactly — instead of the middle or 

596 the shared knot, neither of which is where the design put it.""" 

597 pieces = getattr(self, "_split_ports", None) 

598 at_site = getattr(self, "_split_port_at", None) or {} 

599 if net is None or not pieces: 

600 return net 

601 

602 def meshed_port(name, port): 

603 piece = pieces.get(name) 

604 if piece is None: 

605 return port 

606 frac = at_site.get(name) 

607 if frac is not None: 

608 # A port at the WIRE's own end keeps the POSITIONED spelling on 

609 # BOTH families (AK#1619). A knot engine's `PortAtVertex` is a 

610 # series EMF at the knot two pieces SHARE; a wire's outer end 

611 # is no such knot, and asking momwire for a node gap there is 

612 # refused by name — "wire 0 'start' is not a member of any 

613 # junction group ... for a feed inside a wire use feeds=". It 

614 # is also what this port is spelled as when nothing splits the 

615 # wire, and what `momwire.eznec.serve` hands the solver for 

616 # every network end (AK#1617). 

617 return replace(port, wire=piece, at=frac) 

618 if self.segment_parity == "even": 

619 return PortAtVertex(wire=piece, end="p1") 

620 return replace(port, wire=piece, at=None) 

621 

622 meshed = copy.copy(net) 

623 meshed.ports = {name: meshed_port(name, p) for name, p in net.ports.items()} 

624 return meshed

626 def _authored_currents(self, currents): 

627 """One `WireCurrents` per authored ``build_wires()`` entry (AK#1510). 

628 Every piece of a split wire joins back into that wire, in order, so a 

629 caller indexing currents by the design's own wires finds each one where 

630 the design wrote it. The knot at each cut takes the mean of the two 

631 pieces' end currents, the rule every interior knot follows.""" 

632 owners = getattr(self, "_tup_authored", None) 

633 if not getattr(self, "_split_wires", None) or owners is None: 

634 return currents 

635 out, last = [], None 

636 for owner, wc in zip(owners, currents, strict=True): 

637 if owner == last: 

638 prev = out[-1] 

639 joint = 0.5 * (prev.knot_currents[-1] + wc.knot_currents[0]) 

640 out[-1] = WireCurrents( 

641 knot_positions=np.concatenate( 

642 [prev.knot_positions, wc.knot_positions[1:]] 

643 ), 

644 knot_currents=np.concatenate( 

645 [prev.knot_currents[:-1], [joint], wc.knot_currents[1:]] 

646 ), 

647 ) 

648 else: 

649 out.append(wc) 

650 last = owner 

651 return out

653 def fed_segments(self): 

654 """The segment each feed and port sits on, as THIS engine meshes it 

655 (AK#1456): ``[{"port", "wire", "segments", "length_m", "site"}]``. 

656 

657 Engines differ here by construction. The parity coercion puts a delta 

658 gap mid-segment on an odd count (momwire's bs2, PyNEC, NEC-2) and a 

659 NEC-5 source on the centre knot of an even count, so the house 50 mm 

660 gap wire is one 50 mm segment on one engine and two 25 mm segments on 

661 the other. A near-open driving point is sensitive to that size, so a 

662 row comparing engines records both. 

663 

664 ``site`` is ``"centre"`` (a gap in the middle of a segment 

665 ``length_m`` long), ``"knot"`` (a source between two such segments) 

666 or ``"end"`` (a port at the wire's end). ``port`` is the network 

667 port's name, or None for a legacy ``ex`` feed; ``wire`` indexes the 

668 engine's coerced wire list, or is None where the engine keeps none. 

669 A knot record also carries ``length_after_m``, the segment on the 

670 knot's far side. It equals ``length_m`` except where a wire is split at 

671 its ports (AK#1510, AK#1511) and the source is the knot two pieces 

672 share. Read 

673 from the engine's own coerced wires and its network as meshed, so it 

674 reports what the engine solves, and it needs no solve.""" 

675 tups = getattr(self, "tups", None) 

676 if tups is None: 

677 return [] 

678 return fed_records( 

679 tups, 

680 self._network_as_meshed(_builder_network(self.builder)), 

681 self.segment_parity, 

682 getattr(self, "_tup_authored", None), 

683 )

685 def _split_wire(self, t, at_here, parity, taken): 

686 """Wire entry `t` cut by `split_spans` so every port in `at_here` 

687 (``[(port name, at)]``, None the middle) is fed exactly (AK#1511), 

688 recording the split on `_split_wires` and `_split_ports`. Returns the 

689 pieces in p0 -> p1 order. 

690 

691 Every piece keeps the wire's orientation and spec, and neighbours meet 

692 at the same point. The piece a port feeds is named for it 

693 (`_piece_name`), so the meshed network can point that port at it; 

694 fillers are unnamed. A legacy ``ex`` on the wire rides on the first fed 

695 piece: a positioned port only exists in a `build_network()` design, 

696 where PyNEC and NEC-2 ignore ``ex`` and NEC-5 refuses the mix, and the 

697 refusal must still see it.""" 

698 wire = as_wire(t) 

699 refuse_coincident_ports(wire.name, at_here) 

700 middle = [(port, 0.5 if at is None else float(at)) for port, at in at_here] 

701 ordered = tuple(sorted(middle, key=lambda p: p[1])) 

702 plan = split_spans(wire.n_seg, [at for _port, at in ordered], parity) 

703 # One name per NAMED piece, and the site each port takes on it: the 

704 # middle (None) for a port the piece is cut around, 0.0 or 1.0 for one 

705 # at the wire's own end (AK#1619). A piece can carry an end port and a 

706 # port of its own, so it is named once, for the first port along it, 

707 # and both ports point at that one name. 

708 piece_names = [None] * len(plan.spans) 

709 site = {} 

710 for j, span in enumerate(plan.spans): 

711 on_it = [ 

712 (i, frac) 

713 for i, frac in ( 

714 (span.p0_port, 0.0), 

715 (span.port, None), 

716 (span.p1_port, 1.0), 

717 ) 

718 if i is not None 

719 ] 

720 if not on_it: 

721 continue 

722 if parity == "even" and span.port is not None and len(on_it) > 1: 

723 # A knot engine spells a port at a shared knot as a series 

724 # `PortAtVertex`, and a piece cannot carry one of those AND a 

725 # gap port — `validate_named_wires_referenced` refuses the pair 

726 # by name (issues #579, #898). An end port has nowhere else to 

727 # go, so the port sharing its piece takes the POSITIONED 

728 # spelling instead, at the same knot: `momwire.eznec.serve` 

729 # spells EVERY network end that way, and AK#1617 already moved 

730 # a crowded junction onto it. 

731 on_it = [(i, 1.0 if i == span.port else frac) for i, frac in on_it] 

732 piece_names[j] = _piece_name(wire.name, ordered[on_it[0][0]][0], taken) 

733 for i, frac in on_it: 

734 site[ordered[i][0]] = (piece_names[j], frac) 

735 names = tuple(site[port][0] for port, _at in ordered) 

736 

737 def point(frac): 

738 if frac == 0.0: 

739 return wire.p0 

740 if frac == 1.0: 

741 return wire.p1 

742 return tuple( 

743 float(a + frac * (b - a)) for a, b in zip(wire.p0, wire.p1, strict=True) 

744 ) 

745 

746 pieces = [] 

747 ex = wire.ex 

748 for span, name in zip(plan.spans, piece_names, strict=True): 

749 here, ex = (ex, None) if name is not None else (None, ex) 

750 p0, p1 = point(span.lo), point(span.hi) 

751 if isinstance(t, Wire) or len(t) == 6: 

752 pieces.append( 

753 wire._replace(p0=p0, p1=p1, n_seg=span.n_seg, ex=here, name=name) 

754 ) 

755 else: 

756 pieces.append((p0, p1, span.n_seg, here, name)) 

757 self._split_wires[wire.name] = WireSplit( 

758 ports=ordered, pieces=names, plan=plan, n_seg=int(wire.n_seg) 

759 ) 

760 self._split_ports.update( 

761 {port: name for (port, _at), name in zip(ordered, names, strict=True)} 

762 ) 

763 self._split_port_at.update({port: frac for port, (_n, frac) in site.items()}) 

764 _logger.info( 

765 "%s split wire %r into %d pieces so %s sit(s) exactly on its sites", 

766 type(self).__name__, 

767 wire.name, 

768 len(pieces), 

769 ", ".join(f"{port!r} at {at:.6g}" for port, at in ordered), 

770 ) 

771 return pieces

773 def _parity_exempt_names(self): 

774 """Wire names exempt from parity coercion even though marked. 

775 

776 Coercion exists to land a MID-WIRE attachment on a middle segment; 

777 an engine whose port sits at a wire END (NEC-5's EX-at-knot for 

778 `PortAtVertex`, issue #898) overrides this to exempt wires whose 

779 only attachment is such a port — bumping their count would change 

780 the author's mesh for nothing.""" 

781 return frozenset()

783 def _coerce_wire_tuples(self, tups): 

784 """Returns the input tuples with each n_seg bumped to the engine's 

785 required parity. Logs once per distinct (n_in, n_out) shift so a 

786 converge sweep doesn't spam the log per-edge. Reads 

787 self.segment_parity so subclasses can set it per-instance (e.g. 

788 MomwireEngine, where the parity depends on the chosen solver).""" 

789 parity = self.segment_parity 

790 if parity == "any": 

791 return tups 

792 positions, referenced = _port_positions(getattr(self, "builder", None)) 

793 family = {"odd": "centre", "even": "knot"}.get(parity) 

794 splits = getattr(self, "splits_wire_at_feed", False) 

795 # Authored wire name -> WireSplit for each wire split at its ports, port 

796 # name -> the piece that feeds it, and the authored entry each output 

797 # entry came from (AK#1510, AK#1511). 

798 self._split_wires = {} 

799 self._split_ports = {} 

800 self._split_port_at = {} 

801 self._tup_authored = [] 

802 taken = referenced | {as_wire(t).name for t in tups} - {None} 

803 seen = set() 

804 out = [] 

805 for index, t in enumerate(tups): 

806 w = as_wire(t) 

807 if isinstance(w.n_seg, GradedSegments): 

808 # Graded wires are structural (no ex/name — the geometry 

809 # walk enforces it) and carry explicit per-panel counts; 

810 # parity coercion never applies to them. 

811 out.append(t) 

812 self._tup_authored.append(index) 

813 continue 

814 # Parity coercion exists to land an engine attachment on a wire's 

815 # middle segment: a feed's delta gap (odd parity) or an even-parity 

816 # solver's midpoint. Only wires that HOST an attachment — a legacy 

817 # EX (``ex``) or a named network port (``name``: feed/load/TL/NT) — 

818 # need it. Coercing an UNMARKED wire's segment count is gratuitous, 

819 # and for a deck whose author chose an exact mesh (nec_import) on a 

820 # tightly-coupled structure like a capacity hat it changes the 

821 # modeled coupling enough to corrupt the impedance — reactance sign 

822 # flip on the spiral-hat verticals (issue #450). Preserve unmarked 

823 # wires' n_seg verbatim; NEC-2 and every momwire basis accept any 

824 # per-edge count (the count only has to be valid, i.e. ≥ 1). 

825 # Catalog consequence: a native design whose ``segs_for`` lands an 

826 # EVEN count on an unfed wire now keeps it (was silently bumped to 

827 # odd), so its impedance shifts a hair under the odd-parity engines, 

828 # and a knob move can flip an unfed wire's parity mid-sweep — a 

829 # ~0.01 Ω wobble, not a bug. Measured design deltas: 0.00 Ω for 

830 # PyNEC/Sinusoidal/BSpline-d2, 0.05 Ω for the d=1 tent basis. 

831 marked = ( 

832 w.ex is not None or w.name is not None 

833 ) and w.name not in self._parity_exempt_names() 

834 n_new = self.coerce_n_seg(w.n_seg, parity) if marked else max(1, w.n_seg) 

835 # A wire carrying a positioned port (AK#1469) keeps its authored 

836 # count: no parity bump, and no re-count to put a port on a site, 

837 # which moved the impedance whenever some count happened to fit 

838 # (AK#1511, split-always). When every port on it is already a site 

839 # of that count (a segment centre, or a knot), the wire stays whole 

840 # and each port feeds its site. Otherwise an engine that splits cuts 

841 # the wire so every port on it, one or several, sits exactly on a 

842 # site; a knot engine then cuts at every port, on a knot or not. 

843 at_here = positions.get(w.name) if marked else None 

844 if at_here and family is not None: 

845 n_new = max(1, int(w.n_seg)) 

846 on_grid = all(on_site(n_new, at, family) for _name, at in at_here) 

847 if splits and not on_grid: 

848 pieces = self._split_wire(t, at_here, parity, taken) 

849 self._tup_authored += [index] * len(pieces) 

850 out.extend(pieces) 

851 continue 

852 if n_new != w.n_seg and (w.n_seg, n_new) not in seen: 

853 seen.add((w.n_seg, n_new)) 

854 _logger.info( 

855 "%s bumped n_seg=%d → %d for %s parity", 

856 type(self).__name__, 

857 w.n_seg, 

858 n_new, 

859 parity, 

860 ) 

861 # Preserve the entry's original shape: plain tuples stay plain 

862 # (so tests and shape-sensitive callers see what they passed), 

863 # Wire entries keep name and spec. 

864 if isinstance(t, Wire) or len(t) == 6: 

865 out.append(w._replace(n_seg=n_new)) 

866 elif len(t) == 5: 

867 out.append((w.p0, w.p1, n_new, w.ex, w.name)) 

868 else: 

869 out.append((w.p0, w.p1, n_new, w.ex)) 

870 self._tup_authored.append(index) 

871 return out
