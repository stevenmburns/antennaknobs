1"""delta_looparray driven by two TLs from a central virtual driver. 

2 

3Same antenna geometry as the legacy `build_tls` variant (two slanted delta 

4loops spaced along y), driven the same way (two Z0=100 transmission lines from a 

5single driver, lengths set by `twist`). That legacy Builder now lives only as a 

6test oracle at `tests/fixtures/delta_looparray_with_tls.py` (it is not a catalog 

7design). This Builder is the canonical version: it uses the port-based network 

8spec (`build_network()`), so 

9 

10 - there is no dummy `WWW`-`WW` stub wire in `build_wires()`, 

11 - the loop feed edges are named (`"loop1"`, `"loop2"`), 

12 - the central driver is a `PortVirtual` — exists only as a row/column 

13 in the network Y matrix during the nodal reduction. 

14 

15The geometry is authored with the `Cell`/`Placement` hierarchy (the geometry 

16analog of `Composite`/`Instance`): one delta-loop `Cell` is stamped twice, and 

17the formal feed name `"feed"` is renamed to `"loop1"`/`"loop2"` by the 

18placement map — the single source of truth `build_network()` binds its ports 

19to. The two placements are a mirror pair across y = 0, but expressed as pure 

20placement rather than an explicit `ry` on the geometry: loop2 negates both the 

21y-offset (`-del_y`) and the slant. Negating the slant is what keeps it a true 

22reflection at any tilt — a y-mirror conjugates the tilt (`ry . rotX(-s) . ry = 

23rotX(+s)`). Combined with the base loop's own y-reflection symmetry, this 

24reproduces the legacy mirror construction edge-for-edge (including the fed 

25edge) for every `slant_deg`, not just the default 0. 

26 

27MomwireEngine produces the same impedance as the legacy `build_tls` fixture to 

28numerical precision; the showcase for the network-spec API in #65. 

29""" 

30 

31from antennaknobs import ( 

32 AntennaBuilder, 

33 Cell, 

34 Placement, 

35 Transform, 

36 flatten_placements, 

37) 

38from antennaknobs.network import Driven, Network, PortOnWire, PortVirtual, TL, Wire 

39 

40import math 

41from types import MappingProxyType 

42 

43 

44class Builder(AntennaBuilder): 

45 default_params = MappingProxyType( 

46 { 

47 "design_freq": 28.47, 

48 "freq": 28.47, 

49 "base": 7.0, 

50 "length_factor": 1.0664, 

51 "angle_deg": 61.2377, 

52 "slant_deg": 0.0, 

53 "twist": 0.125, 

54 "del_y": 4.0, 

55 } 

56 ) 

57 

58 def build_wires(self): 

59 eps = 0.05 

60 b = self.base 

61 wavelength = self.design_wavelength 

62 driver = wavelength * self.length_factor 

63 angle = math.radians(self.angle_deg) 

64 cos_t = math.cos(angle) 

65 tan_t = math.tan(angle) 

66 

67 def ry(p): 

68 return p[0], -p[1], p[2] 

69 

70 # One delta loop in LOCAL coordinates, symmetric about y = 0: the 

71 # bottom edge T → S is the named feed, the other three edges close the 

72 # loop. y is the top-corner half-width in closed form. 

73 y = (cos_t * (driver - 2 * eps) + 2 * eps) / (2 * (cos_t + 1)) 

74 S = (0, eps, b - (y - eps) * tan_t) 

75 A = (0, y, b) 

76 B, T = ry(A), ry(S) 

77 loop = Cell( 

78 feeds=("feed",), 

79 wires=[ 

80 Wire(S, A), 

81 Wire(A, B), 

82 Wire(B, T), 

83 Wire(T, S, name="feed"), 

84 ], 

85 ) 

86 

87 # Element pose: raise by the base height, apply the slant, shift along 

88 # y. The two loops are a mirror pair across y = 0, expressed as pure 

89 # placement: loop2 negates BOTH the y-offset and the slant. Negating 

90 # the slant is what makes it a true reflection at any tilt — a y-mirror 

91 # conjugates the tilt, ry . rotX(-s) . ry = rotX(+s), so the reflected 

92 # loop is tilted the opposite way. The placement map renames the formal 

93 # "feed" edge to the per-loop actual build_network() binds a port to. 

94 def pose(dy, slant): 

95 return ( 

96 Transform.translate(0, 0, b) 

97 .postmult(Transform.rotX(-slant)) 

98 .postmult(Transform.translate(0, dy, -b)) 

99 ) 

100 

101 return flatten_placements( 

102 [ 

103 Placement( 

104 "loop1", loop, pose(self.del_y, self.slant_deg), feed="loop1" 

105 ), 

106 Placement( 

107 "loop2", loop, pose(-self.del_y, -self.slant_deg), feed="loop2" 

108 ), 

109 ] 

110 ) 

111 

112 def build_network(self): 

113 wavelength = self.design_wavelength 

114 tl_lengths = ( 

115 self.del_y - wavelength * self.twist, 

116 self.del_y + wavelength * self.twist, 

117 ) 

118 return Network( 

119 ports={ 

120 "loop1": PortOnWire("loop1"), 

121 "loop2": PortOnWire("loop2"), 

122 "driver": PortVirtual("driver"), 

123 }, 

124 branches=[ 

125 TL(a="driver", b="loop1", z0=100.0, length=tl_lengths[0]), 

126 TL(a="driver", b="loop2", z0=100.0, length=tl_lengths[1]), 

127 ], 

128 sources=[Driven(port="driver", voltage=1 + 0j)], 

129 )
