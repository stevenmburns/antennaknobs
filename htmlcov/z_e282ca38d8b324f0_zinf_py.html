1"""The extrapolated impedance Z∞ of a refinement ladder (AK#1781). 

2 

3ONE estimator, implemented twice and kept identical: here (the CLI's density 

4study, ``sweep --param nominal_nsegs``, and the ``ladder`` subcommand) and in 

5``web/frontend/src/lib/zinf.ts`` (the workbench's Z-vs-parameter view and 

6Smith chart). The shared vectors in 

7``web/frontend/src/__tests__/fixtures/zinfVectors.json`` are generated from 

8THIS implementation (``scripts/gen_zinf_vectors.py``) and gate both: pytest 

9(``tests/test_zinf_vectors_1781.py``) and vitest (``zinf.test.ts``) each 

10require their language to reproduce every case to 1e-12. A change to the 

11rule below is a change to both files and a regenerated JSON. 

12 

13The rule (``zinf_estimate``), for rungs ``x_k > 0`` strictly increasing (the 

14refinement variable: segments achieved, or a refinement factor) and complex 

15``Z_k``, ``k = 0..m-1``: 

16 

171. ``m < 3`` (or ``x`` not positive and strictly increasing): 

18 ``insufficient``, no estimate. 

192. ``d_k = |Z_{k+1} - Z_k|``. If the last step ``d_{m-2} <= 1e-6 |Z_{m-1}|``: 

20 ``converged``, ``Z∞ = Z_{m-1}``, no p. 

213. Local slopes ``s_j = -(ln d_{j+1} - ln d_j) / (ln x_{j+1} - ln x_j)``, 

22 each difference ``d_j`` placed at its lower rung ``x_j``; defined only 

23 where both d are > 0. 

244. Oscillating: the last two step vectors point in opposite directions, 

25 ``Re(D_{m-2} conj(D_{m-3})) < 0`` with ``D_k = Z_{k+1} - Z_k`` -> rough. 

265. Straight (needs ``m >= 4``): the last two local slopes ``s_a, s_b`` are 

27 finite, ``|s_a - s_b| <= 0.10 |(s_a + s_b) / 2|``, and p (below) is in 

28 ``[0.25, 4]``. 

296. Straight: ``p = -`` the least-squares slope of ``ln d_k`` on ``ln x_k`` 

30 over the last ``min(3, m-1)`` differences; then ``Z = Z∞ + C x^-p`` 

31 fitted by complex least squares through the last 3 rungs -> 

32 ``asymptotic``, with p. 

337. Otherwise (``m == 3``, not straight, oscillating, p out of range): the 

34 theoretical first order, ``Z∞ = Z_{m-1} + (Z_{m-1} - Z_{m-2}) / 

35 (x_{m-1}/x_{m-2} - 1)`` -> ``rough``, no p. 

36 

37One p per ladder, from the complex step: R and X are not extrapolated 

38separately. Plain ``math`` in double precision, no numpy, in the same 

39operation order as the TypeScript, so the two agree to rounding. 

40 

41A reason beside "rough" (``feed_mesh_step``): the common cause is a feed 

42mesh that does not refine with the ladder. A short feed wire's segment 

43count moves in steps of two to keep the source centred (1 -> 3 on an 

44odd-parity basis, 2 -> 4 on an even one), so the fed segment stays fixed 

45for several rungs and then jumps, and each jump kinks the ladder (AK#1767's 

46feed-gap wires). Given the fed segment's length at each rung, the helper 

47finds the step, within the last ``FEED_MESH_WINDOW`` rungs (the ones the 

48straightness test reads), where that length departs most from the ``1/x`` 

49scaling of a uniform refinement, and reports it when the departure exceeds 

50``FEED_MESH_TOL``. It never changes the estimate. 

51"""

53from __future__ import annotations

55import math 

56from dataclasses import dataclass 

57 

58#: Step 2's floor, relative to the finest |Z|. 

59CONVERGED_REL = 1e-6 

60#: Step 5's straightness tolerance on the last two local slopes. 

61STRAIGHT_REL = 0.10 

62#: Step 5's admissible observed order. 

63P_MIN, P_MAX = 0.25, 4.0 

64#: Step 7's assumed order. 

65P_THEORY = 1.0

68@dataclass(frozen=True) 

69class ZInfEstimate: 

70 """``z_inf`` is None only when ``status == "insufficient"``; ``p`` is set 

71 only when ``status == "asymptotic"``.""" 

72 

73 z_inf: complex | None 

74 p: float | None 

75 status: str # "insufficient" | "converged" | "asymptotic" | "rough"

78def _local_slope(d0, d1, x0, x1): 

79 if not (d0 > 0 and d1 > 0): 

80 return None 

81 return -(math.log(d1) - math.log(d0)) / (math.log(x1) - math.log(x0)) 

82 

83 

84def _fit_order(xs, ds): 

85 """``-`` the least-squares slope of ln d on ln x.""" 

86 n = len(xs) 

87 lx = [math.log(x) for x in xs] 

88 ld = [math.log(d) for d in ds] 

89 # Plain loops, not sum(): since Python 3.12 sum() of floats is 

90 # compensated, and the TypeScript twin adds left to right. 

91 sx = 0.0 

92 sd = 0.0 

93 for i in range(n): 

94 sx += lx[i] 

95 for i in range(n): 

96 sd += ld[i] 

97 mx = sx / n 

98 md = sd / n 

99 num = 0.0 

100 den = 0.0 

101 for i in range(n): 

102 num += (lx[i] - mx) * (ld[i] - md) 

103 den += (lx[i] - mx) * (lx[i] - mx) 

104 return -(num / den) 

105

107def _fit_zinf(xs, zs, p): 

108 """Complex least squares for ``Z = Z∞ + C x^-p``; returns Z∞.""" 

109 s0 = 0.0 

110 s1 = 0.0 

111 s2 = 0.0 

112 t0r = 0.0 

113 t0i = 0.0 

114 t1r = 0.0 

115 t1i = 0.0 

116 for x, z in zip(xs, zs, strict=True): 

117 u = math.pow(x, -p) 

118 s0 += 1.0 

119 s1 += u 

120 s2 += u * u 

121 t0r += z.real 

122 t0i += z.imag 

123 t1r += u * z.real 

124 t1i += u * z.imag 

125 det = s0 * s2 - s1 * s1 

126 return complex((s2 * t0r - s1 * t1r) / det, (s2 * t0i - s1 * t1i) / det) 

127 

128 

129def zinf_estimate(x, z) -> ZInfEstimate: 

130 """Z∞ of the ladder ``(x_k, Z_k)``: the rule in this module's docstring.""" 

131 xs = [float(v) for v in x] 

132 zs = [complex(v) for v in z] 

133 m = len(zs) 

134 if m != len(xs): 

135 raise ValueError(f"x and z differ in length ({len(xs)} vs {m})") 

136 ok = m >= 3 and all(v > 0 for v in xs) 

137 ok = ok and all(xs[k + 1] > xs[k] for k in range(m - 1)) 

138 if not ok: 

139 return ZInfEstimate(None, None, "insufficient") 

140 

141 steps = [zs[k + 1] - zs[k] for k in range(m - 1)] 

142 ds = [abs(s) for s in steps] 

143 z_last = zs[m - 1] 

144 if ds[m - 2] <= CONVERGED_REL * abs(z_last): 

145 return ZInfEstimate(z_last, None, "converged") 

146 

147 a, b = steps[m - 2], steps[m - 3] 

148 oscillating = a.real * b.real + a.imag * b.imag < 0 

149 if not oscillating and m >= 4: 

150 s_a = _local_slope(ds[m - 4], ds[m - 3], xs[m - 4], xs[m - 3]) 

151 s_b = _local_slope(ds[m - 3], ds[m - 2], xs[m - 3], xs[m - 2]) 

152 if ( 

153 s_a is not None 

154 and s_b is not None 

155 and math.isfinite(s_a) 

156 and math.isfinite(s_b) 

157 and abs(s_a - s_b) <= STRAIGHT_REL * abs((s_a + s_b) / 2) 

158 ): 

159 n = min(3, m - 1) 

160 p = _fit_order(xs[m - 1 - n : m - 1], ds[m - 1 - n : m - 1]) 

161 if P_MIN <= p <= P_MAX: 

162 return ZInfEstimate( 

163 _fit_zinf(xs[m - 3 :], zs[m - 3 :], p), p, "asymptotic" 

164 ) 

165 

166 # Componentwise, not complex division: Python's complex / float has 

167 # changed algorithm across versions, and the TypeScript is componentwise. 

168 ratio = math.pow(xs[m - 1] / xs[m - 2], P_THEORY) 

169 z_prev = zs[m - 2] 

170 re = z_last.real + (z_last.real - z_prev.real) / (ratio - 1) 

171 im = z_last.imag + (z_last.imag - z_prev.imag) / (ratio - 1) 

172 return ZInfEstimate(complex(re, im), None, "rough")

175#: ``feed_mesh_step``'s window: the last rungs, as the straightness test reads. 

176FEED_MESH_WINDOW = 4 

177#: ``feed_mesh_step``'s threshold: the factor by which one step's fed-segment 

178#: ratio may differ from the ladder's own ratio before it is reported. 

179FEED_MESH_TOL = 1.25

182def feed_mesh_step(x, fed_len) -> int | None: 

183 """Index ``k`` of the step ``x_k -> x_{k+1}`` where the fed segment least 

184 follows the ladder, or None (see the module docstring). 

185 

186 ``fed_len`` is the fed segment's length at each rung. The departure of a 

187 step is ``|ln(len_k / len_{k+1}) - ln(x_{k+1} / x_k)|``: zero when the 

188 fed segment shrinks with the mesh, ``ln 1.4`` when it stays put while x 

189 grows 1.4x, and larger still at a jump. Only steps departing by more 

190 than ``ln FEED_MESH_TOL`` count. A JUMP (the length changed) is preferred 

191 over a step where it stayed put, being the one a user can find in the 

192 mesh; within each kind the first step with the largest departure wins. 

193 None when no step counts, or when the inputs are short, misaligned or 

194 not positive.""" 

195 m = len(x) 

196 if m < 2 or len(fed_len) != m: 

197 return None 

198 tol = math.log(FEED_MESH_TOL) 

199 best_jump, jump_dev = None, tol 

200 best_held, held_dev = None, tol 

201 for k in range(max(0, m - FEED_MESH_WINDOW), m - 1): 

202 x0, x1 = float(x[k]), float(x[k + 1]) 

203 l0, l1 = float(fed_len[k]), float(fed_len[k + 1]) 

204 if not (x0 > 0 and x1 > x0 and l0 > 0 and l1 > 0): 

205 return None 

206 dev = abs(math.log(l0 / l1) - math.log(x1 / x0)) 

207 if _same_length(l0, l1): 

208 if dev > held_dev: 

209 best_held, held_dev = k, dev 

210 elif dev > jump_dev: 

211 best_jump, jump_dev = k, dev 

212 return best_jump if best_jump is not None else best_held

215def _same_length(a, b): 

216 return abs(a - b) <= 1e-9 * max(a, b)

219def describe_feed_mesh(x, fed_len, k: int, x_name: str = "N") -> str: 

220 """The CLI's line for a ``feed_mesh_step`` hit at step ``k``.""" 

221 l0, l1 = fed_len[k] * 1000.0, fed_len[k + 1] * 1000.0 

222 x0, x1 = x[k], x[k + 1] 

223 if _same_length(l0, l1): 

224 where = f"stayed {l0:.1f} mm from {x_name} = {x0:g} to {x1:g}" 

225 else: 

226 where = f"went {l0:.1f} → {l1:.1f} mm between {x_name} = {x0:g} and {x1:g}" 

227 return ( 

228 f"the fed segment {where}: the feed mesh does not refine with the " 

229 "ladder (AK#1767)" 

230 ) 

231 

232 

233def describe(est: ZInfEstimate, digits: int = 3) -> str: 

234 """The CLI's one-line reading: ``Z∞ = R±Xj (p = 0.98, asymptotic)``, 

235 ``(rough: not yet asymptotic, first order assumed)``, ``(converged)``, 

236 or ``Z∞ unavailable (need >= 3 rungs)``.""" 

237 if est.z_inf is None: 

238 return "Z∞ unavailable (need >= 3 rungs)" 

239 z = est.z_inf 

240 head = f"Z∞ = {z.real:.{digits}f}{z.imag:+.{digits}f}j" 

241 if est.status == "asymptotic": 

242 return f"{head} (p = {est.p:.2f}, asymptotic)" 

243 if est.status == "rough": 

244 return f"{head} (rough: not yet asymptotic, first order assumed)" 

245 return f"{head} (converged)"
