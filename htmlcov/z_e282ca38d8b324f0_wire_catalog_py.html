1"""Wire and cable stock: the physical-materials half of the old `network` 

2module (momwire#456 workstream 2, phase A). 

3 

4`network.py` grew two unrelated populations: the circuit spec (ports, 

5branches, sources, `Network`) that is destined to move down into 

6`momwire.networks`, and this — the catalog of physical wire and feedline 

7stock (`WireSpec`/`WIRES`, `Cable`/`CABLES`), the named `build_wires()` 

8entry (`Wire`/`as_wire`), and the named-wire/port cross-check 

9(`validate_named_wires_referenced`). None of it is circuit math; all of it 

10is geometry and engine-contract code that stays in antennaknobs when the 

11circuit half moves. 

12 

13Import compatibility: `antennaknobs.network` re-exports everything here, so 

14existing `from antennaknobs.network import Wire, CABLES, ...` imports keep 

15working unchanged. 

16"""

52@dataclass(frozen=True)

112def wire_from_catalog(name): 

113 """The `WIRES` entry for `name`, with the same unknown-key ergonomics 

114 as `TL.from_cable`.""" 

115 if name not in WIRES: 

116 raise KeyError(f"unknown wire {name!r}; available: {', '.join(sorted(WIRES))}") 

117 return WIRES[name]

120class Wire(NamedTuple): 

121 """One ``build_wires()`` entry, named (issue #388). A drop-in superset 

122 of the plain-tuple contract: a ``Wire`` IS a tuple, so indexing, 

123 unpacking, and the ``t[4]``-style name access keep working, and designs 

124 may freely mix plain 4/5-tuples and ``Wire`` entries in one list. 

125 

126 With every field after the endpoints defaulted, keyword construction 

127 is the recommended brief spelling — ``Wire(a, b)`` is a structural 

128 wire at the design density, ``Wire(t, s, ex=1 + 0j)`` the feed, 

129 ``Wire(ti, to, name="trap_b0")`` a named attachment wire — with no 

130 positional ``None`` placeholders. (Plain tuples stay 4-6 fields; 

131 short plain tuples are rejected because a bare third element would 

132 be ambiguous between a count and an excitation.) 

133 

134 ``n_seg=None`` means "mesh me at the design density" — resolved by 

135 ``AntennaBuilder.auto_mesh`` as part of the stack (nominal_nsegs 

136 segments per design_freq quarter-wave); an integer is honored 

137 verbatim. 

138 

139 ``spec=None`` means "the design default": engines fall back to 

140 ``build_wire_material()``. Precedence, defined once: an explicit 

141 per-wire ``spec`` wins; the web ``wire_radius`` override only moves 

142 the default. Transforms, array placement, and scale knobs never scale 

143 a ``spec`` — it describes the physical wire stock the antenna is 

144 built from, not the geometry. 

145 """ 

146 

147 p0: tuple 

148 p1: tuple 

149 n_seg: int | None = None 

150 ex: complex | None = None 

151 name: str | None = None 

152 spec: WireSpec | None = None 

153

155def as_wire(t) -> Wire: 

156 """Normalize one ``build_wires()`` entry — plain 4/5/6-tuple or 

157 ``Wire`` — to a ``Wire``. This is the single choke point for 

158 tuple-shape discrimination: consumers should call this instead of 

159 inspecting ``len(t)``, because a ``Wire``'s defaults make its ``len()`` 

160 always 6, which an arity check misreads.""" 

161 if isinstance(t, Wire): 

162 return t 

163 if not 4 <= len(t) <= 6: 

164 raise ValueError( 

165 "build_wires() entry must have 4-6 fields " 

166 f"(p0, p1, n_seg, ex[, name[, spec]]), got {len(t)}" 

167 ) 

168 return Wire(*t)

171def validate_named_wires_referenced(named_wires, network): 

172 """Reject named ``build_wires()`` wires that no `PortOnWire` references 

173 (issue #578). 

174 

175 The geometry layer registers EVERY named wire as a port edge; a port row 

176 the network never references is left electrically open, and an open 

177 series gap CUTS the current path at that wire. The failure is silent and 

178 masquerades as plausible physics (the solve succeeds with the structure 

179 partitioned), so it is an init-time error. A wire named purely for 

180 documentation is almost certainly this bug; a deliberately-open gap 

181 should be an explicit `Load`/branch on a referenced port. 

182 

183 Engines call this once the flattened network and the final wire list are 

184 both known (composite expansion has already namespaced port names). 

185 """ 

186 # The port types live in the circuit module, which imports THIS module 

187 # at scope for the compatibility re-exports — importing back at call 

188 # time keeps the pair acyclic (the same bridge `touchstone._inv` uses). 

189 # It dissolves in momwire#456 ws2 phase B, when the circuit half moves 

190 # to momwire.networks and this becomes an ordinary top-level import. 

191 from .network import PortAtEnd, PortAtVertex, PortOnWire 

192 

193 gap_names = {n for n, p in network.ports.items() if isinstance(p, PortOnWire)} 

194 # End and vertex ports both name geometry without cutting a gap, so 

195 # they share one rule against gap ports (a wire may carry BOTH an end 

196 # and a vertex port — on different or even the same endpoint, where 

197 # momwire's own shunt+series conflict rule then applies). 

198 end_names = { 

199 p.wire 

200 for p in network.ports.values() 

201 if isinstance(p, (PortAtEnd, PortAtVertex)) 

202 } 

203 both = sorted(gap_names & end_names) 

204 if both: 

205 raise ValueError( 

206 f"wire name(s) {both} are referenced by both a PortOnWire and a " 

207 "PortAtEnd/PortAtVertex — a gap port cuts a delta gap in the " 

208 "wire while an end or vertex port must leave it gapless " 

209 "(issues #579, #898); use separate wires." 

210 ) 

211 orphaned = sorted(set(named_wires) - {None} - gap_names - end_names) 

212 if orphaned: 

213 raise ValueError( 

214 f"named wire(s) {orphaned} are not referenced by any PortOnWire, " 

215 "PortAtEnd or PortAtVertex in build_network(); a named wire " 

216 "becomes a port edge, and an unreferenced port is an OPEN gap " 

217 "that cuts the wire there (issue #578). Reference each name in " 

218 "Network.ports, or drop the name." 

219 )
