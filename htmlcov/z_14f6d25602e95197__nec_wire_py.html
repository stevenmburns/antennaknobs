1"""A wire's material as NEC cards: the GW radius, LD 5 and LD 2 (issue #1523). 

2 

3NEC has no insulated-wire card (NEC-2 never had one, and NEC-5 dropped NEC-4's 

4IS), so every deck writer here spells a dielectric jacket with the cards it does 

5have. momwire models the jacket as the Popović–Nešić PAIR (momwire#865): 

6 

7 * the kernel sees the equivalent radius a′ = a·(b/a)^((εr−1)/εr), which is 

8 the jacket's effect on the charge; 

9 * a distributed series inductance L′ = (μ0/2π)·(1 − 1/εr)·ln(b/a) puts back 

10 the inductance that enlarging the radius removed. 

11 

12The NEC writers used to write L′ alone, on the bare radius. That gets the 

13phase velocity right only to first order and overstates the V/I line impedance 

14by 2–9 %. On a jacketed dipole the two spellings differ by 1.6–5 % of |Z|, by 

15the same amount on bs2, razor-2p and NEC-5 (#1523's verdict). This module is 

16the one place the pair is spelled, so the NEC-5 deck, PyNEC's card calls and the 

17exported NEC-2 deck cannot disagree about what a jacket is. 

18 

19WRITING a′ ON THE GW CARD MOVES LD 5 TOO. NEC evaluates a conductor's internal 

20impedance at the GW radius, and at a′ that is a thicker wire: the copper's 

21resistance would fall by 41–80 % on the catalog's PVC wires. Scaling the 

22conductivity by (a/a′)² puts it back exactly. A round conductor's internal 

23impedance is Z = k·I0(ka) / (2πaσ·I1(ka)) with k = √(jωμ0σ), and σ′ = σ·(a/a′)² 

24leaves both k·a and k/(aσ) unchanged at a′. That holds at every frequency, so 

25one card still serves a sweep. 

26""" 

27 

28from __future__ import annotations 

29 

30from dataclasses import dataclass 

31from typing import TYPE_CHECKING 

32 

33from momwire import equivalent_radius, insulation_inductance 

34 

35if TYPE_CHECKING: 

36 from ..wire_catalog import WireSpec

38# A deck that carries a jacket says so, because its GW radius is larger than the 

39# wire gauge's and a reader comparing the two would otherwise suspect a bug. 

40JACKET_COMMENT_CARDS = ( 

41 "CM jacketed wire: GW radius = equivalent radius a' = a*(b/a)^((er-1)/er)", 

42 "CM its LD 5 conductivity is scaled by (a/a')^2; LD 2 = jacket inductance", 

43) 

44 

45 

46@dataclass(frozen=True) 

47class NecWireMaterial: 

48 """One wire's material cards.""" 

49 

50 radius: float # GW card radius, m 

51 conductivity: float | None # LD 5, S/m; None = no card (PEC) 

52 inductance: float | None # LD 2, H/m; None = no card (bare wire) 

53 

54 

55def nec_wire_material( 

56 radius: float, 

57 conductivity: float | None, 

58 spec: WireSpec | None, 

59 *, 

60 pair: bool = True, 

61) -> NecWireMaterial: 

62 """The NEC cards for a wire of conductor radius ``radius`` [m] and 

63 ``conductivity`` [S/m, None = PEC], jacketed as ``spec`` says. 

64 

65 A bare wire (``spec`` None, or no ``insulation_radius``) passes through 

66 unchanged. ``pair=False`` is the inductance-only spelling: the bare radius, 

67 the unscaled conductivity and L′. It is for a consumer that drops the LD 

68 cards (SimNEC's portal), where a′ without its L′ would be half of a model. 

69 """ 

70 if spec is None or not spec.insulation_radius: 

71 return NecWireMaterial(radius, conductivity, None) 

72 b, eps_r = spec.insulation_radius, spec.insulation_eps_r 

73 inductance = float(insulation_inductance(radius, b, eps_r)) 

74 if not pair: 

75 return NecWireMaterial(radius, conductivity, inductance) 

76 a_eq = float(equivalent_radius(radius, b, eps_r)) 

77 if conductivity is not None: 

78 conductivity = conductivity * (radius / a_eq) ** 2 

79 return NecWireMaterial(a_eq, conductivity, inductance)
