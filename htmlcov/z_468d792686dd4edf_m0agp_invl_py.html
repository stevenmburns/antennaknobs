1"""M0AGP's 160 m inverted L, and his "DX gain" comparison with a full-size vertical. 

2 

3From M0AGP's QRZ thread "Inverted L vs full-size vertical — 'DX Gain' 

4comparison" (thread 1005128, first post). He asks how much a 160 m inverted 

5L gives up against a full-size quarter-wave vertical as its vertical section 

6gets shorter, with the top wire re-cut for resonance at every height. This 

7design is his antenna, and its own study (`build_studies`) is his table. 

8 

9The antenna, in the framework's (x, y, z) convention: 

10 

11- a 5 cm driven gap at the foot of the riser, `base` above ground; 

12- the vertical section rises `vert_ft` FEET from the feed along z; 

13- the top wire runs `horiz_ft` FEET along +y from its top. A top wire of 0 ft 

14 is no top wire at all, so the full-size vertical is a setting of this same 

15 design (a state), with the same feed and the same radials; 

16- `n_radials` quarter-wave radials at the feed height, spread evenly from +x. 

17 Two of them run along +x and -x, perpendicular to the top wire, which is 

18 the orientation his numbers fit. 

19 

20The design itself carries no ground (free space, like the rest of the 

21catalog); the study states his: Sommerfeld average ground, 13 / 0.005 S/m. 

22 

23His figure of merit is DX gain: the power average of the gain over 2-10 

24degrees of elevation, 0.1 degree apart, converted back to dB. He reports 

25each inverted L relative to the vertical, two radials 5 ft up over average 

26ground, modelled on NEC-5. His azimuth isn't stated; the elevation cut at 

27azimuth 0 (along the radials, broadside to the top wire) is the one that 

28reproduces his table, so the study states it. 

29 

30The study, `verticals.m0agp_invl:DX gain vs the vertical`, sweeps the 

31vertical section over 20-100 ft and holds the top wire at resonance at every 

32height. Each inverted L reads relative to the vertical (a FIXED reference: it 

33sets the swept knob, so it is solved once and drawn flat). On momwire 0.69.0 

34at azimuth 0, the DX gain relative to the vertical (dB): 

35 

36 vert_ft 100 90 80 70 60 50 40 30 20 

37 dB -0.196 -0.365 -0.616 -0.984 -1.533 -2.369 -3.687 -5.843 -9.494 

38 

39Run it with `antennaknobs analyze --study "verticals.m0agp_invl:DX gain vs 

40the vertical"`, or from the design's Studies tab in the app. 

41""" 

42 

43import math 

44from types import MappingProxyType 

45 

46import antennaknobs.analyses as an 

47from antennaknobs import AntennaBuilder 

48from antennaknobs.network import Wire 

49 

50#: One foot, in metres. 

51FT = 0.3048 

52 

53#: The full-size vertical: resonant with no top wire at 1.83 MHz over 

54#: Sommerfeld 13/0.005 with these two radials (131.22 ft on NEC-5, 131.10 ft 

55#: on momwire). 

56VERTICAL_FT = 131.2 

57 

58 

59class Builder(AntennaBuilder): 

60 default_params = MappingProxyType( 

61 { 

62 "design_freq": 1.83, 

63 "freq": 1.83, 

64 # The feed and the radials: 5 ft up. 

65 "base": 1.524, 

66 # The vertical section and the top wire, in feet. The top wire 

67 # is cut for resonance at 70 ft of vertical section. 

68 "vert_ft": 70.0, 

69 "horiz_ft": 63.9, 

70 "n_radials": 2, 

71 "ui_params": MappingProxyType( 

72 { 

73 # Riser along z, top wire along y, radials along x: no 

74 # single 2D projection shows the L and its radials. 

75 "default_view": "iso", 

76 "base": {"min": 0.5, "max": 10.0, "unit": "m"}, 

77 "vert_ft": {"min": 10.0, "max": 140.0, "unit": "ft"}, 

78 # The hold's bounds: every resonant top from 100 ft of 

79 # vertical (33 ft) to 20 ft (112 ft) is inside, and the 

80 # top end stays clear of the half-wave antiresonance. 

81 "horiz_ft": {"min": 1.0, "max": 135.0, "unit": "ft"}, 

82 "n_radials": {"min": 1, "max": 16}, 

83 } 

84 ), 

85 } 

86 ) 

87 

88 def build_wires(self): 

89 eps = 0.05 

90 z = self.base 

91 vert = self.vert_ft * FT 

92 horiz = self.horiz_ft * FT 

93 radial_len = 0.25 * self.design_wavelength 

94 

95 tups = [Wire((0.0, 0.0, z), (0.0, 0.0, z + eps), ex=1 + 0j)] 

96 tups.append(Wire((0.0, 0.0, z + eps), (0.0, 0.0, z + vert))) 

97 if horiz > 0: 

98 # No top wire at all is the plain vertical (a zero-length wire 

99 # would not mesh). 

100 tups.append(Wire((0.0, 0.0, z + vert), (0.0, horiz, z + vert))) 

101 n = int(self.n_radials) 

102 for i in range(n): 

103 a = 2 * math.pi / n * i 

104 tups.append( 

105 Wire( 

106 (0.0, 0.0, z), 

107 (radial_len * math.cos(a), radial_len * math.sin(a), z), 

108 ) 

109 ) 

110 return tups 

111 

112 def build_studies(self): 

113 """DX gain against the vertical section, 20-100 ft, each inverted L 

114 relative to the full-size vertical (a FIXED reference: it sets the 

115 swept knob, so it is solved once and drawn flat), the top wire held 

116 at resonance at every height.""" 

117 me = "verticals.m0agp_invl" 

118 dx = an.ElevationWindow("DX gain", 2, 10, step=0.1, mean="power", az=0) 

119 return [ 

120 an.Analysis( 

121 "DX gain vs the vertical", 

122 an.Sweep( 

123 "vert_ft", 

124 values=(20.0, 30.0, 40.0, 50.0, 60.0, 70.0, 80.0, 90.0, 100.0), 

125 ), 

126 cross=an.Cross( 

127 states=( 

128 an.State("inverted L", design=me), 

129 an.State( 

130 "vertical", design=me, vert_ft=VERTICAL_FT, horiz_ft=0.0 

131 ), 

132 ) 

133 ), 

134 hold=an.Hold("resonance", adjust=("horiz_ft",)), 

135 views=(an.MetricPlot(dx, relative_to="vertical"), an.Knobs()), 

136 ground="finite:13,0.005", 

137 ), 

138 ]
