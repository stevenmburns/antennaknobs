1"""16-element linear bowtie array — sixteen identical bowtie elements on a 

2single row (constant height), each fed in phase. 

3 

4Same Module/lattice construction as `arrays.bowtie4x4`, but where that panel 

5stacks elements across 4 heights (nz=4) — which splits momwire's block-shape 

6classes by height under ANY ground model and drops the lattice-FFT coupling 

7path (array_block.py's height-refined shape classes; see issue #613) — this 

8array keeps every element at the SAME height (nz=1): one geometric shape, one 

9height, one block-shape class, always. The FFT lattice path engages in free 

10space AND stays engaged over ground (issue #613's other worked case: a 

11same-height array keeps the fast path even over Sommerfeld ground). 

12 

13Sixteen identical, uniformly-spaced elements on a regular row is exactly the 

14structure momwire's lattice-FFT coupling operator recognizes (P >= 16, one 

15shape class): solve it through `ArrayBlockSolver` and the block-Toeplitz/FFT 

16path engages automatically, ground or no ground. The default engine still 

17uses the dense B-spline solver; the FFT path is opt-in via `solver_kwargs`. 

18 

19All sixteen feeds are driven at unit amplitude and zero relative phase 

20(broadside); `impedance()` returns the sixteen per-element driving-point 

21values. 

22""" 

23 

24from types import MappingProxyType 

25 

26from antennaknobs import ( 

27 AntennaBuilder, 

28 Cell, 

29 Module, 

30 expand_modules, 

31 lattice, 

32) 

33from antennaknobs.designs.specialty import bowtie 

34from antennaknobs.network import Driven, Network, PortOnWire 

35 

36#: Fixed row — the name says 16x1, and 16 elements is the lattice-FFT floor. 

37_NX, _NZ = 16, 1 

38 

39#: The namespaced feed-port names lattice() produces (instance "e{i}_0" + 

40#: formal feed "feed"). Declared to the web UI as ``ui_params["feed_ports"]`` 

41#: so all sixteen feed markers render — build_network() designs don't infer 

42#: feed topology, they trust this list (adapter._declared_feed_ports). 

43_FEED_PORTS = tuple(f"e{i}_0.feed" for i in range(_NX)) 

44 

45 

46class Builder(AntennaBuilder): 

47 NX = _NX 

48 NZ = _NZ 

49 

50 default_params = MappingProxyType( 

51 { 

52 "freq": 28.47, 

53 # Geometry is hand-tuned in absolute metres (inherited from the 

54 # bowtie element); design_freq only anchors auto_mesh's density. 

55 "design_freq": 28.47, 

56 "angle_deg": 28.2625, # bowtie arm droop 

57 "length": 5.4213, # bowtie half-span 

58 "base": 9.0, # element height (z of the fan centre) — same for every element 

59 "del_y": 4.0, # element spacing along the row 

60 "ui_params": MappingProxyType( 

61 {"design_freq": {"hidden": True}, "feed_ports": _FEED_PORTS} 

62 ), 

63 } 

64 ) 

65 

66 def _element_module(self): 

67 # One bowtie element, meshed at this array's density, with its feed 

68 # edge named and its inline excitation dropped (the port drives it). 

69 elem_params = dict(bowtie.Builder.default_params) 

70 elem_params.update( 

71 freq=self.freq, 

72 design_freq=self.design_freq, 

73 angle_deg=self.angle_deg, 

74 length=self.length, 

75 base=self.base, 

76 ) 

77 for k in self.FRAMEWORK_PARAMS: 

78 if k in self._params: 

79 elem_params[k] = self._params[k] 

80 

81 wires = [ 

82 w._replace(name="feed", ex=None) if w.ex is not None else w 

83 for w in bowtie.Builder(elem_params).build_wires() 

84 ] 

85 return Module( 

86 cell=Cell(feeds=("feed",), wires=wires), 

87 ports={"feed": PortOnWire("feed")}, 

88 ) 

89 

90 def _elements(self): 

91 # Single row (nz=1): every element lands at the same z regardless of 

92 # dz, so it's passed as 0.0 rather than exposed as a knob. 

93 return lattice( 

94 self._element_module(), 

95 nx=self.NX, 

96 nz=self.NZ, 

97 dy=self.del_y, 

98 dz=0.0, 

99 ) 

100 

101 def build_wires(self): 

102 return expand_modules(self._elements()).wires 

103 

104 def build_network(self): 

105 a = expand_modules(self._elements()) 

106 return Network( 

107 ports=a.ports, 

108 branches=a.branches, 

109 sources=[Driven(port=f, voltage=1 + 0j) for f in a.feeds], 

110 )
