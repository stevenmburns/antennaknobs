1"""Import an MMANA-GAL model (``.maa``) as a NEC deck (AK#1897). 

2 

3MMANA-GAL is MININEC-based, and a ``.maa`` file is its saved model: wires, 

4sources, loads, its automatic-segmentation parameters and a ground line, in a 

5fixed POSITIONAL layout. ``read_maa`` turns one into the same 

6:class:`~antennaknobs.nec_import.NecDeck` a ``.nec`` deck becomes, so a 

7``.maa`` opens wherever a deck does (the CLI's ``@file``, the designs folder, 

8the workbench's Open...). 

9 

10How the file is read (the field meanings are the MMANA-GAL help plus the 

11oracle sitting recorded with the census, ``scratch/maa-census/REPORT.md`` §7): 

12 

13* **Layout.** Title; a bare ``*``; the design frequency (MHz); then the 

14 Wires, Source, Load, Segmentation and G/H/M/R/AzEl/X blocks, in that order, 

15 each behind a header line; then an optional ``$$$`` taper-wire table and an 

16 optional ``###`` comment. The header lines are cosmetic and localised 

17 (``***Wires***``, ``* Провода *``), so a block is found by its position, 

18 never by its header's text. Fields split on commas, tabs and spaces. 

19* **Wires** ``X1 Y1 Z1 X2 Y2 Z2 R SEG``, metres. R is the radius in metres (the 

20 GUI shows mm); a negative R points into the taper-wire table; R = 0 is 

21 MMANA's insulator wire and is refused. SEG: -1 tapered automatic (both 

22 ends), -2 / -3 tapered at the start / end only, 0 regular automatic, > 0 a 

23 manual segment count. 

24* **Positions** ``w<wire><b|c|e>[offset]``: wire start, centre or end, and an 

25 offset in pulses (``c+1`` toward the end, ``e1`` inward). MMANA puts sources 

26 and loads on PULSES, which straddle the junction between two segments; here 

27 each becomes a port at that segment end (a knot), which is what NEC-5's 

28 segment-end sources are, so the deck is built in NEC-5's spelling. 

29* **Sources** ``POS, phase (deg), volts``: phase BEFORE magnitude. 

30* **Loads** by type: 0 = ``L uH, C pF, Q`` (C = 0 an inductor, L = 0 a 

31 capacitor, both a parallel L||C trap); 1 = ``R, jX`` ohms, fixed at every 

32 frequency; 2 = MMANA's Laplace (S) form, refused. The load count line's 

33 second field is the "Use loads" switch: 0 means the file's loads are off, 

34 and they are not applied. 

35* **Segmentation** ``DM1, DM2, SC, EC``. MMANA's own mesher is not emulated 

36 (the 09-23 no-re-mesh-emulation precedent): every automatic wire is 

37 meshed by `_auto_mesh` from these four numbers, and the note says so. 

38* **Ground** ``G, H, M, R, Az, El, X``: G 0 free, 1 perfect, 2 real; H is 

39 added to every z; M is the wire material; R the reference impedance for 

40 SWR; Az/El the F/B rear ranges (pattern readouts, not physics). MMANA's real 

41 ground is MININEC-type -- perfect ground for the currents, the soil only in 

42 the pattern -- and the soil is NOT in the file (MMANA keeps it in its own 

43 settings), so a stated default soil is assumed. 

44 

45Whatever the file says that cannot be modelled is refused BY NAME, with its 

46line: a stack, a Laplace load, an insulator wire, a user-defined or iron 

47material, an offset on an automatically meshed wire (its pulse depends on 

48MMANA's own mesh), a source or load at a free wire end, ... Nothing is 

49snapped: a position that is not on the mesh cuts the wire there. 

50"""

124def decode_maa(raw: bytes) -> str: 

125 """A ``.maa`` file's text from its bytes. MMANA writes ASCII or cp1251 

126 (Cyrillic headers, titles and comments; no file in the corpus is UTF-8), 

127 so ASCII and UTF-8 are tried first and cp1251 is the fallback -- every 

128 byte decodes in cp1251 but one, which is replaced.""" 

129 if raw.startswith(b"\xef\xbb\xbf"): 

130 raw = raw[3:] 

131 try: 

132 return raw.decode("utf-8") 

133 except UnicodeDecodeError: 

134 return raw.decode("cp1251", errors="replace")

282def _parse(text: str, name: str) -> _MaaFile: 

283 lines = _Lines(text, name) 

284 if not lines.lines: 

285 raise lines.err(None, "the file is empty") 

286 # Title: the line(s) before the first bare '*'. One corpus file omits it. 

287 title_parts = [] 

288 while True: 

289 k, s = lines.take("'*' line after the title") 

290 if s == "*": 

291 break 

292 title_parts.append(s) 

293 if len(title_parts) > 3: 

294 raise lines.err( 

295 k, 

296 "no bare '*' line after the title -- this is not an MMANA-GAL " 

297 ".maa model", 

298 ) 

299 title = " ".join(title_parts) 

300 k, f = lines.fields("design frequency") 

301 freq = _num(f[0], k, "the design frequency", lines) 

302 if freq <= 0.0: 

303 # MMANA silently solves a 0.0 at its compiled-in 14.15 MHz and writes 

304 # that back (oracle sitting E12); the frequency is the model's, so a 

305 # file without one is refused rather than given a guess. 

306 raise lines.err( 

307 k, 

308 f"the design frequency is {f[0]} MHz; MMANA would silently use its " 

309 "built-in 14.15 MHz, and that is not this model's frequency -- set " 

310 "the model's frequency in MMANA and save it again", 

311 ) 

312 

313 # Wires. 

314 lines.header("Wires") 

315 k, f = lines.fields("wire count line") 

316 nw = _count(k, f, "wire", lines) 

317 if len(f) == 6: 

318 rows = _num(f[1], k, "the stack's first count", lines) 

319 cols = _num(f[2], k, "the stack's second count", lines) 

320 if rows * cols != 1: 

321 raise lines.err( 

322 k, 

323 f"the wire count line {', '.join(f)} is MMANA's Make Stack " 

324 f"setting: MMANA solves a {f[1]} x {f[2]} stack of these " 

325 f"{nw} wires. The stack dialog's spacing axes and placement " 

326 "are not yet mapped, so it is not built here -- and reading " 

327 "only the first field would model one antenna of the stack", 

328 ) 

329 elif len(f) != 1: 

330 raise lines.err( 

331 k, 

332 f"the wire count line has {len(f)} fields ({', '.join(f)}); MMANA " 

333 "writes 1 (the count) or 6 (the count and a stack)", 

334 ) 

335 if nw == 0: 

336 raise lines.err(k, "the model has no wires") 

337 wires = [] 

338 for _ in range(nw): 

339 k, f = lines.fields("wire line") 

340 if len(f) != 8: 

341 raise lines.err( 

342 k, 

343 f"a wire line has 8 fields (X1 Y1 Z1 X2 Y2 Z2 R SEG), this one " 

344 f"{len(f)}", 

345 ) 

346 v = [_num(t, k, "a wire field", lines) for t in f[:7]] 

347 seg = _int(f[7], k, "the wire's segmentation (SEG)", lines) 

348 wires.append(_Wire(tuple(v[0:3]), tuple(v[3:6]), v[6], seg, k)) 

349 

350 # Sources: count line `n, flag`. The flag is unidentified (no GUI change 

351 # moved it in the oracle sitting, and MMANA rewrites it on a plain save), 

352 # so it is read past. 

353 lines.header("Source") 

354 k, f = lines.fields("source count line") 

355 ns = _count(k, f, "source", lines) 

356 sources = [] 

357 for _ in range(ns): 

358 k, f = lines.fields("source line") 

359 if len(f) != 3: 

360 raise lines.err( 

361 k, 

362 f"a source line has 3 fields (position, phase in degrees, " 

363 f"volts), this one {len(f)}", 

364 ) 

365 sources.append( 

366 _Source( 

367 _pos(f[0], k, lines), 

368 _num(f[1], k, "the source's phase", lines), 

369 _num(f[2], k, "the source's voltage", lines), 

370 k, 

371 ) 

372 ) 

373 

374 # Loads: count line `n, use-loads`. 

375 lines.header("Load") 

376 k, f = lines.fields("load count line") 

377 nl = _count(k, f, "load", lines) 

378 loads_on = True 

379 if len(f) >= 2: 

380 flag = _int(f[1], k, "the load switch (Use loads)", lines) 

381 if flag not in (0, 1): 

382 raise lines.err( 

383 k, f"the load switch (Use loads) is {flag}; MMANA writes 0 or 1" 

384 ) 

385 loads_on = flag == 1 

386 loads = [] 

387 for _ in range(nl): 

388 k, f = lines.fields("load line") 

389 if len(f) < 2: 

390 raise lines.err(k, "a load line needs a position and a type") 

391 pos = _pos(f[0], k, lines) 

392 kind = _int(f[1], k, "the load type", lines) 

393 want = {0: 5, 1: 4, 2: 12}.get(kind) 

394 if want is None: 

395 raise lines.err( 

396 k, 

397 f"load type {kind} is not one MMANA writes (0 = L/C/Q, " 

398 "1 = R+jX, 2 = Laplace)", 

399 ) 

400 if len(f) != want: 

401 raise lines.err( 

402 k, f"a type {kind} load line has {want} fields, this one {len(f)}" 

403 ) 

404 vals = tuple(_num(t, k, "a load value", lines) for t in f[2:]) 

405 loads.append(_Load(pos, kind, vals, k)) 

406 

407 lines.header("Segmentation") 

408 k, f = lines.fields("segmentation line (DM1, DM2, SC, EC)") 

409 if len(f) != 4: 

410 raise lines.err( 

411 k, 

412 f"the segmentation line has 4 fields (DM1, DM2, SC, EC), this one {len(f)}", 

413 ) 

414 dm1, dm2, sc, ec = (_num(t, k, "a segmentation field", lines) for t in f) 

415 seg_line = k 

416 

417 lines.header("G/H/M/R/AzEl/X") 

418 k, f = lines.fields("ground line (G, H, M, R, Az, El, X)") 

419 if len(f) != 7: 

420 raise lines.err( 

421 k, 

422 f"the ground line has 7 fields (G, H, M, R, Az, El, X), this one " 

423 f"{len(f)} -- the line is incomplete", 

424 ) 

425 g = _int(f[0], k, "the ground type (G)", lines) 

426 h = _num(f[1], k, "the added height (H)", lines) 

427 mat = _int(f[2], k, "the wire material (M)", lines) 

428 ref_z = _num(f[3], k, "the reference impedance (R)", lines) 

429 x = _num(f[6], k, "the ground line's last field (X)", lines) 

430 ground_line = k 

431 

432 tapers = [] 

433 nxt = lines.peek() 

434 if nxt is not None and nxt[1].startswith("$"): 

435 lines.take("taper wire set header") 

436 k, f = lines.fields("taper wire set count") 

437 nt = _count(k, f, "taper wire set", lines) 

438 for _ in range(nt): 

439 k, f = lines.fields("taper wire set row") 

440 if len(f) < 4 or len(f) % 2: 

441 raise lines.err( 

442 k, 

443 "a taper wire set row is the pointer, the type, then " 

444 "(length, radius) pairs", 

445 ) 

446 vals = [_num(t, k, "a taper wire set field", lines) for t in f] 

447 secs = tuple(zip(vals[2::2], vals[3::2], strict=True)) 

448 tapers.append(_Taper(vals[0], int(vals[1]), secs, k)) 

449 nxt = lines.peek() 

450 if nxt is not None and not nxt[1].startswith("#"): 

451 raise lines.err( 

452 nxt[0], 

453 f"unexpected line after the ground line: {nxt[1][:40]!r} (only the " 

454 "taper wire set and the comment follow it)", 

455 ) 

456 

457 # The fields that are read but cannot be modelled, by name. 

458 if g not in _GROUNDS: 

459 raise lines.err( 

460 ground_line, f"the ground type (G) is {g}; MMANA writes 0, 1 or 2" 

461 ) 

462 if mat in _MATERIALS_REFUSED: 

463 what, why = _MATERIALS_REFUSED[mat] 

464 raise lines.err( 

465 ground_line, f"the wire material (M = {mat}) is MMANA's {what}: {why}" 

466 ) 

467 if mat not in _MATERIALS: 

468 raise lines.err( 

469 ground_line, 

470 f"the wire material (M) is {mat}; MMANA's list runs 0 to 7", 

471 ) 

472 if x != 0.0: 

473 raise lines.err( 

474 ground_line, 

475 f"the ground line's last field (X) is {f[6]}; every MMANA file seen " 

476 "writes 0, and what a non-zero value means is not settled", 

477 ) 

478 if not (dm1 > 0 and dm2 > 0 and ec >= 1 and sc >= 1): 

479 raise lines.err( 

480 seg_line, 

481 f"the segmentation line ({dm1:g}, {dm2:g}, {sc:g}, {ec:g}) needs " 

482 "DM1 > 0, DM2 > 0, SC >= 1 and EC >= 1", 

483 ) 

484 return _MaaFile( 

485 title=title, 

486 freq_mhz=freq, 

487 wires=tuple(wires), 

488 sources=tuple(sources), 

489 loads=tuple(loads), 

490 loads_on=loads_on, 

491 dm1=dm1, 

492 dm2=dm2, 

493 sc=sc, 

494 ec=ec, 

495 ground=g, 

496 height=h, 

497 material=mat, 

498 ref_z=ref_z, 

499 tapers=tuple(tapers), 

500 )

508def _auto_mesh(length: float, lam: float, seg: int, f: _MaaFile, radius: float): 

509 """The segment lengths of one automatically meshed wire (SEG <= 0). 

510 

511 MMANA's mesher is its own and is not emulated. Its parameters are read 

512 for what they say: 

513 

514 * SEG = 0, regular: equal segments no longer than lambda / DM2. 

515 * SEG = -1 / -2 / -3, tapered at both ends / the start / the end: at each 

516 tapered end the segments start at lambda / (DM1 * EC) and grow by the 

517 factor SC per segment until they reach lambda / DM2, which is the 

518 segment length over the rest of the wire. A both-ends wire is meshed 

519 symmetrically, with an even count in the middle, so its centre is 

520 always a segment end (a pulse) -- where ``w<n>c`` puts its port. 

521 

522 The help states the taper in two inconsistent forms (lambda / (DM1 * EC) 

523 to lambda / (SC * DM2), and lambda / (SC * DM1) to lambda / DM2); this 

524 takes the finer end of the first and the coarser end of the second. 

525 

526 No segment is shorter than two radii: below that no thin-wire kernel 

527 here is valid (NEC's own limit for its extended kernel).""" 

528 floor = 2.0 * radius 

529 dmax = max(lam / f.dm2, floor) 

530 if seg == 0: 

531 n = math.ceil(length / dmax - 1e-9) 

532 if n > _HARD_SEGMENT_CAP: 

533 raise _Refusal(f"would take {n} segments") 

534 n = max(1, n) 

535 return [length / n] * n 

536 dmin = max(lam / (f.dm1 * f.ec), floor) 

537 run = [] 

538 s = dmin 

539 while s < dmax * (1 - 1e-12): 

540 run.append(s) 

541 if len(run) > _HARD_SEGMENT_CAP: 

542 raise _Refusal("its taper would take too many segments") 

543 s *= f.sc 

544 if f.sc == 1.0: 

545 break 

546 ends = 2 if seg == -1 else 1 

547 while run and ends * sum(run) > length * (1 + 1e-12): 

548 run.pop() 

549 middle = length - ends * sum(run) 

550 n_mid = math.ceil(middle / dmax - 1e-9) if middle > length * 1e-9 else 0 

551 if n_mid > _HARD_SEGMENT_CAP: 

552 raise _Refusal(f"would take {n_mid} segments") 

553 if seg == -1: 

554 if n_mid % 2: 

555 n_mid += 1 

556 if n_mid == 0 and not run: 

557 n_mid = 2 

558 elif n_mid == 0 and not run: 

559 n_mid = 1 

560 mid = [middle / n_mid] * n_mid if n_mid else [] 

561 if seg == -1: 

562 return [*run, *mid, *reversed(run)] 

563 if seg == -2: 

564 return [*run, *mid] 

565 return [*mid, *reversed(run)]

571 along the WHOLE wire (0 .. 1) and each segment's radius.""" 

572 

573 knots: list[float] 

574 radii: list[float] 

575 manual: int | None # the manual count, when SEG > 0

578def _sections(w: _Wire, length: float, tapers: dict, line_err): 

579 """``[(t0, t1, radius)]``: the wire's radius sections along it, as 

580 fractions. A plain wire is one section; a negative R points into the 

581 taper wire set (``$$$`` rows ``pointer, type, L0, R0, L1, R1, ..., 

582 99999.9, Rlast``). Type 0 (``<>``) is symmetric about the centre with L0 

583 the WHOLE centre section and each later L a section on each side; type 1 

584 (``->``) runs from the wire's start. Types 2 and 3 are refused: what 

585 their ``*`` changes is not settled.""" 

586 if w.radius > 0: 

587 return [(0.0, 1.0, w.radius)] 

588 if w.radius == 0: 

589 raise line_err( 

590 w.line, 

591 "this wire has radius 0, MMANA's insulator wire (it joins wires " 

592 "without electrical contact), which nothing here models", 

593 ) 

594 row = tapers.get(w.radius) 

595 if row is None: 

596 raise line_err( 

597 w.line, 

598 f"this wire's radius {w.radius:g} points into the taper wire set, " 

599 "which has no row for it", 

600 ) 

601 secs = list(row.sections) 

602 if row.kind not in (0, 1): 

603 raise line_err( 

604 row.line, 

605 f"taper wire set type {row.kind} (MMANA's starred <>* / ->* forms) " 

606 "is not read here: what the star changes is not yet settled", 

607 ) 

608 if any(r <= 0 for _, r in secs) or any(ln <= 0 for ln, _ in secs): 

609 raise line_err( 

610 row.line, "a taper wire set section has a non-positive length or radius" 

611 ) 

612 if row.kind == 1: 

613 out, at = [], 0.0 

614 for ln, r in secs: 

615 t1 = min(1.0, at + ln / length) 

616 if t1 > at: 

617 out.append((at, t1, r)) 

618 at = t1 

619 if at >= 1.0: 

620 break 

621 if at < 1.0: 

622 out[-1] = (out[-1][0], 1.0, out[-1][2]) 

623 return out 

624 # Type 0: symmetric. The centre section first, then outward. 

625 half = 0.5 

626 bounds = [] # (half-width fraction reached, radius) 

627 reach = min(half, 0.5 * secs[0][0] / length) 

628 bounds.append((reach, secs[0][1])) 

629 for ln, r in secs[1:]: 

630 if reach >= half: 

631 break 

632 reach = min(half, reach + ln / length) 

633 bounds.append((reach, r)) 

634 if bounds[-1][0] < half: 

635 bounds[-1] = (half, bounds[-1][1]) 

636 out = [] 

637 inner = 0.0 

638 right = [] 

639 for reach, r in bounds: 

640 right.append((0.5 + inner, 0.5 + reach, r)) 

641 inner = reach 

642 left = [(1.0 - b, 1.0 - a, r) for a, b, r in reversed(right)] 

643 # The centre section spans both sides as one. 

644 centre = (left[-1][0], right[0][1], right[0][2]) 

645 out = [*left[:-1], centre, *right[1:]] 

646 return [s for s in out if s[1] - s[0] > _SAME_T]

649def _mesh_wire(w: _Wire, f: _MaaFile, lam: float, tapers: dict, line_err) -> _Piece: 

650 length = math.dist(w.p1, w.p2) 

651 if length <= 0.0: 

652 raise line_err(w.line, "this wire has zero length") 

653 if w.seg < -3: 

654 raise line_err( 

655 w.line, 

656 f"SEG is {w.seg}; MMANA writes -1, -2 or -3 (tapered), 0 (regular) " 

657 "or a manual segment count", 

658 ) 

659 secs = _sections(w, length, tapers, line_err) 

660 if w.seg > 0: 

661 if len(secs) > 1: 

662 raise line_err( 

663 w.line, 

664 f"a manual segment count ({w.seg}) on a stepped-radius taper " 

665 "wire: how MMANA shares it among the sections is not settled", 

666 ) 

667 if w.seg > _HARD_SEGMENT_CAP: 

668 raise line_err(w.line, f"SEG asks for {w.seg} segments") 

669 # MMANA rounds an odd count UP to even, so every wire has a pulse at 

670 # its centre (measured, MMANA-GAL Basic 3.5, 2026-10-06: SEG 7 gave 8 

671 # segments in the currents CSV; AK#1919). SEG counts segments. 

672 n = w.seg + (w.seg % 2) 

673 return _Piece([k / n for k in range(n + 1)], [secs[0][2]] * n, n) 

674 knots, radii = [0.0], [] 

675 for t0, t1, r in secs: 

676 try: 

677 lens = _auto_mesh((t1 - t0) * length, lam, w.seg, f, r) 

678 except _Refusal as e: 

679 raise line_err(w.line, f"this wire {e}") from None 

680 span = sum(lens) 

681 at = t0 

682 for k, ln in enumerate(lens): 

683 at = t1 if k == len(lens) - 1 else at + (t1 - t0) * ln / span 

684 knots.append(at) 

685 radii.append(r) 

686 return _Piece(knots, radii, None)

713 r = repr(float(v)) 

714 return "0" if r in ("0.0", "-0.0") else r

716 

717def read_maa( 

718 text: str, *, name: str = "MMANA model", limits: GeometryLimits | None = None 

719) -> MaaImport: 

720 """Read the text of an MMANA-GAL ``.maa`` model (see the module 

721 docstring) into a `MaaImport`. Raises ``ValueError`` naming the file, the 

722 line and the field for anything malformed or anything the file asks for 

723 that cannot be modelled. ``limits`` bounds the structure, in MMANA's own 

724 wires and the expanded mesh's segments, before the deck is built.""" 

725 f = _parse(text, name) 

726 

727 def line_err(line, msg): 

728 return ValueError(f"{name} line {line}: {msg}" if line else f"{name}: {msg}") 

729 

730 lam = C_LIGHT_MHZ_M / f.freq_mhz 

731 h = f.height 

732 tapers = {t.pointer: t for t in f.tapers} 

733 if limits is not None and len(f.wires) > limits.max_wires: 

734 raise GeometryLimitError( 

735 f"{name}: this model has {len(f.wires)} wires (the limit is " 

736 f"{limits.max_wires})" + (f"; {limits.note}" if limits.note else ""), 

737 ) 

738 

739 # Geometry: H is added to every z (the help's "add height"). MMANA's own 

740 # 20 m dipole (DP20, G = 0, H = 20) lies at z = 0 in the file and solves 

741 # over perfect ground as a dipole 20 m up, so H is geometry. 

742 wires = [ 

743 replace( 

744 w, 

745 p1=(w.p1[0], w.p1[1], w.p1[2] + h), 

746 p2=(w.p2[0], w.p2[1], w.p2[2] + h), 

747 ) 

748 for w in f.wires 

749 ] 

750 grounded = f.ground != 0 

751 # Ends closer than this are one point; a z this close to 0 is on the 

752 # ground (H added to a file's z = -H need not cancel to the last bit). 

753 extent = max((abs(c) for w in wires for c in (*w.p1, *w.p2)), default=1.0) or 1.0 

754 tol = 1e-6 * extent 

755 if grounded: 

756 

757 def flat(p): 

758 return (p[0], p[1], 0.0) if abs(p[2]) <= tol else p 

759 

760 wires = [replace(w, p1=flat(w.p1), p2=flat(w.p2)) for w in wires] 

761 for w in wires: 

762 if min(w.p1[2], w.p2[2]) < 0.0: 

763 raise line_err( 

764 w.line, 

765 "this wire goes below the ground (z + H < 0); MMANA's " 

766 "ground models nothing buried", 

767 ) 

768 

769 pieces = [_mesh_wire(w, f, lam, tapers, line_err) for w in wires] 

770 total = sum(len(p.radii) for p in pieces) 

771 if limits is not None and total > limits.max_segments: 

772 raise GeometryLimitError( 

773 f"{name}: this model meshes to {total} segments (the limit is " 

774 f"{limits.max_segments})" + (f"; {limits.note}" if limits.note else ""), 

775 ) 

776 

777 # Shared ends: one coordinate per junction (MMANA joins wires where their 

778 # ends coincide), matched to `tol`, well inside any real gap. 

779 nodes: list[tuple[float, float, float]] = [] 

780 ends: dict[tuple[int, int], int] = {} 

781 for i, w in enumerate(wires): 

782 for e, p in ((0, w.p1), (1, w.p2)): 

783 for k, q in enumerate(nodes): 

784 if math.dist(p, q) <= tol: 

785 ends[(i, e)] = k 

786 break 

787 else: 

788 ends[(i, e)] = len(nodes) 

789 nodes.append(p) 

790 degree: dict[int, int] = {} 

791 for k in ends.values(): 

792 degree[k] = degree.get(k, 0) + 1 

793 # A wire end on another wire's INTERIOR: MMANA joins wires only at their 

794 # ends, and a mesh here could put a segment end on that very point and 

795 # join them, so the two programs would disagree. Refused, by name. 

796 for i, w in enumerate(wires): 

797 for j, v in enumerate(wires): 

798 if i == j: 

799 continue 

800 d = [b - a for a, b in zip(v.p1, v.p2, strict=True)] 

801 ln2 = sum(c * c for c in d) 

802 for e, p in ((0, w.p1), (1, w.p2)): 

803 t = sum((p[k] - v.p1[k]) * d[k] for k in range(3)) / ln2 

804 if not (1e-9 < t < 1 - 1e-9): 

805 continue 

806 q = tuple(v.p1[k] + t * d[k] for k in range(3)) 

807 if math.dist(p, q) <= tol: 

808 raise line_err( 

809 w.line, 

810 f"wire {i + 1}'s {'start' if e == 0 else 'end'} lies on " 

811 f"wire {j + 1} between its ends; MMANA joins wires only " 

812 f"at their ends -- split wire {j + 1} there", 

813 ) 

814 # Two wires crossing between their ends: MMANA does not join them there 

815 # (its own library has such crossovers, e.g. DL2KQ's multi-turn loops), 

816 # so it solves two wires at zero spacing; a mesh here would join them 

817 # wherever both put a segment end on the crossing. Neither is the 

818 # antenna, so the model is refused by name. 

819 for i, w in enumerate(wires): 

820 u = [b - a for a, b in zip(w.p1, w.p2, strict=True)] 

821 for j in range(i + 1, len(wires)): 

822 v = wires[j] 

823 vv = [b - a for a, b in zip(v.p1, v.p2, strict=True)] 

824 r = [a - b for a, b in zip(w.p1, v.p1, strict=True)] 

825 a_ = sum(c * c for c in u) 

826 b_ = sum(x * y for x, y in zip(u, vv, strict=True)) 

827 c_ = sum(c * c for c in vv) 

828 d_ = sum(x * y for x, y in zip(u, r, strict=True)) 

829 e_ = sum(x * y for x, y in zip(vv, r, strict=True)) 

830 den = a_ * c_ - b_ * b_ 

831 if den <= 1e-12 * a_ * c_: 

832 continue # parallel: an overlap puts an end inside, above 

833 sp = (b_ * e_ - c_ * d_) / den 

834 tq = (a_ * e_ - b_ * d_) / den 

835 if not (1e-9 < sp < 1 - 1e-9 and 1e-9 < tq < 1 - 1e-9): 

836 continue 

837 pp = [w.p1[k] + sp * u[k] for k in range(3)] 

838 qq = [v.p1[k] + tq * vv[k] for k in range(3)] 

839 if math.dist(pp, qq) <= tol: 

840 raise line_err( 

841 w.line, 

842 f"wires {i + 1} and {j + 1} cross between their ends. MMANA " 

843 "does not join wires there, so it solves two wires passing " 

844 "through each other at zero spacing, which no wire model " 

845 "here solves faithfully. Split both there to join them, or " 

846 "move one a few radii off the other (an insulated " 

847 "crossover)", 

848 ) 

849 

850 def on_ground(i, e): 

851 return grounded and nodes[ends[(i, e)]][2] == 0.0 

852 

853 notes: list[str] = [] 

854 cut_notes: list[str] = [] 

855 cut_wires: set[int] = set() 

856 

857 def resolve(pos: _Pos, line: int, what: str) -> tuple[int, int]: 

858 """``(wire index, knot)`` for a position, cutting the wire at an 

859 exact centre that is not on the mesh.""" 

860 if not 1 <= pos.wire <= len(wires): 

861 raise line_err( 

862 line, 

863 f"{what} {pos.text} names wire {pos.wire}; the model has " 

864 f"{len(wires)} wires", 

865 ) 

866 i = pos.wire - 1 

867 pc = pieces[i] 

868 n = len(pc.radii) 

869 if pos.offset and pc.manual is None: 

870 raise line_err( 

871 line, 

872 f"{what} {pos.text} is {abs(pos.offset)} pulse" 

873 f"{'s' if abs(pos.offset) != 1 else ''} from the wire's " 

874 f"{_ANCHOR_WORD[pos.anchor]} on an automatically segmented wire " 

875 f"(SEG {wires[i].seg}): where that pulse is depends on MMANA's " 

876 "own mesh, which is not emulated here. Give the wire a manual " 

877 "segment count in MMANA to place it", 

878 ) 

879 if pos.anchor == "b": 

880 knot = pos.offset 

881 elif pos.anchor == "e": 

882 knot = n - pos.offset 

883 else: 

884 mid = next( 

885 (k for k, t in enumerate(pc.knots) if abs(t - 0.5) <= _SAME_T), None 

886 ) 

887 if mid is None or (pos.offset and i in cut_wires): 

888 if pos.offset: 

889 raise line_err( 

890 line, 

891 f"{what} {pos.text} counts pulses from the centre of a " 

892 f"wire with an odd segment count ({n}), which has no " 

893 "pulse at its centre; which pulse MMANA takes as the " 

894 "centre there is not settled", 

895 ) 

896 # Cut, never snap: the wire gets a segment end at its centre. 

897 k = next(k for k, t in enumerate(pc.knots) if t > 0.5) 

898 pc.knots.insert(k, 0.5) 

899 pc.radii.insert(k, pc.radii[k - 1]) 

900 cut_wires.add(i) 

901 cut_notes.append( 

902 f"{pos.text}: wire {pos.wire} has no segment end at its " 

903 "centre, so it is cut there" 

904 ) 

905 mid = k 

906 knot = mid + pos.offset 

907 n = len(pc.radii) 

908 if not 0 <= knot <= n: 

909 raise line_err( 

910 line, 

911 f"{what} {pos.text} is off wire {pos.wire}, which has {n} segments", 

912 ) 

913 if knot in (0, n): 

914 e = 0 if knot == 0 else 1 

915 node = ends[(i, e)] 

916 if degree[node] < 2 and not on_ground(i, e): 

917 raw_z = (f.wires[i].p1 if e == 0 else f.wires[i].p2)[2] 

918 if grounded and raw_z == 0.0 and h != 0.0: 

919 why = ( 

920 f" It is at z = 0 in the file, but the added height " 

921 f"H = {h:g} m lifts it off the ground; whether MMANA " 

922 "still puts a pulse there is not settled" 

923 ) 

924 elif not grounded and raw_z == 0.0: 

925 why = ( 

926 " It is at z = 0, where MMANA puts a pulse over a " 

927 "ground; what it does there in free space (G = 0) is " 

928 "not settled" 

929 ) 

930 else: 

931 why = "" 

932 raise line_err( 

933 line, 

934 f"{what} {pos.text} is at a free wire end (wire {pos.wire}'s " 

935 f"{'start' if e == 0 else 'end'} joins no other wire and " 

936 "is not on the ground), where MININEC has no pulse." + why, 

937 ) 

938 if on_ground(i, e) and h != 0.0: 

939 raise line_err( 

940 line, 

941 f"{what} {pos.text} is where wire {pos.wire} meets the " 

942 f"ground at z + H = 0, with the added height H = {h:g} m; " 

943 "whether MMANA puts a pulse there (z = 0 in the file, or " 

944 "z + H = 0) is not settled", 

945 ) 

946 return i, knot 

947 

948 # Sources. 

949 if not f.sources: 

950 raise line_err(None, "the model has no source") 

951 feeds = [] 

952 for s in f.sources: 

953 if s.volts == 0.0: 

954 raise line_err( 

955 s.line, 

956 f"the source at {s.pos.text} has zero volts, which drives nothing", 

957 ) 

958 at = resolve(s.pos, s.line, "source") 

959 feeds.append((at, cmath.rect(s.volts, math.radians(s.phase_deg)), s)) 

960 seen = {} 

961 for at, _v, s in feeds: 

962 if at in seen: 

963 raise line_err( 

964 s.line, 

965 f"two sources ({seen[at]} and {s.pos.text}) are on one pulse", 

966 ) 

967 seen[at] = s.pos.text 

968 

969 # Loads. 

970 ld_cards = [] # (at, card mnemonic fields) 

971 q_notes = [] 

972 omega0 = 2.0 * math.pi * f.freq_mhz * 1e6 

973 if f.loads and not f.loads_on: 

974 notes.append( 

975 f"The file's {len(f.loads)} load{'s are' if len(f.loads) != 1 else ' is'} " 

976 "switched off (MMANA's Use loads is unticked), so " 

977 f"{'they are' if len(f.loads) != 1 else 'it is'} not applied." 

978 ) 

979 for ld in f.loads if f.loads_on else (): 

980 if ld.kind == 2: 

981 raise line_err( 

982 ld.line, 

983 f"the load at {ld.pos.text} is MMANA's Laplace (S) form, a " 

984 "rational impedance Z(s) that no element here spells", 

985 ) 

986 at = resolve(ld.pos, ld.line, "load") 

987 if ld.kind == 1: 

988 r, x = ld.values 

989 if r < 0: 

990 raise line_err( 

991 ld.line, f"the load at {ld.pos.text} has a negative resistance" 

992 ) 

993 if r == 0 and x == 0: 

994 continue 

995 ld_cards.append((at, 4, r, x, 0.0, ld.pos.text)) 

996 continue 

997 l_uh, c_pf, q = ld.values 

998 if l_uh < 0 or c_pf < 0 or q < 0: 

999 raise line_err( 

1000 ld.line, 

1001 f"the load at {ld.pos.text} has a negative L, C or Q", 

1002 ) 

1003 lh, cf = l_uh * 1e-6, c_pf * 1e-12 

1004 if lh == 0 and cf == 0: 

1005 continue 

1006 if lh and cf: 

1007 # A parallel L||C trap. Q is the coil's: 4nec2's LD 6 reading, 

1008 # a parallel loss R = Q * w0 * L at the file's frequency. 

1009 rp = q * omega0 * lh if q else 0.0 

1010 ld_cards.append((at, 1, rp, lh, cf, ld.pos.text)) 

1011 if q: 

1012 q_notes.append(f"{ld.pos.text} (trap, {rp:.4g} ohm in parallel)") 

1013 elif lh: 

1014 rs = omega0 * lh / q if q else 0.0 

1015 ld_cards.append((at, 0, rs, lh, 0.0, ld.pos.text)) 

1016 if q: 

1017 q_notes.append(f"{ld.pos.text} (coil, {rs:.4g} ohm in series)") 

1018 else: 

1019 rs = 1.0 / (omega0 * cf * q) if q else 0.0 

1020 ld_cards.append((at, 0, rs, 0.0, cf, ld.pos.text)) 

1021 if q: 

1022 q_notes.append(f"{ld.pos.text} (capacitor, {rs:.4g} ohm in series)") 

1023 

1024 # Objects on DIFFERENT wires' ends at one junction of three or more wire 

1025 # ends. Each is a series gap at its own wire's end, which is what MMANA's 

1026 # position names; the port model here serves one gap per junction 

1027 # (AK#1886), so the model is refused rather than half-built. 

1028 at_node: dict[int, dict[int, list[str]]] = {} 

1029 for at, text_ in [(at, s.pos.text) for at, _v, s in feeds] + [ 

1030 (at, pos) for at, *_rest, pos in ld_cards 

1031 ]: 

1032 i, knot = at 

1033 n = len(pieces[i].radii) 

1034 if knot in (0, n): 

1035 node = ends[(i, 0 if knot == 0 else 1)] 

1036 at_node.setdefault(node, {}).setdefault(i, []).append(text_) 

1037 for node, by_wire in at_node.items(): 

1038 if len(by_wire) > 1 and degree[node] >= 3: 

1039 named = " and ".join(p for ps in by_wire.values() for p in ps) 

1040 raise line_err( 

1041 None, 

1042 f"{named} sit at one junction of {degree[node]} wire ends, each " 

1043 "at its own wire's end: that is a series gap in each of those " 

1044 "wires, and this route serves one gap per junction (AK#1886). " 

1045 "Put them on one wire's end there, or move one of them along " 

1046 "its wire", 

1047 ) 

1048 

1049 # Panels: one GW per run of equal segments of one radius, so every knot 

1050 # a port lands on is a segment end of some GW. `seg_card` maps a wire's 

1051 # 1-based segment to its (tag, segment) on the panels. 

1052 gw_lines = [] 

1053 seg_card: dict[tuple[int, int], tuple[int, int]] = {} 

1054 tag = 0 

1055 

1056 def pt(i, t): 

1057 if t == 0.0: 

1058 return nodes[ends[(i, 0)]] 

1059 if t == 1.0: 

1060 return nodes[ends[(i, 1)]] 

1061 w = wires[i] 

1062 return tuple(a + (b - a) * t for a, b in zip(w.p1, w.p2, strict=True)) 

1063 

1064 for i, pc in enumerate(pieces): 

1065 ks = pc.knots 

1066 n = len(pc.radii) 

1067 start = 0 

1068 while start < n: 

1069 stop = start + 1 

1070 seg_len = ks[start + 1] - ks[start] 

1071 while ( 

1072 stop < n 

1073 and pc.radii[stop] == pc.radii[start] 

1074 and abs((ks[stop + 1] - ks[stop]) - seg_len) <= 1e-9 * seg_len 

1075 ): 

1076 stop += 1 

1077 tag += 1 

1078 a, b = pt(i, ks[start]), pt(i, ks[stop]) 

1079 gw_lines.append( 

1080 f"GW {tag} {stop - start} " 

1081 + " ".join(_fmt(c) for c in (*a, *b)) 

1082 + f" {_fmt(pc.radii[start])}" 

1083 ) 

1084 for k in range(start + 1, stop + 1): 

1085 seg_card[(i, k)] = (tag, k - start) 

1086 start = stop 

1087 

1088 # Which side of its knot each port is spelled from: the end of the 

1089 # segment BEFORE it (NEC-5's I4 = 2), or the start of the one after 

1090 # (I4 = 1). The port model hosts one port per wire piece end, so two 

1091 # ports on the two ends of one segment must not both ride that segment: 

1092 # a port on knot 0 can only be spelled from segment 1, and the knot after 

1093 # a knot spelled that way is spelled from its own segment after, too. 

1094 claimed: dict[int, set[int]] = {} 

1095 for at, *_ in [*feeds, *ld_cards]: 

1096 claimed.setdefault(at[0], set()).add(at[1]) 

1097 side: dict[tuple[int, int], int] = {} 

1098 for i, ks in claimed.items(): 

1099 n = len(pieces[i].radii) 

1100 after = False 

1101 prev = None 

1102 for k in sorted(ks): 

1103 after = k == 0 or (after and prev == k - 1 and k < n) 

1104 side[(i, k)] = 1 if after else 2 

1105 prev = k 

1106 

1107 def addr(at): 

1108 i, k = at 

1109 if side[at] == 1: 

1110 return (*seg_card[(i, k + 1)], 1) 

1111 return (*seg_card[(i, k)], 2)
