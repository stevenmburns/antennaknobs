1"""Antenna example registry. 

2 

3Each antenna geometry exposed by the web UI is defined in its own module 

4under this package and registered via `register(EXAMPLE)`. The dispatchers 

5in `web.server` and `web.pynec_backend` look up the example by its `name` 

6field, so removing an antenna is a two-line change: delete the module file 

7and the matching `from . import <name>` line below. 

8 

9The registry is intentionally web-layer (not solver-core) — each example 

10parses the request dict, calls shared helpers in `web.server` / 

11`web.pynec_backend`, and produces the JSON-shaped response the frontend 

12consumes. The solver package `momwire/` stays free of UI concerns. 

13""" 

14 

15from __future__ import annotations 

16 

17from ._base import AntennaExample, ParamSpec 

18 

19REGISTRY: dict[str, AntennaExample] = {} 

20 

21 

22class UnknownGeometryError(ValueError): 

23 """A request named a design the registry does not hold (issue #1343). 

24 

25 Raised instead of quietly answering for the first registered design, 

26 which is what every `EXAMPLES.get(key) or next(iter(EXAMPLES.values()))` 

27 used to do: a request for ``invvee`` instead of ``dipoles.invvee`` came 

28 back with solved numbers for ``arrays.bowtie16x1`` and no error, and the 

29 Windows box session nearly reported the invvee as broken on NEC-5 on the 

30 strength of it (2026-09-09). The server maps this to a 400 naming the 

31 key and the nearest registered names; the solve channel formats it like 

32 any other solve error. 

33 """ 

34 

35 

36def example_for(geometry) -> AntennaExample: 

37 """The registered design for ``geometry``, or UnknownGeometryError. 

38 

39 An ABSENT key is a different thing from an unknown one: callers that 

40 default a missing key to the first design (``req.get("geometry", 

41 next(iter(REGISTRY)))``) keep that convention; this only refuses a key 

42 that was given and does not resolve. 

43 """ 

44 ex = REGISTRY.get(geometry) 

45 if ex is not None: 

46 return ex 

47 import difflib 

48 

49 near = difflib.get_close_matches(str(geometry), list(REGISTRY), n=3, cutoff=0.4) 

50 tail = f"; did you mean {', '.join(repr(n) for n in near)}?" if near else "" 

51 raise UnknownGeometryError( 

52 f"unknown geometry {geometry!r}: not a registered design{tail} " 

53 f"(the registry holds {len(REGISTRY)} designs, named like 'dipoles.invvee')" 

54 ) 

55 

56 

57def register(example: AntennaExample) -> AntennaExample: 

58 if example.name in REGISTRY: 

59 raise ValueError(f"duplicate antenna example: {example.name}") 

60 REGISTRY[example.name] = example 

61 return example

66# default_params and wraps MomwireEngine + PyNECEngine in the SolveFn / 

67# SweepFn contract above. See web/adapter.py for the bridge and what each 

68# Builder can opt into via its `ui_params` dict. 

69from .. import adapter # noqa: E402

71adapter.register_all()
