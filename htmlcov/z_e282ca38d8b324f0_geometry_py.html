1"""Translate the flat (p0, p1, n_seg, excitation) wire list used by 

2AntennaBuilder.build_wires() into the polyline + feed-arclength shape 

3that momwire's solver classes consume. 

4 

5The flat list expresses connectivity implicitly: two tuples are part of 

6the same electrical wire when they share an endpoint (within `eps`). 

7momwire wants each electrical wire as one (M, 3) polyline with junction 

8information (KCL at shared nodes) explicit; this module recovers the 

9graph, decomposes it into maximal chains between junction/endpoint 

10nodes, and emits the junction list. 

11 

12Supported topologies: 

13 * Open chains and any number of junctions of any degree (tees, X's, 

14 hentenna-style multi-junction, fandipole-style multi-spoke feeds). 

15 * Pure cycles (closed loops). A cycle is cut at one edge into two 

16 polylines joined by a junction at each cut node, so momwire's KCL 

17 carries the current around the loop. The cut edge is the loop's port 

18 edge when it has one (driven loops); for a PARASITIC loop, which 

19 radiates only through mutual coupling, the cut edge is arbitrary. 

20 A cycle with two or more port edges is not yet handled. 

21 * One or more excited segments per geometry (each becomes a delta-gap 

22 feed in the returned `feeds` list). The geometry as a whole must 

23 carry at least one excitation, but individual loop components need 

24 not (a parasitic loop is excited only by coupling). 

25""" 

26 

27from __future__ import annotations 

28 

29import numpy as np 

30 

31from .network import as_wire 

32 

33 

34def _round_point(p, eps): 

35 # Quantize endpoints onto an eps-spaced grid so 1e-14 floating-point 

36 # noise doesn't fragment what is logically a shared node. 

37 return tuple(round(float(c) / eps) * eps for c in p) 

38 

39 

40def flat_wires_to_polylines(tups, *, eps=1e-6, end_ports=None): 

41 """Convert flat wire tuples to momwire polyline form. 

42 

43 ``end_ports`` (issue #579): iterable of ``(wire_name, "p0"|"p1")`` pairs 

44 naming wire ENDPOINTS that must become junction-node ports. Each such 

45 node is forced to be a polyline boundary (so momwire gives it junction 

46 directional bases) and its shared-node group is emitted in `junctions` 

47 even when only one polyline end lives there (a lone conductor end — the 

48 1-entry groups momwire#172 made legal). A name that appears in 

49 ``end_ports`` is NOT registered as a gap-feed port edge (no delta gap is 

50 cut in it); it identifies the wire only. The returned 

51 ``end_port_junctions`` maps each ``(wire_name, end)`` to its junction 

52 index — pass those straight to a momwire solver's ``junction_ports=``. 

53 

54 Returns a dict with keys: 

55 polylines : list of (M, 3) np.ndarray 

56 edge_segments : list of list[int] — n_seg per edge per polyline 

57 feeds : list of (polyline_idx, arclength, voltage) — 

58 one entry per excited tuple, in registration 

59 order. Suitable to pass directly to a 

60 momwire solver's feeds=... kwarg. 

61 feed_dirs : list of int — +1/-1 per feed: whether the walk 

62 traversed the authored tuple p0->p1 (+1) or 

63 p1->p0 (-1); engines use it to normalize each 

64 port's sign convention to the authored 

65 direction (issue #580). 

66 feed_wire_index : int — polyline holding the first excited 

67 segment (back-compat: feeds[0][0]) 

68 feed_arclength : float — arclength of the first feed 

69 (back-compat: feeds[0][1]) 

70 feed_voltage : complex — voltage of the first feed 

71 (back-compat: feeds[0][2]) 

72 junctions : list of list[(wire_idx, "start"|"end")] — 

73 shared-node groups, suitable to pass directly 

74 to a momwire solver's junctions=... kwarg. Empty list 

75 if every component is a simple path. 

76 end_port_junctions : dict (wire_name, "p0"|"p1") -> junction index 

77 for every requested end port (issue #579). 

78 end_port_members : dict (wire_name, "p0"|"p1") -> 

79 (polyline_idx, "start"|"end") — the momwire 

80 member each named end became; what a series 

81 node gap addresses (momwire#305, issue #898). 

82 """ 

83 if not tups: 

84 raise ValueError("no wires to translate") 

85 

86 # Build endpoint->node map and per-tuple edge list. 

87 node_of = {} 

88 nodes = [] # list of np.ndarray(3,) 

89 edges = [] # list of (a, b, n_seg, ev, tup_index) 

90 

91 def node_id(p): 

92 key = _round_point(p, eps) 

93 if key not in node_of: 

94 node_of[key] = len(nodes) 

95 nodes.append(np.asarray(p, dtype=float)) 

96 return node_of[key] 

97 

98 # Names are an optional 5th field, per-wire specs an optional 6th 

99 # (issue #388); entries may be plain 4/5-tuples or `Wire` named tuples, 

100 # normalized here at the single choke point. 

101 tup_names = [] 

102 tup_specs = [] 

103 for i, t in enumerate(tups): 

104 try: 

105 p0, p1, n_seg, ev, name, spec = as_wire(t) 

106 except ValueError as e: 

107 raise ValueError(f"tuple {i}: {e}") from None 

108 a = node_id(p0) 

109 b = node_id(p1) 

110 if a == b: 

111 raise ValueError(f"tuple {i}: degenerate edge (p0==p1 within eps)") 

112 edges.append((a, b, int(n_seg), ev, i)) 

113 tup_names.append(name) 

114 tup_specs.append(spec) 

115 

116 # End ports (issue #579): resolve each (wire_name, "p0"|"p1") to its 

117 # node id. These names identify wires only — they are excluded from 

118 # gap-feed registration below, and their nodes are forced to be 

119 # polyline boundaries so momwire can host a junction port there. 

120 end_ports = list(end_ports or []) 

121 end_port_names = set() 

122 end_port_nodes = [] # (name, which, node_id) 

123 if end_ports: 

124 name_to_tup = {} 

125 for i, nm in enumerate(tup_names): 

126 if nm is not None: 

127 name_to_tup.setdefault(nm, []).append(i) 

128 for nm, which in end_ports: 

129 if which not in ("p0", "p1"): 

130 raise ValueError( 

131 f"end port on {nm!r}: end must be 'p0' or 'p1', got {which!r}" 

132 ) 

133 idxs = name_to_tup.get(nm) 

134 if not idxs: 

135 raise ValueError( 

136 f"end port references wire name {nm!r} but no build_wires() " 

137 f"tuple carries that name" 

138 ) 

139 if len(idxs) > 1: 

140 raise ValueError( 

141 f"end port references wire name {nm!r} which is carried by " 

142 f"{len(idxs)} tuples — end-port wire names must be unique" 

143 ) 

144 a, b, *_ = edges[idxs[0]] 

145 end_port_names.add(nm) 

146 end_port_nodes.append((nm, which, a if which == "p0" else b)) 

147 

148 # adj[nid] = list of (other_node, edge_index), in registration order. 

149 adj = [[] for _ in nodes] 

150 for ei, (a, b, _, _, _) in enumerate(edges): 

151 adj[a].append((b, ei)) 

152 adj[b].append((a, ei)) 

153 

154 for nid, neigh in enumerate(adj): 

155 if len(neigh) == 0: 

156 raise ValueError(f"node {nid} is isolated") 

157 

158 # Polyline boundaries are exactly the non-degree-2 nodes: degree-1 

159 # ends an open polyline, degree>=3 is a junction that ends one 

160 # polyline and starts another. Walk every edge out of every boundary 

161 # node, threading through degree-2 nodes until the next boundary. 

162 is_boundary = [len(a) != 2 for a in adj] 

163 

164 # A wire-spec change is a polyline boundary too (issue #388): momwire 

165 # consumes one spec per wire, so a degree-2 node whose two edges carry 

166 # different specs must end one polyline and start another. The node is 

167 # then registered as a 2-entry junction below — exactly like a cycle 

168 # cut — so KCL still carries the current through it. 

169 for nid, neigh in enumerate(adj): 

170 if not is_boundary[nid]: 

171 e0, e1 = neigh[0][1], neigh[1][1] 

172 if tup_specs[edges[e0][4]] != tup_specs[edges[e1][4]]: 

173 is_boundary[nid] = True 

174 

175 # An end-port node is a boundary too (issue #579): the port must land on 

176 # a junction node, so a degree-2 node splits its chain here (registered 

177 # as a 2-entry junction below — KCL still carries the current through, 

178 # and the port injects into the shared node). Degree-1 nodes become 

179 # 1-entry junction groups (legal since momwire#172). 

180 for _nm, _which, nid in end_port_nodes: 

181 is_boundary[nid] = True 

182 

183 edge_seen = [False] * len(edges) 

184 

185 polylines = [] 

186 edge_segments = [] 

187 # One spec per polyline: uniform by construction, since a spec change 

188 # at a degree-2 node was marked as a boundary above and any other 

189 # meeting point is a junction (a boundary already). 

190 polyline_specs = [] 

191 # junction_ends[node_id] -> list of (polyline_index, "start"|"end"). 

192 # Filled as we walk; only meaningful for degree>=3 nodes, but we 

193 # collect it for all boundary nodes and filter later. 

194 junction_ends = {nid: [] for nid in range(len(nodes)) if is_boundary[nid]} 

195 # tup_index -> (polyline_index, edge_index_within) 

196 edge_to_polyline = {} 

197 # tup_index -> +1 when the walk traversed the tuple in its authored 

198 # p0 -> p1 direction, -1 when reversed. A port edge's delta-gap sign 

199 # convention (EMF direction / positive-current direction) follows the 

200 # POLYLINE direction, i.e. the walk — so this factor is what an engine 

201 # needs to normalize each port to the authored direction (issue #580: 

202 # "the port's + terminal is toward p1" is the design-visible contract). 

203 edge_walk_dir = {} 

204 

205 def walk_from(start_nid, first_edge): 

206 path_nodes = [start_nid] 

207 path_edges = [] 

208 prev_edge = None 

209 cur = start_nid 

210 next_edge = first_edge 

211 while True: 

212 edge_seen[next_edge] = True 

213 path_edges.append(next_edge) 

214 a, b, _, _, _ = edges[next_edge] 

215 nxt = b if a == cur else a 

216 path_nodes.append(nxt) 

217 cur = nxt 

218 if is_boundary[cur]: 

219 return path_nodes, path_edges 

220 prev_edge = next_edge 

221 # Degree-2 interior: take the unique outgoing edge. 

222 next_edge = None 

223 for _nb, ei in adj[cur]: 

224 if ei != prev_edge: 

225 next_edge = ei 

226 break 

227 assert next_edge is not None, f"degree-2 node {cur} had no continuation" 

228 

229 for start in range(len(nodes)): 

230 if not is_boundary[start]: 

231 continue 

232 for _nb, ei in adj[start]: 

233 if edge_seen[ei]: 

234 continue 

235 path_nodes, path_edges = walk_from(start, ei) 

236 

237 polyline_idx = len(polylines) 

238 polylines.append(np.stack([nodes[n] for n in path_nodes], axis=0)) 

239 edge_segments.append([edges[e][2] for e in path_edges]) 

240 polyline_specs.append(tup_specs[edges[path_edges[0]][4]]) 

241 for k, e in enumerate(path_edges): 

242 edge_to_polyline[edges[e][4]] = (polyline_idx, k) 

243 # Edge k was walked path_nodes[k] -> path_nodes[k+1]; 

244 # authored direction is edges[e][0] -> edges[e][1]. 

245 edge_walk_dir[edges[e][4]] = 1 if edges[e][0] == path_nodes[k] else -1 

246 junction_ends[path_nodes[0]].append((polyline_idx, "start")) 

247 junction_ends[path_nodes[-1]].append((polyline_idx, "end")) 

248 

249 # Pure-cycle components — every node degree 2, no boundary to start 

250 # the walk from — are left untouched by the loop above. Cut each at 

251 # its excited edge: the excited edge becomes one polyline (A→B), the 

252 # rest of the cycle becomes a second polyline walked B→A the long 

253 # way. The two cut nodes A and B are each registered as 2-entry 

254 # junctions so momwire's KCL enforces current continuity around the 

255 # loop. 

256 while not all(edge_seen): 

257 seed = next(i for i, seen in enumerate(edge_seen) if not seen) 

258 # Flood the component reachable from `seed` through unseen edges. 

259 comp_edges = [] 

260 stack = [seed] 

261 in_comp = {seed} 

262 while stack: 

263 ei = stack.pop() 

264 comp_edges.append(ei) 

265 edge_seen[ei] = True 

266 a, b, _, _, _ = edges[ei] 

267 for endpoint in (a, b): 

268 for _nb, eo in adj[endpoint]: 

269 if eo not in in_comp and not edge_seen[eo]: 

270 in_comp.add(eo) 

271 stack.append(eo) 

272 

273 # A "port edge" carries either a voltage (legacy build_tls path) or a 

274 # network-spec name. To open the cycle we cut ONE edge into its own 

275 # polyline and register the two cut nodes as junctions, so momwire's KCL 

276 # enforces current continuity around the loop. We PREFER to cut at a 

277 # port edge: it has to become its own polyline anyway, to host the 

278 # delta-gap feed. A PARASITIC loop has no port edge, so the cut point 

279 # is arbitrary (any edge breaks the cycle) -- we cut the first one, and 

280 # since it carries no voltage/name it simply stays a passive polyline 

281 # whose only role is to anchor the two cut-node junctions. Any extra 

282 # port edges (a feed + a termination, as in a terminated rhombic/T2FD) 

283 # stay inside the long-way polyline and are registered as feeds by 

284 # arclength below, so a multi-port loop is handled too. 

285 excited_in_comp = [ 

286 ei 

287 for ei in comp_edges 

288 if edges[ei][3] is not None or tup_names[edges[ei][4]] is not None 

289 ] 

290 cut_ei = excited_in_comp[0] if excited_in_comp else comp_edges[0] 

291 cut_a, cut_b, cut_n_seg, _, cut_tup_idx = edges[cut_ei] 

292 

293 # Polyline 0: the cut edge alone, A → B. It hosts the delta-gap feed 

294 # when the cut was a port edge; for a parasitic loop it is just passive. 

295 cut_pl_idx = len(polylines) 

296 polylines.append(np.stack([nodes[cut_a], nodes[cut_b]], axis=0)) 

297 edge_segments.append([cut_n_seg]) 

298 polyline_specs.append(tup_specs[cut_tup_idx]) 

299 edge_to_polyline[cut_tup_idx] = (cut_pl_idx, 0) 

300 # The cut polyline is stacked [A, B] = authored p0 -> p1. 

301 edge_walk_dir[cut_tup_idx] = 1 

302 

303 # Polyline 1 (long way): walk B → ... → A via the remaining edges. 

304 # The cut nodes are now polyline boundaries; the walker stops there. 

305 is_boundary[cut_a] = True 

306 is_boundary[cut_b] = True 

307 junction_ends.setdefault(cut_a, []) 

308 junction_ends.setdefault(cut_b, []) 

309 

310 # Undo the flood-fill's seen marks on the cycle remainder so the 

311 # walker can traverse them. Keep the cut edge marked since it's 

312 # already become polyline 0. 

313 for ei in comp_edges: 

314 if ei != cut_ei: 

315 edge_seen[ei] = False 

316 

317 first = next( 

318 (eo for _nb, eo in adj[cut_b] if eo != cut_ei and not edge_seen[eo]), 

319 None, 

320 ) 

321 # In a pure cycle every node has degree 2, so there's exactly one 

322 # remaining edge at cut_b after consuming the cut edge. 

323 assert first is not None, "cycle cut left no continuation" 

324 path_nodes, path_edges = walk_from(cut_b, first) 

325 loop_pl_idx = len(polylines) 

326 polylines.append(np.stack([nodes[n] for n in path_nodes], axis=0)) 

327 edge_segments.append([edges[e][2] for e in path_edges]) 

328 polyline_specs.append(tup_specs[edges[path_edges[0]][4]]) 

329 for k, e in enumerate(path_edges): 

330 edge_to_polyline[edges[e][4]] = (loop_pl_idx, k) 

331 edge_walk_dir[edges[e][4]] = 1 if edges[e][0] == path_nodes[k] else -1 

332 

333 # Register the cut endpoints as junctions: the cut polyline has 

334 # path [A, B] so its start=A, end=B; loop polyline was walked 

335 # B → A so its start=B, end=A. 

336 junction_ends[cut_a].append((cut_pl_idx, "start")) 

337 junction_ends[cut_a].append((loop_pl_idx, "end")) 

338 junction_ends[cut_b].append((cut_pl_idx, "end")) 

339 junction_ends[cut_b].append((loop_pl_idx, "start")) 

340 

341 # Junctions = nodes where >= 2 polylines meet. Single-end records 

342 # (degree-1 free ends) and lone polyline starts aren't junctions — 

343 # EXCEPT end-port nodes (issue #579), whose group is emitted even with a 

344 # single member so a momwire junction port can live there. 

345 end_port_nids = {nid for _nm, _w, nid in end_port_nodes} 

346 junctions = [] 

347 junction_index_of_node = {} 

348 for nid, ends in junction_ends.items(): 

349 if len(ends) >= 2 or (nid in end_port_nids and len(ends) >= 1): 

350 junction_index_of_node[nid] = len(junctions) 

351 junctions.append(ends) 

352 end_port_junctions = { 

353 (nm, which): junction_index_of_node[nid] for nm, which, nid in end_port_nodes 

354 } 

355 # The MEMBER each named end became: (polyline_idx, "start"|"end") in 

356 # momwire's numbering — what a series node gap (momwire#305, issue 

357 # #898's PortAtVertex) addresses, where a junction PORT (#579) needs 

358 # only the junction index above. Derivation: the named endpoint sits at 

359 # node position edge_pos or edge_pos+1 of its polyline's walk depending 

360 # on whether the walk traversed the authored tuple p0→p1; a forced 

361 # boundary means that position is the walk's first or last node. 

362 end_port_members = {} 

363 if end_port_nodes: 

364 name_to_tup = {} 

365 for i, nm in enumerate(tup_names): 

366 if nm is not None: 

367 name_to_tup.setdefault(nm, []).append(i) 

368 for nm, which, _nid in end_port_nodes: 

369 tup_index = name_to_tup[nm][0] 

370 pl_idx, edge_pos = edge_to_polyline[tup_index] 

371 walk_dir = edge_walk_dir[tup_index] 

372 pos = edge_pos + (0 if (which == "p0") == (walk_dir == 1) else 1) 

373 n_edges_pl = len(edge_segments[pl_idx]) 

374 assert pos in (0, n_edges_pl), (nm, which, pos, n_edges_pl) 

375 end_port_members[(nm, which)] = ( 

376 pl_idx, 

377 "start" if pos == 0 else "end", 

378 ) 

379 

380 # Locate the excitation(s) and convert each to (polyline_idx, 

381 # arclength, voltage). PyNEC feeds at segment `(n_seg+1)//2` of the 

382 # excited tuple — the middle segment 1-indexed, i.e. the physical 

383 # midpoint of the wire. The excited tuple is one edge of its 

384 # polyline; feed at that edge's midpoint. 

385 # 

386 # A tuple is a feed if either (a) it has a non-None ex value (legacy 

387 # voltage-driven feed) or (b) it has a non-None name (network-port 

388 # placeholder, voltage gets set later by `build_network()`). 

389 feeds = [] 

390 feed_names = [] 

391 feed_edges = [] 

392 feed_dirs = [] 

393 for tup_index, edge in enumerate(edges): 

394 voltage = edge[3] 

395 name = tup_names[tup_index] 

396 # An end-port name identifies its wire for the junction port only — 

397 # no gap is cut in the wire (issue #579), so it is not a port edge. 

398 if name in end_port_names: 

399 name = None 

400 if voltage is None and name is None: 

401 continue 

402 feed_pl, feed_edge_idx = edge_to_polyline[tup_index] 

403 polyline = polylines[feed_pl] 

404 edge_lengths = np.linalg.norm(np.diff(polyline, axis=0), axis=1) 

405 feed_arclength = float( 

406 edge_lengths[:feed_edge_idx].sum() + 0.5 * edge_lengths[feed_edge_idx] 

407 ) 

408 feeds.append( 

409 (feed_pl, feed_arclength, complex(voltage if voltage is not None else 0)) 

410 ) 

411 feed_names.append(name) 

412 feed_edges.append((feed_pl, feed_edge_idx)) 

413 feed_dirs.append(edge_walk_dir[tup_index]) 

414 

415 if not feeds and not end_port_nodes: 

416 raise ValueError("no excitation found in wire list") 

417 

418 return { 

419 "polylines": polylines, 

420 "edge_segments": edge_segments, 

421 "polyline_specs": polyline_specs, 

422 "feeds": feeds, 

423 "feed_names": feed_names, 

424 # Which (polyline, edge) each feed sits on — a distributed port 

425 # (issue #477) needs the whole edge's extent, not just its midpoint. 

426 "feed_edges": feed_edges, 

427 # +1/-1 per feed: whether the walk traversed the authored tuple 

428 # p0 -> p1 (+1) or p1 -> p0 (-1). A feed's solver-side sign 

429 # convention follows the walk; engines multiply by this factor to 

430 # normalize every port to the AUTHORED direction (issue #580). 

431 "feed_dirs": feed_dirs, 

432 # Back-compat scalars — first feed (None when the design is driven 

433 # entirely through end ports, issue #579). 

434 "feed_wire_index": feeds[0][0] if feeds else None, 

435 "feed_arclength": feeds[0][1] if feeds else None, 

436 "feed_voltage": feeds[0][2] if feeds else None, 

437 "junctions": junctions, 

438 "end_port_junctions": end_port_junctions, 

439 "end_port_members": end_port_members, 

440 }
