1"""Holding an objective at every sweep point (AK#1757, sweep-framework step 6). 

2 

3An `analyses.Hold` says "keep it resonant (or matched) while this knob 

4sweeps": at each swept x the hold's knobs are re-solved for its objective, 

5and the analysis's views are drawn at that optimised point. Nothing here is a 

6second optimizer. Each point IS one call of the workbench's own 

7``web.optimize.optimize`` (its scalar secant / bracket for ``resonance`` on 

8one knob, its two-component Newton for ``match_z0`` on two), with: 

9 

10- ``warm=True`` when the point starts from the previous point's solution 

11 (``Hold.warm_start``, continuation, as the workbench's track-while-drag 

12 does), so the two-knob path tries Newton from that start before sampling 

13 the box; 

14- ``fallback=False``, so a point with no root is a GAP, not a simplex's 

15 best effort drawn as a plausible wrong value. 

16 

17The rules (docs/design/sweep-framework-step6.md): 

18 

191. A point the optimizer does not converge at (its root path's own verdict), 

20 or whose solved residual is over `HOLD_TOL`, is a gap with its reason. The 

21 next point warm-starts from the last CONVERGED point. After 

22 `FAILS_BEFORE_COLD` failures in a row, each remaining point cold-starts 

23 from the defaults once (a converged one resumes the warm chain); one that 

24 fails then is given up on, a gap. 

252. The bounds are the knobs' own ``ui_params`` min/max, as the workbench's 

26 optimizer reads them (its knob menu's opt extents default to the slider's). 

27 A held knob without both is refused by name. 

283. What the optimizer cannot serve is refused by name, not run: ``swr`` (a 

29 minimisation, no root and so no verdict a point could be drawn on), a 

30 frequency or density sweep, a map, a multi-feed design. 

31 

32Framework-free: the CLI (`analysis_run`) and the workbench's ``/param_sweep`` 

33(``web.server``) both call `hold_line`, with a ``solve_fn`` of their own. 

34"""

44#: The residual (ohms) a converged point must be within: the workbench's 

45#: hold tolerance (``web.tracker.TRACK_TOL``, the user-facing one the #1202 

46#: study measured against). The optimizer's own stop is far tighter 

47#: (``web.optimize._ROOT_FTOL``, 1e-3 ohm); this is the backstop for a root 

48#: called on a step size (``xtol``) whose solved residual is still large. 

49HOLD_TOL = 1.0 

50 

51#: Failures in a row after which the remaining points cold-start (rule 1). 

52FAILS_BEFORE_COLD = 3

59def analysis_refusal(a: an.Analysis, builder=None) -> str | None: 

60 """Why ``a``'s hold cannot run at all (beyond `an.problems`), or None. 

61 ``builder`` (the analysis's design) resolves a role or a knob named 

62 directly, to tell a density knob; without it only the spec is read.""" 

63 h = a.hold 

64 if h is None: 

65 return None 

66 if h.objective == "swr": 

67 return ( 

68 "hold swr: SWR is a minimisation, not a root, so the optimizer has " 

69 "no verdict a held point could be drawn or left a gap on; hold " 

70 "resonance (one knob) or match_z0 (two)" 

71 ) 

72 if len(a.sweeps) != 1: 

73 return ( 

74 "a hold on a two-sweep map: not in v1 (sweep-framework step 6 holds " 

75 "along one swept knob); cross the second knob as a family" 

76 ) 

77 s = a.sweep 

78 if s.knob == an.FREQUENCY: 

79 return ( 

80 "a hold on a frequency sweep: not in v1 (the frequency sweep is one " 

81 "build solved across the band, and a hold rebuilds and re-solves at " 

82 "every point; sweep a knob, or hold at the measurement frequency)" 

83 ) 

84 dens = an.density_knob(builder) if builder is not None else None 

85 swept = an.resolve(s.knob, builder).knob if builder is not None else None 

86 if s.knob == an.DENSITY or (swept is not None and swept in (dens, "nominal_nsegs")): 

87 return ( 

88 "a hold on a density ladder: not in v1 (a convergence study reads " 

89 "Z as the mesh refines, and a hold would move the design under it)" 

90 ) 

91 return None

94def free_of(hold: an.Hold, builder) -> list[dict]: 

95 """The hold's knobs on ``builder`` as the optimizer takes them, 

96 ``[{name, min, max}]``, bounded by each knob's ``ui_params`` min/max 

97 (rule 2); `HoldRefused` by name for a knob that does not resolve, the 

98 density knob, or a knob with no range of its own.""" 

99 params = an._params(builder) 

100 ui = an._ui(params) 

101 dens = an.density_knob(builder) 

102 out = [] 

103 for k in hold.adjust: 

104 r = an.resolve(k, builder) 

105 if r.knob is None: 

106 raise HoldRefused(r.reason) 

107 if r.knob in (dens, "nominal_nsegs"): 

108 raise HoldRefused( 

109 f"the hold adjusts {r.knob}, the density knob: the engine holds " 

110 "every solve at its own density, so it is not a knob to hold" 

111 ) 

112 meta = ui.get(r.knob) 

113 lo = meta.get("min") if isinstance(meta, Mapping) else None 

114 hi = meta.get("max") if isinstance(meta, Mapping) else None 

115 if lo is None or hi is None or not float(lo) < float(hi): 

116 raise HoldRefused( 

117 f"the hold adjusts {r.knob}, which has no ui_params min/max to " 

118 "bound it: the optimizer searches within a knob's own range; " 

119 f'give {r.knob} a "min" and "max" in ui_params' 

120 ) 

121 out.append({"name": r.knob, "min": float(lo), "max": float(hi)}) 

122 return out

126class HeldPoint: 

127 """One swept point of a held line. ``start`` is where the optimizer 

128 started (``cold``: the defaults; else the last converged point's knobs); 

129 ``params`` and ``z`` are its answer (port 0), the best SOLVED point even 

130 when it did not converge; ``converged`` False makes it a gap, with 

131 ``reason``.""" 

132 

133 x: float 

134 start: dict 

135 cold: bool 

136 params: dict 

137 z: complex 

138 residual: float | None 

139 converged: bool 

140 reason: str | None 

141 method: str 

142 n_solves: int

145def _gap_reason(res: Mapping, free: Sequence[Mapping], objective: str) -> str | None: 

146 """Why ``res`` (an ``optimize`` result) is a gap, or None when it holds.""" 

147 resid = res.get("residual_after") 

148 if res.get("converged"): 

149 if resid is not None and resid <= HOLD_TOL: 

150 return None 

151 return ( 

152 f"the root search stopped at a residual of {resid:.3g} ohm, over the " 

153 f"hold's {HOLD_TOL:g} ohm" 

154 ) 

155 why = res.get("root_reason") or "no root path ran" 

156 words = { 

157 "no-sign-change": "X does not change sign anywhere in the knob's range", 

158 "no-crossing": "R = Z0 and X = 0 do not cross within the knobs' ranges", 

159 "stalled": "the root search stalled", 

160 "singular": "the knobs do not move R and X independently here", 

161 "flat": "X does not respond to the knob", 

162 "budget": "the optimizer ran out of solves", 

163 "multi-feed": "the design has several feeds", 

164 }.get(why, why) 

165 target = "resonance" if objective == "resonance" else "match" 

166 text = f"no {target} held ({words})" 

167 params = res.get("params") or {} 

168 edge = [] 

169 for f in free: 

170 v = params.get(f["name"]) 

171 if v is None: 

172 continue 

173 span = f["max"] - f["min"] 

174 if v <= f["min"] + 0.002 * span: 

175 edge.append(f"{f['name']} at its ui_params min {f['min']:g}") 

176 elif v >= f["max"] - 0.002 * span: 

177 edge.append(f"{f['name']} at its ui_params max {f['max']:g}") 

178 if edge: 

179 text += "; " + ", ".join(edge) 

180 return text

183def hold_point( 

184 x, 

185 knob: str, 

186 start: Mapping, 

187 free: Sequence[Mapping], 

188 objective: str, 

189 *, 

190 solve_fn: Callable[[dict], dict],

193 max_evals: int | None = None,

195 """ONE held point, exactly as `hold_line` runs it: ``(request, optimize 

196 result)``. The request is ``base`` with the swept ``knob`` at ``x`` and

201 req = {**(base or {}), knob: x, **start} 

202 res = optimize(

207 max_evals=max_evals, 

208 warm=warm, 

209 fallback=False, 

210 ) 

211 return req, res

214def hold_line( 

215 xs: Sequence, 

216 knob: str, 

217 free: Sequence[Mapping], 

218 objective: str, 

219 *, 

220 solve_fn: Callable[[dict], dict], 

221 defaults: Mapping, 

222 warm_start: bool = True, 

223 base: Mapping | None = None, 

224 max_evals: int | None = None, 

225 on_point: Callable[[HeldPoint], None] | None = None, 

226) -> list[HeldPoint]: 

227 """The held line along ``xs`` of ``knob`` (module docstring, rule 1). 

228 ``defaults`` are the held knobs' starting values (the design's own); 

229 ``on_point`` sees each point as it lands (the workbench streams them).""" 

230 points: list[HeldPoint] = [] 

231 last: dict | None = None 

232 fails = 0 

233 cold_mode = False 

234 failed: list[Exception] = [] 

235 for x in xs: 

236 warm = warm_start and last is not None and not cold_mode 

237 start = dict(last) if warm else {f["name"]: defaults[f["name"]] for f in free} 

238 try: 

239 _req, res = hold_point( 

240 x, 

241 knob, 

242 start, 

243 free, 

244 objective, 

245 solve_fn=solve_fn, 

246 warm=warm, 

247 base=base, 

248 max_evals=max_evals, 

249 ) 

250 except HoldRefused: 

251 raise 

252 except (ValueError, NotImplementedError) as e: 

253 # A solve the search asked for failed: a candidate the engine 

254 # refuses (a wire end in the ground plane, say). The optimizer has 

255 # no notion of a failed probe, so the POINT failed: a gap with the 

256 # engine's words, never a value. Every point failing is the 

257 # engine refusing the cell, re-raised below. 

258 failed.append(e) 

259 pt = HeldPoint( 

260 x=x, 

261 start=start, 

262 cold=not warm, 

263 params=dict(start), 

264 z=complex(math.nan, math.nan), 

265 residual=None, 

266 converged=False, 

267 reason=f"a solve inside the search failed: {_short(e)}", 

268 method="failed", 

269 n_solves=0, 

270 ) 

271 else: 

272 reason = _gap_reason(res, free, objective) 

273 m = res["metrics_after"] 

274 pt = HeldPoint( 

275 x=x, 

276 start=start, 

277 cold=not warm, 

278 params=dict(res["params"]), 

279 z=complex(m["z_in_re"], m["z_in_im"]), 

280 residual=res.get("residual_after"), 

281 converged=reason is None, 

282 reason=reason, 

283 method=str(res.get("method")), 

284 n_solves=int(res.get("n_solves") or 0), 

285 ) 

286 points.append(pt) 

287 if pt.converged: 

288 last, fails, cold_mode = pt.params, 0, False 

289 else: 

290 fails += 1 

291 if warm_start and fails >= FAILS_BEFORE_COLD: 

292 # Rule 1's recovery: from here each point cold-starts from 

293 # the defaults, once; a converged one resumes the warm chain. 

294 cold_mode = True 

295 if on_point is not None: 

296 on_point(pt) 

297 if points and len(failed) == len(points): 

298 raise failed[0] 

299 return points

302def _short(e: Exception, limit: int = 160) -> str: 

303 """An engine's refusal, cut to its first sentence (they run long).""" 

304 text = " ".join(str(e).split()) 

305 cut = text.find(": ", 40) 

306 if 0 < cut < limit: 

307 text = text[:cut] 

308 return text if len(text) <= limit else text[: limit - 1] + "…"

311def builder_solve_fn(builder, factory: Callable, z0: float) -> Callable[[dict], dict]: 

312 """The CLI's ``solve_fn``: each knob of the request set on ``builder`` 

313 (a cell's own), solved through the cell's engine ``factory``, as 

314 ``sweep._solve_at`` solves a knob sweep, and answered in the optimizer's 

315 response shape. A design with several feeds is refused by name: the 

316 optimizer's root paths hold one feed's Z.""" 

317 params = an._params(builder) 

318 

319 def solve(req: dict) -> dict: 

320 for k, v in req.items(): 

321 if k in params: 

322 setattr(builder, k, v) 

323 zs = factory(builder).impedance() 

324 zs = list(zs) if hasattr(zs, "__len__") else [zs] 

325 if len(zs) != 1: 

326 raise HoldRefused( 

327 f"this design drives {len(zs)} feeds: a hold drives one feed's " 

328 "Z to its target (the optimizer's root paths are single-feed)" 

329 ) 

330 z = complex(zs[0]) 

331 return {"z_in_re": z.real, "z_in_im": z.imag, "z0_ohms": float(z0)} 

332 

333 return solve
