1"""Locating a user-supplied engine binary — shared by the NEC-5 and NEC-2 wrappers. 

2 

3Both wrappers drive an executable antennaknobs never ships: NEC-5 because it is 

4licensed (#825), NEC-2 because bundling one would make the zip a GPL combined 

5work (#1354). The question "where is it" has the same two answers in both cases 

6— an explicit path from the caller, else an environment variable — and the same 

7two non-answers, which is why this is one function rather than two. 

8 

9The distinction the return value draws is deliberate and is the reason a bare 

10`os.environ.get` will not do: **None means "no binary here", never "the variable 

11was empty"**, so a variable pointing at a directory, a missing file or a 

12non-executable is an absence rather than a path a caller will later fail to 

13run. What None does NOT mean is "this is the wrong program" — that needs a run, 

14and each wrapper's own probe is where that lives (`probe_nec5`, `probe_nec2`). 

15""" 

16 

17from __future__ import annotations 

18 

19import os 

20import subprocess 

21import time 

22from contextlib import contextmanager 

23from contextvars import ContextVar 

24from pathlib import Path 

25 

26 

27def find_exe(env_var: str, explicit: str | None = None) -> str | None: 

28 """The executable named explicitly, else by ``$env_var``, else by the 

29 settings file's ``[engines]`` table (AK#1492); None if none resolves to an 

30 executable file.""" 

31 from ..settings_file import engine_exe 

32 

33 cand = explicit or os.environ.get(env_var) or engine_exe(env_var) 

34 if not cand: 

35 return None 

36 p = Path(cand).expanduser() 

37 if p.is_file() and os.access(p, os.X_OK): 

38 return str(p) 

39 return None 

40 

41 

42# The cancel token of the solve this thread is running, if any (AK#1712). 

43# 

44# An external engine is a child process, so the momwire token the server's 

45# lane hands a solve cannot reach it through a kernel checkpoint. Threading a 

46# `cancel=` argument down through every backend, adapter and engine method 

47# that can end in a deck run would touch dozens of signatures for one poll 

48# loop at the bottom, so the server binds the token for the duration of the 

49# call instead (`cancel_scope`) and `run_exe` — the one place a binary is 

50# launched — reads it. A context variable, not a global: each threadpool 

51# worker runs its own solve, and a token must never reach another session's 

52# process. 

53_CANCEL: ContextVar = ContextVar("antennaknobs_engine_cancel", default=None) 

54 

55# How often a running binary checks its token. The cost is one wakeup; the 

56# benefit bounds how long a cancelled NEC-5 fill keeps a core. 

57_POLL_S = 0.1 

58 

59 

60@contextmanager 

61def cancel_scope(token): 

62 """Bind ``token`` (anything with a ``cancelled`` flag, normally a 

63 ``momwire.CancelToken``) as the cancel for every binary launched inside 

64 the block on this thread. ``None`` is accepted and binds nothing.""" 

65 reset = _CANCEL.set(token) 

66 try: 

67 yield 

68 finally: 

69 _CANCEL.reset(reset)

72def run_exe( 

73 argv: list[str], *, stdin_text: str, cwd: str, timeout: float 

74) -> subprocess.CompletedProcess: 

75 """``subprocess.run(argv, input=stdin_text, text=True, capture_output=True, ...)`` 

76 that also dies with the solve that asked for it. 

77 

78 With no token bound this IS ``subprocess.run`` (the availability probes, 

79 the CLI). With one bound, the process is polled every ``_POLL_S`` and 

80 killed the moment the token trips, raising ``momwire.SolveAborted`` — the 

81 same exception a cancelled momwire solve raises, so every caller's abort 

82 handling applies unchanged. ``timeout`` keeps ``subprocess.run``'s 

83 meaning, ``TimeoutExpired`` included. 

84 """ 

85 token = _CANCEL.get() 

86 if token is None: 

87 return subprocess.run( 

88 argv, 

89 input=stdin_text, 

90 text=True, 

91 capture_output=True, 

92 cwd=cwd, 

93 timeout=timeout, 

94 ) 

95 import momwire 

96 

97 if token.cancelled: 

98 raise momwire.SolveAborted() 

99 deadline = time.monotonic() + timeout 

100 with subprocess.Popen( 

101 argv, 

102 stdin=subprocess.PIPE, 

103 stdout=subprocess.PIPE, 

104 stderr=subprocess.PIPE, 

105 text=True, 

106 cwd=cwd, 

107 ) as proc: 

108 pending_input: str | None = stdin_text 

109 while True: 

110 try: 

111 out, err = proc.communicate(pending_input, timeout=_POLL_S) 

112 break 

113 except subprocess.TimeoutExpired: 

114 # communicate() refuses input once it has started; the first 

115 # call already handed it over. 

116 pending_input = None 

117 if token.cancelled: 

118 proc.kill() 

119 proc.communicate() 

120 raise momwire.SolveAborted() 

121 if time.monotonic() >= deadline: 

122 proc.kill() 

123 out, err = proc.communicate() 

124 raise subprocess.TimeoutExpired(argv, timeout, output=out, stderr=err) 

125 return subprocess.CompletedProcess(argv, proc.returncode, out, err)
