1"""What only the hosted instance does: crash reports from visitors' browsers, 

2session pinning across a fleet of machines, and usage counters. 

3 

4Everything here is dormant on a local workbench. Crash reports and counters 

5are gated on ``ANTENNAKNOBS_HOSTED`` (server._HOSTED), so nothing a local 

6install does ever reports anywhere (the #846 beacon rule: nothing phones home 

7from a local install). Pinning is gated on ``FLY_MACHINE_ID``, which Fly sets 

8on its machines and nothing else does. 

9 

10Crash reports (AK#1851) 

11----------------------- 

12The frontend's error boundary and its global ``error`` / 

13``unhandledrejection`` handlers POST ``/client-error``; the hosted server logs 

14ONE line per report and stores nothing. The line carries exactly three 

15fields, each bounded and scrubbed: 

16 

17- ``message``: the error's message; 

18- ``component_stack``: React's component stack, when the boundary caught it; 

19- ``version``: the antennaknobs version the page was served by. 

20 

21No IP, no URL, no user agent, no deck text, no knob values. The ``?deck=`` 

22link carries a visitor's whole deck, so anything shaped like a URL query or 

23fragment is cut out of both text fields before the line is written (the 

24frontend cuts it too; the server does not trust that). The client address 

25keys the per-client rate limit in memory and is never written. 

26 

27Session pinning (AK#405) 

28------------------------ 

29Three things assume one page's requests reach one machine: the lane rule 

30(one solve at a time per session, ``LaneRegistry`` is per process), the 

31opened-deck cache (``deck.<sha12>`` is parsed into one machine's memory), and 

32the per-machine limits. With more than one machine, Fly's edge routes each 

33request to the nearest healthy one, which a session's requests need not share. 

34 

35So the page pins itself, and the server only says where it is: 

36 

37- every HTTP response carries ``X-AK-Machine: <FLY_MACHINE_ID>``; 

38- the ``/ws`` channel's first message is ``{"_kind": "machine", "id": ...}``; 

39- the page sends ``fly-force-instance-id: <id>`` on its requests, which Fly's 

40 proxy routes to that machine (https://fly.io/docs/networking/dynamic-request-routing/); 

41- a browser cannot set a header on a WebSocket, so the page asks for its 

42 machine in the ``/ws`` URL (``?fly_instance=<id>``), and a machine that is 

43 not that one answers the upgrade with ``fly-replay: instance=<id>`` instead 

44 of accepting it, which the proxy replays to the right machine (the same 

45 page: "an application returning fly-replay headers should not negotiate a 

46 web socket upgrade itself"). 

47 

48Off Fly there is no machine id: no header, no first message, no replay, and 

49the page never pins, so every request is byte-identical to before. 

50 

51Usage counters (AK#405) 

52----------------------- 

53Prometheus counters, per machine, served on their own port (``[metrics]`` in 

54fly.toml points Fly's scraper at it; it is not behind the public 

55``[http_service]``). No IP, no session id, no deck content: every label is 

56drawn from a fixed roster, so the series count is bounded by the catalog and 

57the engine and ground rosters. 

58 

59- ``ak_solves_total{design, engine, ground}``: live solves the server paid 

60 for (cache hits and failures excluded); 

61- ``ak_sessions_total{kind}``: live-solve connections, by what their first 

62 solve was of (``catalog`` / ``user`` / ``deck``); 

63- ``ak_deck_opens_total{dialect, outcome}``: ``POST /deck``, by the dialect it 

64 was read in and ``ok`` / ``refused`` / ``busy``; 

65- ``ak_refusals_total{reason}``: requests the server turned away for capacity 

66 or limits, by why (``REFUSAL_REASONS``, a closed set, one count per 

67 refusal at the point it is made): ``busy`` (the deck slot is held by 

68 another client), ``deck_watchdog`` (an opened deck's solve killed at the 

69 hosted solve budget), ``deck_cap`` (a deck over the byte, segment or wire 

70 caps), ``deck_rate`` (opens per minute), ``sweep_budget`` (a sweep stopped 

71 at the hosted wall-time budget; its partial results stand), 

72 ``optimize_budget`` (an optimize run stopped at its time budget), 

73 ``cost_refused`` (the cost model refused a solve or batch outright) and 

74 ``cost_withheld`` (a poor-match batch held until the client approves it). 

75 A malformed deck, a design build error and a superseded or cancelled solve 

76 are not refusals and are not counted. ``ak_deck_opens_total``'s ``busy`` 

77 outcome overlaps ``deck_rate`` for ``POST /deck`` (that counter is about 

78 opens, this one about turning away, and the two are never summed). 

79 

80``design`` is the catalog name, or the literal ``deck`` or ``user``, never a 

81file name or hash. Off unless hosted. 

82"""

98# Field caps. A message is a sentence; a component stack is a few dozen 

99# component frames. Longer is either a runaway or someone's payload. 

100MESSAGE_MAX = 500 

101STACK_MAX = 4000 

102# The body the endpoint reads at all (the two caps above, JSON-escaped, with 

103# room to spare). Larger is refused unread. 

104BODY_MAX = 16 * 1024 

105 

106# Per-client and fleet-wide budgets, per minute. The client side already sends 

107# at most a few per page load; these bound what a script could make the log 

108# say, per address and in total. 

109PER_CLIENT_PER_MIN = 6 

110TOTAL_PER_MIN = 120 

111 

112_VERSION_RE = re.compile(r"[0-9A-Za-z.+\-]{1,40}") 

113# A URL's query and fragment, and any bare `?k=v...` / `deck=...` run: the 

114# opened-deck link carries a whole deck in its query. 

115_URL_TAIL_RE = re.compile( 

116 r"((?:[a-z][a-z0-9+.\-]*:)?//[^\s?#)'\"]*)[?#][^\s)'\"]*", re.I 

117) 

118_QUERY_RE = re.compile(r"\?[^\s)'\"]*=[^\s)'\"]*") 

119_DECK_PARAM_RE = re.compile(r"\bdeck=[^\s&)'\"]*") 

120

122def scrub(text: str) -> str: 

123 """``text`` with every URL query and fragment, and any deck parameter, 

124 cut out.""" 

125 text = _URL_TAIL_RE.sub(r"\1", text) 

126 text = _QUERY_RE.sub("", text) 

127 return _DECK_PARAM_RE.sub("deck=…", text) 

128 

129 

130def _field(value, cap: int) -> str: 

131 if not isinstance(value, str): 

132 return "" 

133 text = scrub(value) 

134 return text if len(text) <= cap else text[:cap] + "…"

137def client_error_record(payload) -> dict: 

138 """The three fields a report is logged with, bounded and scrubbed. 

139 Anything else in ``payload`` is dropped unread.""" 

140 if not isinstance(payload, dict): 

141 payload = {} 

142 version = payload.get("version") 

143 return { 

144 "message": _field(payload.get("message"), MESSAGE_MAX), 

145 "component_stack": _field(payload.get("component_stack"), STACK_MAX), 

146 "version": ( 

147 version 

148 if isinstance(version, str) and _VERSION_RE.fullmatch(version) 

149 else "unknown" 

150 ), 

151 }

154class ClientErrorSink: 

155 """Rate-limits and logs crash reports. One per process.""" 

156 

157 def __init__( 

158 self, 

159 per_client: int = PER_CLIENT_PER_MIN, 

160 total: int = TOTAL_PER_MIN, 

161 ) -> None: 

162 self._per_client = _decks.RateLimiter(per_client) 

163 self._total = _decks.RateLimiter(total) 

164 

165 def report(self, payload, client: str) -> bool: 

166 """Log ``payload``'s record unless ``client`` (an address, used as a 

167 key only) or the fleet is over its budget. True if logged.""" 

168 if not self._per_client.allow(client) or not self._total.allow("*"): 

169 return False 

170 record = client_error_record(payload) 

171 _logger.warning("client-error: %s", json.dumps(record, ensure_ascii=False)) 

172 return True

199def _ws_pin(scope) -> str | None: 

200 raw = scope.get("query_string", b"").decode("latin-1") 

201 want = parse_qs(raw).get(WS_PIN_PARAM, [None])[0] 

202 return want if want and _MACHINE_ID_RE.fullmatch(want) else None

205class HostingMiddleware: 

206 """Pure ASGI. On Fly: stamps ``X-AK-Machine`` on every HTTP response and 

207 replays a ``/ws`` upgrade pinned to another machine there. Off Fly it 

208 passes HTTP and WebSocket scopes through untouched. 

209 

210 ``on_startup`` runs once at the server's lifespan startup (the hosted 

211 metrics endpoint), never at import, so importing the app in a test or a 

212 second interpreter binds no port.""" 

213 

214 def __init__(self, app, on_startup=None) -> None: 

215 self.app = app 

216 self.on_startup = on_startup 

217 

218 async def __call__(self, scope, receive, send): 

219 kind = scope["type"] 

220 if kind == "lifespan": 

221 if self.on_startup is not None: 

222 self.on_startup() 

223 await self.app(scope, receive, send) 

224 return 

225 mid = machine_id() 

226 if mid is None: 

227 await self.app(scope, receive, send) 

228 return 

229 if kind == "websocket": 

230 want = _ws_pin(scope) 

231 if want is not None and want != mid: 

232 await self._replay(scope, send, want) 

233 return 

234 await self.app(scope, receive, send) 

235 return 

236 if kind != "http": 

237 await self.app(scope, receive, send) 

238 return 

239 stamp = (MACHINE_HEADER.encode(), mid.encode()) 

240 

241 async def stamped(message): 

242 if message["type"] == "http.response.start": 

243 message = dict(message) 

244 message["headers"] = [*message.get("headers", []), stamp] 

245 await send(message) 

246 

247 await self.app(scope, receive, stamped) 

248 

249 @staticmethod 

250 async def _replay(scope, send, want: str) -> None: 

251 if "websocket.http.response" in scope.get("extensions", {}): 

252 await send( 

253 { 

254 "type": "websocket.http.response.start", 

255 "status": 409, 

256 "headers": [ 

257 (b"fly-replay", f"instance={want}".encode()), 

258 (b"content-length", b"0"), 

259 ], 

260 } 

261 ) 

262 await send({"type": "websocket.http.response.body", "body": b""}) 

263 else: 

264 # A server without the denial-response extension cannot replay; 

265 # refusing the upgrade sends the page back unpinned. 

266 await send({"type": "websocket.close", "code": 1013})

370class UsageCounters: 

371 """The hosted instance's counters. ``enabled=False`` (a local workbench) 

372 makes every ``inc`` a no-op and the exposition empty.""" 

373 

374 def __init__(self, enabled: bool) -> None: 

375 self.enabled = enabled 

376 self._lock = threading.Lock() 

377 self._counts: dict[str, dict[tuple[str, ...], int]] = {n: {} for n in _HELP} 

378 

379 def inc(self, name: str, *labels: str) -> None: 

380 names = _HELP[name][1] 

381 if len(labels) != len(names): 

382 raise ValueError(f"{name} takes labels {names}, got {labels}") 

383 # Checked before the enabled test, so a local run and a test both 

384 # trip on a reason outside the set. 

385 if name == "ak_refusals_total" and labels[0] not in REFUSAL_REASONS: 

386 raise ValueError(f"unknown refusal reason {labels[0]!r}") 

387 if not self.enabled: 

388 return 

389 with self._lock: 

390 series = self._counts[name] 

391 series[labels] = series.get(labels, 0) + 1 

392 

393 def value(self, name: str, *labels: str) -> int: 

394 with self._lock: 

395 return self._counts[name].get(labels, 0) 

396 

397 def series(self, name: str) -> dict[tuple[str, ...], int]: 

398 with self._lock: 

399 return dict(self._counts[name]) 

400 

401 def exposition(self) -> str: 

402 """The Prometheus text format (version 0.0.4).""" 

403 if not self.enabled: 

404 return "" 

405 lines = [] 

406 with self._lock: 

407 for name, (help_text, names) in _HELP.items(): 

408 lines.append(f"# HELP {name} {help_text}") 

409 lines.append(f"# TYPE {name} counter") 

410 for labels, n in sorted(self._counts[name].items()): 

411 body = ",".join( 

412 f'{k}="{_escape(v)}"' 

413 for k, v in zip(names, labels, strict=True) 

414 ) 

415 lines.append(f"{name}{{{body}}} {n}") 

416 return "\n".join(lines) + "\n"

429def serve_metrics( 

430 counters: UsageCounters, host: str = "0.0.0.0", port: int | None = None 

431) -> ThreadingHTTPServer: 

432 """Serve ``counters`` at ``/metrics`` on their own port, from a daemon 

433 thread. Fly scrapes it on 0.0.0.0 (https://fly.io/docs/monitoring/metrics/); 

434 it is not behind the public ``[http_service]``, so the edge never serves 

435 it. Returns the server (its ``server_address`` has the bound port).""" 

436 

437 class _Handler(BaseHTTPRequestHandler): 

438 def do_GET(self) -> None: 

439 if self.path.split("?", 1)[0] != METRICS_PATH: 

440 self.send_error(404) 

441 return 

442 body = counters.exposition().encode() 

443 self.send_response(200) 

444 self.send_header("Content-Type", "text/plain; version=0.0.4; charset=utf-8") 

445 self.send_header("Content-Length", str(len(body))) 

446 self.end_headers() 

447 self.wfile.write(body) 

448 

449 def log_message(self, *_args) -> None: 

450 # A scrape every 15 s would otherwise be a log line every 15 s. 

451 return 

452 

453 srv = ThreadingHTTPServer( 

454 (host, metrics_port() if port is None else port), _Handler 

455 ) 

456 srv.daemon_threads = True 

457 threading.Thread(target=srv.serve_forever, name="ak-metrics", daemon=True).start() 

458 return srv
