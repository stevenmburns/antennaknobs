1"""Currents below the interface in a far-field readout (issue #1341). 

2 

3Both far-field readouts in this package — `MomwireEngine.far_field` (the 

4grid behind the compare table and the CLI plots) and the web server's 

5`_mag2_at_directions` (the polar-chart cuts) — build the pattern from every 

6solved segment's current moment through the above-ground formula: a PEC 

7image mirrored in the ground plane, then Fresnel reflection on the image 

8wave. That formula is the far field of a current ABOVE the interface. The 

9far field above ground of a current INSIDE the medium is the transmitted 

10field, refracted at the boundary and attenuated through the soil, and its 

11stationary-phase form is momwire#570 — not written. Neither readout ever 

12looked at which side of the plane a segment was on, so the app served for a 

13wholly buried dipole a −21 dBi pattern that means nothing, and imaged the 

14buried radials of the ground-mounted vertical as if they stood in air. 

15 

16This module is the one place the rule lives: 

17 

18* every current below the plane → the pattern is REFUSED by name (the 

19 impedance, currents and charges stay served); 

20* a mixture → the pattern is evaluated twice, with and without the 

21 in-medium segments, and the share of the radiated power (over the lit 

22 hemisphere, sin θ weighted) that the imaged in-medium currents account 

23 for is the measured dependence of this pattern on currents the readout 

24 cannot place honestly. Under `IN_MEDIUM_POWER_SHARE_BAR` the full 

25 readout is served (the status quo — excluding the segments is not the 

26 transmitted answer either, and on the ground-mounted vertical it moves 

27 the peak 0.37 dB AWAY from NEC-5's) with a note carrying the share and 

28 the change at the peak direction; over the bar it is refused, because a 

29 pattern that depends materially on in-medium currents needs momwire#570. 

30 

31The measure is a POWER SHARE and not a worst-direction dB change, and 

32not the share of current moment below the plane, for two reasons found 

33by fire. A dB change over the lit hemisphere compares against nulls: a 

34monopole over one buried radial has an exact zenith null on its 

35above-ground currents alone and a filled one with the radial imaged, and 

36that read as "254 dB" and refused a pattern the physics barely notices 

37(the main-only power-balance suite, 2026-09-10). A moment fraction sees 

3838 % of Σ|I·dl| below ground on the shipped buried-radial vertical, whose 

39symmetric screen cancels in the far field and accounts for 8 % of the 

40radiated power as imaged. The power share is bounded, integrates over 

41the whole lit hemisphere, and is 100 % exactly when nothing is above 

42the plane. 

43"""

47from dataclasses import dataclass

49import numpy as np

51# The share of the lit hemisphere's radiated power that the imaged in-medium 

52# currents account for, past which the pattern is refused. Half: past that 

53# the in-medium currents are not a detail of the pattern, they are the 

54# pattern. Measured on the catalog at 13/0.005 (2026-09-10): 

55# buried_radial_vertical 8.3 % (4 radials; 8.8 % at one, 8.1 % at two), 

56# elevated_buried_counterpoise 1.6 %, buried_dipole 100 % (refused: nothing 

57# above the plane). 

58IN_MEDIUM_POWER_SHARE_BAR = 0.5 

59 

60# Directions more than this far below the pattern's own peak are ignored by 

61# the sensitivity measure: a null moves by tens of dB when anything moves, 

62# and a null is not what a pattern is read for. 

63IN_MEDIUM_LIT_WINDOW_DB = 20.0 

64 

65TRANSMITTED_FAR_FIELD_ISSUE = "momwire#570" 

66 

67 

68class InMediumPatternRefusal(ValueError): 

69 """The far field of this solve cannot be read out honestly (issue #1341). 

70 

71 Raised by `MomwireEngine.far_field`; the web solve path stores the same 

72 sentence under ``pattern_refusal`` instead of raising, so impedance and 

73 currents still ship. 

74 """ 

75 

76 

77@dataclass(frozen=True) 

78class InMediumAssessment: 

79 """What the readout found below the plane. ``fraction`` is the share of 

80 Σ|I·dl| below the interface; ``power_share`` the share of the lit 

81 hemisphere's radiated power the imaged in-medium currents account for; 

82 ``delta_db`` the change at the full pattern's peak direction between 

83 the full readout and the above-ground currents alone (negative when 

84 the imaged currents add there); exactly one of ``refusal`` / ``note`` 

85 is set when ``fraction`` is non-zero, neither when it is zero.""" 

86 

87 fraction: float 

88 power_share: float 

89 delta_db: float 

90 refusal: str | None 

91 note: str | None 

92 

93 @property 

94 def served(self) -> bool: 

95 return self.refusal is None

98def below_surface_mask( 

99 mid: np.ndarray, ground_z: float | None, *, tol=None 

100) -> np.ndarray: 

101 """Segments whose midpoint lies below the ground plane at ``ground_z``. 

102 

103 ``tol`` defaults to 1e-6 of the structure's largest extent (at least 

104 1 m): a segment whose midpoint is a float-noise below the plane is ON 

105 the plane, and a deliberate burial is orders of magnitude deeper. 

106 """ 

107 mid = np.asarray(mid, dtype=float) 

108 if ground_z is None or mid.size == 0: 

109 return np.zeros(mid.shape[0], dtype=bool) 

110 if tol is None: 

111 extent = float(np.max(np.ptp(mid, axis=0))) if mid.shape[0] > 1 else 0.0 

112 tol = 1e-6 * max(extent, 1.0) 

113 return mid[:, 2] < float(ground_z) - tol 

114

116def moment_fraction(dr: np.ndarray, i_mid: np.ndarray, mask: np.ndarray) -> float: 

117 """Share of Σ|I|·|dl| carried by the masked segments.""" 

118 w = np.abs(np.asarray(i_mid)) * np.linalg.norm(np.asarray(dr, dtype=float), axis=1) 

119 total = float(np.sum(w)) 

120 return float(np.sum(w[mask]) / total) if total > 0.0 else 0.0 

121 

122 

123def power_share( 

124 mag2_full, mag2_above, weights=None, *, window_db=IN_MEDIUM_LIT_WINDOW_DB 

125): 

126 """The share of the full readout's radiated power, over the directions 

127 within ``window_db`` of its peak, that the imaged in-medium currents 

128 account for: Σ w·|full − above| / Σ w·full. ``weights`` is the solid- 

129 angle weight per direction (sin θ per row on a θ×φ grid), broadcast 

130 against the arrays; None means a uniform direction set.""" 

131 full = np.asarray(mag2_full, dtype=float) 

132 above = np.asarray(mag2_above, dtype=float) 

133 if weights is None: 

134 w = np.ones_like(full) 

135 else: 

136 w = np.broadcast_to(np.asarray(weights, dtype=float), full.shape) 

137 peak = float(np.max(full)) 

138 if peak <= 0.0: 

139 return 0.0 

140 lit = full >= peak * 10.0 ** (-window_db / 10.0) 

141 denominator = float(np.sum(w[lit] * full[lit])) 

142 if denominator <= 0.0: 

143 return 0.0 

144 return float(np.sum(w[lit] * np.abs(full[lit] - above[lit])) / denominator) 

145 

146 

147def peak_delta_db(mag2_full, mag2_above) -> float: 

148 """The change, in dB, at the full readout's peak direction when the 

149 in-medium currents are dropped: 10·log10(above/full) there. Bounded 

150 below at −99 for an exact null.""" 

151 full = np.asarray(mag2_full, dtype=float) 

152 above = np.asarray(mag2_above, dtype=float) 

153 idx = np.unravel_index(int(np.argmax(full)), full.shape) 

154 if full[idx] <= 0.0: 

155 return 0.0 

156 if above[idx] <= 0.0: 

157 return -99.0 

158 return float(10.0 * np.log10(above[idx] / full[idx]))

161def wholly_buried_sentence() -> str: 

162 return ( 

163 "every current in this solve lies below the interface, and the far " 

164 "field above ground of a source inside the medium is the transmitted " 

165 "field — refracted at the interface and attenuated through the soil — " 

166 f"which this readout does not compute ({TRANSMITTED_FAR_FIELD_ISSUE}). " 

167 "Impedance, currents and charges are served; the pattern is not." 

168 )

171def over_bar_sentence(fraction: float, share: float, delta_db: float) -> str: 

172 return ( 

173 f"{fraction:.0%} of the current moment lies below the interface, and " 

174 "imaged as if above ground rather than through the interface it " 

175 f"accounts for {share:.0%} of the radiated power ({delta_db:+.1f} dB at " 

176 "the peak). A pattern that depends on in-medium currents needs the " 

177 f"transmitted far field ({TRANSMITTED_FAR_FIELD_ISSUE}), which this " 

178 "readout does not compute; the pattern is not served." 

179 )

181 

182def served_note(fraction: float, share: float, delta_db: float) -> str: 

183 return ( 

184 f"{fraction:.0%} of the current moment lies below the interface and is " 

185 f"imaged as if above ground; that accounts for {share:.0%} of the " 

186 f"radiated power ({delta_db:+.1f} dB at the peak). The transmitted far " 

187 f"field ({TRANSMITTED_FAR_FIELD_ISSUE}) is not computed." 

188 )

191def assess( 

192 mid, dr, i_mid, ground_z, evaluate, *, weights=None, bar=IN_MEDIUM_POWER_SHARE_BAR 

193): 

194 """Assess one solve's far-field readout against the currents below the 

195 plane. ``evaluate(mid, dr, i_mid) -> |M_perp|²`` over a fixed direction 

196 set is the readout under test, called at most twice; ``weights`` is that 

197 direction set's solid-angle weight (sin θ per row on a θ×φ grid).""" 

198 mid = np.asarray(mid, dtype=float) 

199 mask = below_surface_mask(mid, ground_z) 

200 if not np.any(mask): 

201 return InMediumAssessment(0.0, 0.0, 0.0, None, None) 

202 fraction = moment_fraction(dr, i_mid, mask) 

203 if not np.any(~mask): 

204 return InMediumAssessment(fraction, 1.0, -99.0, wholly_buried_sentence(), None) 

205 above = ~mask 

206 full = evaluate(mid, dr, i_mid) 

207 part = evaluate(mid[above], np.asarray(dr)[above], np.asarray(i_mid)[above]) 

208 share = power_share(full, part, weights) 

209 delta = peak_delta_db(full, part) 

210 if share > bar: 

211 return InMediumAssessment( 

212 fraction, share, delta, over_bar_sentence(fraction, share, delta), None 

213 ) 

214 return InMediumAssessment( 

215 fraction, share, delta, None, served_note(fraction, share, delta) 

216 )
