1"""Discovery and path-loading of user-authored antenna designs. 

2 

3Engine/UI-agnostic: pure filesystem + ``importlib`` with no dependency on the 

4``web`` package, so both the CLI (``antennaknobs.cli``) and the web adapter 

5resolve ``user.<name>`` designs from the same folders. The web layer 

6(``web/user_designs.py``) adds REGISTRY registration, scaffolding, and 

7error-panel formatting on top of this. 

8 

9User designs live *outside* the installed package (so a ``pip`` upgrade never 

10clobbers them) and are loaded by file path. The ``user`` namespace can never 

11collide with or shadow a built-in family design. 

12""" 

13

29# Antenna DATA files that count as designs in a user folder (issue #1419): a 

30# NEC card deck, a SimNEC circuit or an MMANA-GAL model (AK#1897) dropped 

31# beside the ``.py`` designs shows up as ``user.<stem>`` with no stub, through 

32# the same pure-data loader the CLI's ``@file`` spec uses. Data never 

33# executes, so these bypass the trust gate that Python designs go through. 

34DECK_SUFFIXES = (".nec", ".ssn", ".maa", ".ez")

37def default_user_dir() -> Path: 

38 """The primary user-design folder: ``$ANTENNAKNOBS_USER_DIR`` if set, 

39 else ``~/.antennaknobs/designs``. Read fresh each call so tests can 

40 redirect it via the environment.""" 

41 env = os.environ.get("ANTENNAKNOBS_USER_DIR") 

42 if env: 

43 return Path(env).expanduser() 

44 return Path.home() / ".antennaknobs" / "designs"

47def user_design_dirs() -> list[Path]: 

48 """Folders scanned for user designs, in priority order: the default (or 

49 the env override) plus ``./antenna_designs`` if it exists in the cwd.""" 

50 dirs = [default_user_dir()] 

51 seen = {d.resolve() for d in dirs if d.exists()} 

52 local = Path.cwd() / "antenna_designs" 

53 if local.is_dir() and local.resolve() not in seen: 

54 dirs.append(local) 

55 return dirs

58class NoBuilderError(AttributeError): 

59 """The file imported cleanly but defines no ``Builder``. 

60 

61 Its own class, not a bare ``AttributeError``, because the two callers want 

62 opposite things and only the *cause* tells them apart. A CLI user who typed 

63 a design name deserves the message; the web's registry sweep is looking at 

64 every ``.py`` in the folder, and a user's scratch program sitting next to 

65 their designs is not a broken design — it is not a design at all. An 

66 ``AttributeError`` raised from INSIDE a user's module (a typo in their 

67 ``Builder``) must keep reaching the error panel, which is exactly what a 

68 shared exception type could not distinguish. 

69 

70 Subclasses ``AttributeError`` so existing handlers and the CLI's message 

71 are unchanged. 

72 """

83def _is_design_file(path: Path, *, top: bool = True) -> bool: 

84 """Skip private files and, at the folder's top level (``top``), the 

85 copied authoring templates.""" 

86 stem = path.stem 

87 return not (stem.startswith("_") or (top and stem in _TEMPLATE_STEMS))

90def is_deck_file(path: Path) -> bool: 

91 """True for a data-file design (``.nec`` / ``.ssn`` / ``.maa`` / ``.ez``) as 

92 opposed to a Python one — the two load differently and only the Python 

93 one needs trusting.""" 

94 return path.suffix.lower() in DECK_SUFFIXES

97def load_builder(path: Path, *, trust: bool | None = None): 

98 """Import a single user file by path and return its ``Builder`` class. 

99 Re-executes the file on every call, so edits are picked up live. 

100 

101 A terrain file (see `load_user_file`) is not a design: it raises 

102 `NoBuilderError`, saying what the file is. 

103 """ 

104 kind, obj = load_user_file(path, trust=trust) 

105 if kind == "terrain": 

106 raise NoBuilderError( 

107 f"{path.name} defines a terrain (`{TERRAIN_ATTR}`), not an antenna " 

108 "design; pick it as the ground's terrain in the web app" 

109 ) 

110 return obj

113def _terrain_preset(path: Path, preset): 

114 """The file's `TERRAIN`, named after the file, or the reason it cannot 

115 be used, in words for the error panel.""" 

116 from .terrain import BUILTIN_PRESET_NAMES, TerrainPreset 

117 

118 stem = _user_name(path) 

119 if not isinstance(preset, TerrainPreset): 

120 raise TypeError( 

121 f"`{TERRAIN_ATTR}` must be a TerrainPreset (from antennaknobs.terrain), " 

122 f"and it is a {type(preset).__name__}" 

123 ) 

124 for what in ("name", "label"): 

125 given = getattr(preset, what) 

126 if given not in ("", stem): 

127 raise ValueError( 

128 f"`{TERRAIN_ATTR}` sets {what}={given!r}, but a terrain file is " 

129 f"named by its file name ({stem!r}); leave {what} out, or rename " 

130 "the file" 

131 ) 

132 if stem in BUILTIN_PRESET_NAMES: 

133 raise ValueError( 

134 f"{stem!r} is a built-in terrain and a file cannot replace it; " 

135 f"rename {path.name} (for example to {stem}_mine.py)" 

136 ) 

137 return replace(preset, name=stem, label=stem)

140def load_user_file(path: Path, *, trust: bool | None = None): 

141 """Import a single user file by path and say what it is: 

142 ``("design", Builder)`` or ``("terrain", TerrainPreset)``. 

143 

144 A folder holds two kinds of Python file (issue #1292). A design defines 

145 ``Builder``. A terrain defines ``TERRAIN = TerrainPreset(...)`` and no 

146 ``Builder``; there is one per file, and the file's name is the preset's 

147 name and its label in the ground panel. A file with both is refused, and 

148 a file with neither raises `NoBuilderError`. 

149 

150 Re-executes the file on every call, so edits are picked up live. 

151 

152 A user design runs with full privileges, so it executes only if it is 

153 *trusted* (see ``design_trust``): the file's contents are recorded in the 

154 trust store, or the blanket ``ANTENNAKNOBS_TRUST_USER_DESIGNS`` env flag is 

155 set. An untrusted file raises ``DesignNotTrustedError`` — carrying the 

156 advisory screen report so a UI/CLI can show what the design does before the 

157 user decides — and is NOT executed. Pass ``trust=True`` to bypass the gate 

158 for a file you already vouch for (e.g. programmatic/test use). 

159 

160 A ``.nec`` / ``.ssn`` file is data, not code: it goes through 

161 ``file_designs.builder_from_file`` (the CLI's ``@file`` loader), which 

162 parses and never executes, so the trust gate does not apply and ``trust`` 

163 is ignored. 

164 """ 

165 if is_deck_file(path): 

166 from .file_designs import builder_from_file 

167 

168 return "design", builder_from_file(str(path)) 

169 if trust is None: 

170 trust = design_trust.is_trusted(path) 

171 if not trust: 

172 # Advisory only: the screen report never blocks on its own, it just 

173 # tells the user what they're being asked to trust. 

174 report = design_screen.screen_file(path) 

175 raise design_trust.DesignNotTrustedError(path, report) 

176 # Named after the design, subfolders included, so two ``x.py`` files in 

177 # different subfolders never share a `sys.modules` entry (AK#2064). 

178 modname = f"{_MODULE_PREFIX}.{_user_name(path).replace('.', '__')}" 

179 spec = importlib.util.spec_from_file_location(modname, path) 

180 if spec is None or spec.loader is None: 

181 raise ImportError(f"could not create import spec for {path.name}") 

182 mod = importlib.util.module_from_spec(spec) 

183 sys.modules[modname] = mod 

184 try: 

185 spec.loader.exec_module(mod) 

186 except BaseException: 

187 sys.modules.pop(modname, None) 

188 raise 

189 cls = getattr(mod, "Builder", None) 

190 preset = getattr(mod, TERRAIN_ATTR, None) 

191 if cls is not None and preset is not None: 

192 raise ValueError( 

193 f"this file defines both `Builder` and `{TERRAIN_ATTR}`; a file is " 

194 "one antenna design or one terrain, so move one of them to a file " 

195 "of its own" 

196 ) 

197 if preset is not None: 

198 return "terrain", _terrain_preset(path, preset) 

199 if cls is None: 

200 raise NoBuilderError( 

201 "no `Builder` class found — define `class Builder(AntennaBuilder): ...`" 

202 ) 

203 return "design", cls

206def _folder_files(d: Path, roots: frozenset[Path] = frozenset()) -> list[Path]: 

207 """The candidate files under user folder ``d``, at any depth (AK#2064): 

208 one folder at a time, its ``.py`` designs before its data-file designs, 

209 the top level first. A subfolder whose name starts with ``_`` or ``.`` 

210 is private and skipped with everything under it, and so is one whose 

211 name carries a ``.``, since the dot separates a design name's parts. A 

212 subfolder in ``roots`` is a scanned folder of its own (``./antenna_designs`` 

213 can sit inside the designs folder), and is left to its own scan.""" 

214 py = sorted(d.glob("*.py")) 

215 decks = sorted(p for p in d.iterdir() if p.is_file() and is_deck_file(p)) 

216 out = [p for p in py + decks if _is_design_file(p, top=False)] 

217 for sub in sorted(p for p in d.iterdir() if p.is_dir()): 

218 if sub.name.startswith(("_", ".")) or "." in sub.name: 

219 continue 

220 if sub.resolve() not in roots: 

221 out += _folder_files(sub, roots) 

222 return out

225def design_name(path: Path, root: Path) -> str: 

226 """The name of user design ``path`` under folder ``root``, without the 

227 ``user.`` prefix: its file name, after its subfolders, dot-separated. 

228 ``mydesigns/my_dipole.py`` is ``mydesigns.my_dipole`` (AK#2064); a file 

229 at the top level keeps the name it always had.""" 

230 rel = path.relative_to(root) 

231 return ".".join((*rel.parent.parts, rel.stem))

234def _scan() -> Iterator[tuple[str, Path, Path | None]]: 

235 """``(name, path, winner)`` for every candidate file in every folder: 

236 ``winner`` is None for the file that carries its name, and that file's 

237 path for each later one of the same name.""" 

238 seen: dict[str, Path] = {} 

239 dirs = [d for d in user_design_dirs() if d.is_dir()] 

240 roots = frozenset(d.resolve() for d in dirs) 

241 for d in dirs: 

242 for path in _folder_files(d, roots): 

243 if path.parent == d and not _is_design_file(path): 

244 continue 

245 name = design_name(path, d) 

246 winner = seen.setdefault(name, path) 

247 yield name, path, (None if winner is path else winner)

250def iter_design_files() -> Iterator[tuple[str, Path]]: 

251 """Yield ``(stem, path)`` for each user design file across all folders, 

252 first-folder-wins on a duplicate stem (matching the web's priority order). 

253 

254 Within one folder the ``.py`` designs come first and the data-file designs 

255 (``.nec`` / ``.ssn``, issue #1419) after them, so a stub beside its deck — 

256 the documented ``my_yagi.py`` + ``my_yagi.nec`` recipe — is the design and 

257 the bare deck of the same stem is not listed twice.""" 

258 for stem, path, winner in _scan(): 

259 if winner is None: 

260 yield stem, path

263def iter_shadowed_files() -> Iterator[tuple[str, Path, Path]]: 

264 """Yield ``(stem, path, winner)`` for each file `iter_design_files` leaves 

265 out because ``winner`` already carries its stem.""" 

266 for stem, path, winner in _scan(): 

267 if winner is not None: 

268 yield stem, path, winner

271def _user_name(path: Path) -> str: 

272 """`design_name` under whichever user folder holds ``path``; its file 

273 name when none does (a file loaded by path from elsewhere).""" 

274 resolved = path.resolve() 

275 # The innermost folder that holds it, as `_scan` names it: a scanned 

276 # folder nested in another is scanned on its own (`_folder_files`). 

277 holders = [ 

278 d.resolve() 

279 for d in user_design_dirs() 

280 if d.is_dir() and resolved.is_relative_to(d.resolve()) 

281 ] 

282 if not holders: 

283 return path.stem 

284 return design_name(resolved, max(holders, key=lambda d: len(d.parts)))

287def find_design_file(stem: str) -> Path | None: 

288 """The file backing user design ``stem``, or None if there is no such file.""" 

289 for s, path in iter_design_files(): 

290 if s == stem: 

291 return path 

292 return None

295def resolve_user_design(stem: str): 

296 """Return the ``Builder`` class for user design ``stem``, or None if no 

297 such file exists. Load errors (syntax error, missing ``Builder``) propagate 

298 so someone debugging their own file sees the real cause.""" 

299 path = find_design_file(stem) 

300 if path is None: 

301 return None 

302 return load_builder(path)
