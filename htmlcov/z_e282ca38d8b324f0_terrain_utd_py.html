1"""Ray geometry and wedge diffraction for the faceted-terrain far field 

2(issue #1373, the follow-on to #534). 

3 

4The specular composer in ``engines/momwire.py`` is geometric optics: for 

5every direction it adds the antenna's direct radiation and one reflection 

6off a facet. Two things are missing there, and both are why the uphill band 

7below the slope angle is hatched on the slope page and why the downhill 

8curves on M0AGP's hillside pages have notches: 

9 

10* nothing is ever **shadowed** — uphill below the slope angle the composer 

11 still adds a direct ray the crest blocks, and reports grazing cancellation 

12 where the sky is actually behind the hill; 

13* nothing is **diffracted** — where the specular point steps from one facet 

14 onto the next the field steps with it, and at a shadow boundary the field 

15 would switch off like a light. 

16 

17This module supplies the pieces that fix both, for one azimuth cut at a 

18time (the composer treats each observation azimuth's terrain profile as a 

19two-dimensional problem, and so does this): 

20 

21* the cut's break points and per-facet planes; 

22* visibility tests for the direct ray, the incident and reflected legs of a 

23 specular reflection, and a diffracted ray; 

24* the exact specular reflection off each facet as a *tilted mirror* — the 

25 antenna's reference point imaged across the facet's own plane, the 

26 reflected ray required to leave from a point inside the facet's span. The 

27 #534 composer located the specular point with the horizontal-mirror 

28 formula at the facet's mid-height and always picked *some* facet, which 

29 is exact for horizontal facets and increasingly wrong for a mast standing 

30 on a slope; 

31* the uniform-theory-of-diffraction (UTD) wedge coefficient at every break 

32 between facets, in Luebbers' form for lossy faces: the Kouyoumjian–Pathak 

33 four-term coefficient with the two faces' Fresnel reflection coefficients 

34 on the reflection terms and Fresnel transition functions on all four, so 

35 the total field is continuous across each shadow and reflection boundary 

36 (at a shadow boundary the diffracted field is exactly half the field the 

37 boundary switches off). 

38 

39Conventions match ``terrain.py``: in a cut, ``x`` is metres outward from the 

40mast axis along the azimuth, ``z`` metres above the crest plane; the profile 

41starts at ``(0, 0)``. Elevation ``psi`` is measured from the horizontal; 

42the composer's zenith angle is ``theta = pi/2 - psi``. The source is the 

43antenna's reference point ``S = (0, h)`` with ``h`` the current-weighted 

44mean segment height above the crest plane — the same ``h_ref`` the specular 

45composer uses. Time convention ``exp(+j omega t)`` with far fields going 

46as ``exp(-jkr)/r``, which is the composer's ``exp(+jk r_hat . r')`` phase. 

47 

48What this is not: no back-diffraction across the mast (an edge in the 

49opposite azimuth's cut scattering into this one), no double diffraction 

50(an edge's diffracted ray blocked by a further edge is simply dropped), no 

51surface wave, and the incident field at an edge is the antenna's far-field 

52radiation vector at that direction with ``1/s'`` decay — exact for edges 

53many antenna sizes away, an approximation for a break under the mast. 

54"""

81class Plane:

83 pointing outward, unit normal ``n`` pointing into the air (``n_z > 0``), 

84 ``length`` (``inf`` for the last facet) and its medium."""

88 n: np.ndarray 

89 length: float

91 sigma: float

94@dataclass(frozen=True) 

95class Wedge: 

96 """An edge between two facets: its position ``p``, the angle ``a0`` of 

97 the 0-face (the inner facet, seen from the edge going back toward the 

98 mast), the exterior wedge parameter ``n`` (exterior angle ``n*pi``, 

99 measured from the 0-face through the air to the n-face), and the two 

100 faces' media.""" 

101 

102 p: np.ndarray 

103 a0: float 

104 n: float 

105 eps0: float 

106 sig0: float 

107 epsn: float 

108 sign: float 

109 sense: int = 1 # +1 front side of the cut, -1 the mirrored back side

112def _side_vertices(sector: Sector) -> tuple[list[float], list[float]]: 

113 xs, zs = [0.0], [0.0] 

114 for f in sector.facets: 

115 if f.x1 is None: 

116 if abs(f.z1 - zs[-1]) > 0: 

117 # a vertical step into the infinite facet 

118 xs.append(xs[-1] + _STEP_RUN) 

119 zs.append(float(f.z1)) 

120 break 

121 xs.append(float(f.x1)) 

122 zs.append(float(f.z1)) 

123 return xs, zs

126def _side_planes(sector: Sector, sense: int) -> list[Plane]: 

127 """One sector's facets as planes in the full cut's coordinates: the 

128 front sector (``sense=+1``) at ``x > 0``, the back sector (``sense=-1``) 

129 mirrored to ``x < 0``. Each plane's ``p0`` is its end nearer the mast 

130 and ``t`` points away from the mast, so ``p0 + t * length`` is the outer 

131 end and consecutive planes meet there.""" 

132 xs, zs = _side_vertices(sector) 

133 n_int = len(xs) - 1 

134 finite = [f for f in sector.facets if f.x1 is not None] 

135 last = sector.facets[-1] 

136 media = [(f.eps_r, f.sigma) for f in finite] 

137 if len(media) < n_int: 

138 media.append((last.eps_r, last.sigma)) # the step into the last facet 

139 out: list[Plane] = [] 

140 for i in range(n_int): 

141 p0 = np.array([sense * xs[i], zs[i]]) 

142 p1 = np.array([sense * xs[i + 1], zs[i + 1]]) 

143 d = p1 - p0 

144 length = float(np.hypot(*d)) 

145 t = d / length 

146 n = np.array([-t[1], t[0]]) 

147 if n[1] < 0: 

148 n = -n 

149 eps, sig = media[i] 

150 # a subdivided ramp is one mirror: merge a facet into the previous 

151 # plane when it continues the same line with the same medium, so a 

152 # 48-facet ramp is one reflection and one pair of edges 

153 if out: 

154 prev = out[-1] 

155 if ( 

156 np.allclose(prev.t, t, atol=1e-12) 

157 and prev.eps_r == eps 

158 and prev.sigma == sig 

159 and abs(float(np.dot(p0 - prev.p0, prev.n))) < 1e-9 

160 ): 

161 out[-1] = Plane(prev.p0, prev.t, prev.n, prev.length + length, eps, sig) 

162 continue 

163 out.append(Plane(p0, t, n, length, eps, sig)) 

164 p0 = np.array([sense * xs[-1], zs[-1]]) 

165 out.append( 

166 Plane( 

167 p0, 

168 np.array([float(sense), 0.0]), 

169 np.array([0.0, 1.0]), 

170 math.inf, 

171 last.eps_r, 

172 last.sigma, 

173 ) 

174 ) 

175 return out

178def _side_wedges(planes_: list[Plane], sense: int) -> list[Wedge]: 

179 out: list[Wedge] = [] 

180 for a, b in itertools.pairwise(planes_): 

181 p = a.p0 + a.t * a.length 

182 u0 = -a.t # from the edge back toward the mast along the inner facet 

183 un = b.t # from the edge away from the mast along the outer facet 

184 a0 = math.atan2(u0[1], u0[0]) 

185 an = math.atan2(un[1], un[0]) 

186 ext = (sense * (a0 - an)) % (2 * math.pi) 

187 n = ext / math.pi 

188 if abs(n - 1.0) < 1e-9: 

189 continue 

190 out.append(Wedge(p, a0, n, a.eps_r, a.sigma, b.eps_r, b.sigma, sense)) 

191 return out

194@dataclass(frozen=True) 

195class Cut: 

196 """The full two-dimensional profile through the mast for one observation 

197 azimuth: the sector in that azimuth at ``x > 0`` and the sector in the 

198 opposite azimuth mirrored to ``x < 0``. Observation directions always 

199 point to ``+x``; the back side matters because a facet behind the mast 

200 is the mirror that throws a mast-on-a-slope's radiation into the 

201 downhill sky, and its crest can diffract over the mast.""" 

202 

203 xb: np.ndarray 

204 zb: np.ndarray 

205 planes: tuple[Plane, ...] 

206 wedges: tuple[Wedge, ...] 

207 

208 

209def build_cut(front: Sector, back: Sector) -> Cut: 

210 fx, fz = _side_vertices(front) 

211 bx, bz = _side_vertices(back) 

212 xb = np.asarray([-x for x in bx[1:]][::-1] + fx) 

213 zb = np.asarray(bz[1:][::-1] + fz) 

214 pf = _side_planes(front, +1) 

215 pb = _side_planes(back, -1) 

216 return Cut( 

217 xb, zb, tuple(pf + pb), tuple(_side_wedges(pf, +1) + _side_wedges(pb, -1)) 

218 ) 

219 

220 

221def break_points(sector: Sector) -> tuple[np.ndarray, np.ndarray]: 

222 """One sector's vertices ``(x_b, z_b)`` from the mast axis outward, 

223 starting at ``(0, 0)`` (the front half of a cut).""" 

224 xs, zs = _side_vertices(sector) 

225 return np.asarray(xs), np.asarray(zs) 

226 

227 

228def planes(sector: Sector) -> list[Plane]: 

229 """One sector's facets as front-side planes.""" 

230 return _side_planes(sector, +1) 

231 

232 

233def wedges(sector: Sector) -> list[Wedge]: 

234 """One sector's edges, front side; collinear breaks (a subdivided ramp) 

235 diffract nothing (``n == 1`` makes the coefficient vanish) and are 

236 skipped.""" 

237 return _side_wedges(_side_planes(sector, +1), +1)

245def ray_clear(px, pz, wx, wz, xb, zb, *, x_from=None): 

246 """Is the ray from ``(px, pz)`` in direction ``(wx, wz)`` (arrays, any 

247 common shape) clear of the profile vertices ``(xb, zb)``? A vertex at 

248 ``x > px`` (or ``x > x_from`` when given) blocks the ray if it lies 

249 above the ray's height there. Vertices behind the start are ignored. 

250 Rays going inward (``wx <= 0``) are treated as clear — this module only 

251 reads outward directions.""" 

252 px, pz, wx, wz = np.broadcast_arrays( 

253 np.asarray(px, float), 

254 np.asarray(pz, float), 

255 np.asarray(wx, float), 

256 np.asarray(wz, float), 

257 ) 

258 start = ( 

259 px if x_from is None else np.broadcast_to(np.asarray(x_from, float), px.shape) 

260 ) 

261 clear = np.ones(px.shape, dtype=bool) 

262 slope = np.where(wx > 0, wz / np.where(wx > 0, wx, 1.0), np.inf) 

263 for x, z in zip(xb, zb, strict=True): 

264 ahead = (x > start + 1e-9) & (wx > 0) 

265 zr = pz + (x - px) * slope 

266 clear &= ~(ahead & (z > zr + 1e-9)) 

267 return clear

270def segment_clear(px, pz, qx, qz, xb, zb): 

271 """Is the straight path from ``(px, pz)`` to ``(qx, qz)`` clear of the 

272 profile? Vertices strictly between the two x's block if above the 

273 chord.""" 

274 px, pz, qx, qz = np.broadcast_arrays( 

275 np.asarray(px, float), 

276 np.asarray(pz, float), 

277 np.asarray(qx, float), 

278 np.asarray(qz, float), 

279 ) 

280 clear = np.ones(px.shape, dtype=bool) 

281 lo, hi = np.minimum(px, qx), np.maximum(px, qx) 

282 dx = qx - px 

283 safe = np.where(np.abs(dx) > 1e-12, dx, 1.0) 

284 for x, z in zip(xb, zb, strict=True): 

285 between = (x > lo + 1e-9) & (x < hi - 1e-9) 

286 zc = pz + (x - px) * (qz - pz) / safe 

287 clear &= ~(between & (z > zc + 1e-9)) 

288 return clear

296@dataclass(frozen=True) 

297class Reflection: 

298 """A specular ray path off one facet (``planes`` of length 1) or off two 

299 in sequence (length 2, the doubly reflected ray a concave corner such 

300 as a slope's toe sends skyward), for a set of source points ``src`` and 

301 elevations ``psi``. ``image`` (P, 2) is each source imaged across the 

302 planes in order; ``mask`` (P, T) marks where the path exists (every 

303 reflection point inside its facet, every leg clear of the profile); 

304 ``cos_tis`` are the cosines of the local incidence angle from each 

305 facet's normal, (P, T), in path order.""" 

306 

307 planes: tuple[Plane, ...] 

308 image: np.ndarray 

309 mask: np.ndarray 

310 cos_tis: tuple[np.ndarray, ...] 

311 

312 @property 

313 def plane(self) -> Plane: 

314 return self.planes[0] 

315 

316 @property 

317 def cos_ti(self) -> np.ndarray: 

318 return self.cos_tis[0]

321def _sources(src, h=None): 

322 """Source points as (P, 2); a bare height ``h`` means one point on the 

323 mast axis, ``(0, h)``.""" 

324 if src is None: 

325 src = np.array([[0.0, float(h)]]) 

326 src = np.asarray(src, float) 

327 return src.reshape(-1, 2)

339def _line_key(pl: Plane): 

340 return (tuple(np.round(pl.n, 9)), round(float(np.dot(pl.p0, pl.n)), 6))

343def reflections( 

344 cut, psi: np.ndarray, h: float | None = None, *, src=None 

345) -> list[Reflection]: 

346 """Every specular path from the source points (``src`` (P, 2), or the 

347 single point ``(0, h)``) toward elevations ``psi`` (radians above the 

348 horizontal): single reflections off each facet, then double reflections 

349 off each ordered pair of distinct facets. A path exists when the source 

350 (or its running image) is on the air side of each plane, each reflected 

351 direction leaves its plane on the air side, each reflection point lies 

352 within its facet's span, and no leg is blocked by the profile. 

353 

354 Double reflections are what a concave corner (a slope meeting a plain) 

355 does with the rays that hit the slope: without them the toe wedge's 

356 diffraction coefficient would be patching a boundary the GO sum does not 

357 have, and the total would step there instead of at nothing.""" 

358 cut = _as_cut(cut) 

359 S = _sources(src, h) # (P, 2) 

360 psi = np.asarray(psi, float) 

361 w = np.stack([np.cos(psi), np.sin(psi)], axis=-1) # (T, 2) 

362 xb, zb = cut.xb, cut.zb 

363 P, T = S.shape[0], psi.shape[0] 

364 out: list[Reflection] = [] 

365 # Two facets on one line (the front and back halves of a flat plane, 

366 # meeting at the mast foot) would both claim the specular point that 

367 # sits exactly on their shared vertex; a path is claimed once, keyed on 

368 # the line(s) it reflects off. 

369 claimed: dict = {} 

370 

371 def _claim(key, inside): 

372 prev = claimed.get(key) 

373 if prev is not None: 

374 inside = inside & ~prev 

375 claimed[key] = prev | inside 

376 else: 

377 claimed[key] = inside.copy() 

378 return inside 

379 

380 wn_all = {id(pl): (w @ pl.n)[None, :] for pl in cut.planes} # (1, T) 

381 for pl in cut.planes: 

382 d0 = (S - pl.p0[None, :]) @ pl.n # (P,) 

383 above = d0 > 0.0 

384 if not np.any(above): 

385 continue 

386 image = S - 2.0 * d0[:, None] * pl.n[None, :] # (P, 2) 

387 wn = wn_all[id(pl)] 

388 ok = above[:, None] & (wn > _EPS_ANGLE) 

389 lam = np.where(ok, d0[:, None] / np.where(ok, wn, 1.0), 0.0) # (P, T) 

390 Q = image[:, None, :] + lam[..., None] * w[None, :, :] # (P, T, 2) 

391 along = (Q - pl.p0) @ pl.t 

392 inside = ok & (along >= -1e-9) & (along <= pl.length + 1e-9) 

393 inside &= segment_clear( 

394 S[:, 0, None], S[:, 1, None], Q[..., 0], Q[..., 1], xb, zb 

395 ) 

396 inside &= ray_clear(Q[..., 0], Q[..., 1], w[None, :, 0], w[None, :, 1], xb, zb) 

397 inside = _claim(_line_key(pl), inside) 

398 if np.any(inside): 

399 out.append( 

400 Reflection( 

401 (pl,), 

402 image, 

403 inside, 

404 (np.clip(np.broadcast_to(wn, (P, T)), 0.0, 1.0),), 

405 ) 

406 ) 

407 for A in cut.planes: 

408 dA = (S - A.p0[None, :]) @ A.n 

409 if not np.any(dA > 0.0): 

410 continue 

411 SA = S - 2.0 * dA[:, None] * A.n[None, :] 

412 for B in cut.planes: 

413 if B is A or _line_key(B) == _line_key(A): 

414 continue 

415 dAB = (SA - B.p0[None, :]) @ B.n 

416 good = (dA > 0.0) & (dAB > 0.0) 

417 if not np.any(good): 

418 continue 

419 SAB = SA - 2.0 * dAB[:, None] * B.n[None, :] 

420 wn = wn_all[id(B)] 

421 ok = good[:, None] & (wn > _EPS_ANGLE) 

422 lam2 = np.where(ok, dAB[:, None] / np.where(ok, wn, 1.0), 0.0) 

423 Q2 = SAB[:, None, :] + lam2[..., None] * w[None, :, :] 

424 along2 = (Q2 - B.p0) @ B.t 

425 ok &= (along2 >= -1e-9) & (along2 <= B.length + 1e-9) 

426 v1 = Q2 - SA[:, None, :] 

427 l1 = np.hypot(v1[..., 0], v1[..., 1]) 

428 w1 = v1 / np.where(l1 > 0, l1, 1.0)[..., None] 

429 wn1 = w1 @ A.n 

430 ok &= wn1 > _EPS_ANGLE 

431 lam1 = np.where(ok, dA[:, None] / np.where(ok, wn1, 1.0), 0.0) 

432 ok &= lam1 < l1 

433 Q1 = SA[:, None, :] + lam1[..., None] * w1 

434 along1 = (Q1 - A.p0) @ A.t 

435 ok &= (along1 >= -1e-9) & (along1 <= A.length + 1e-9) 

436 ok &= segment_clear( 

437 S[:, 0, None], S[:, 1, None], Q1[..., 0], Q1[..., 1], xb, zb 

438 ) 

439 ok &= segment_clear(Q1[..., 0], Q1[..., 1], Q2[..., 0], Q2[..., 1], xb, zb) 

440 ok &= ray_clear( 

441 Q2[..., 0], Q2[..., 1], w[None, :, 0], w[None, :, 1], xb, zb 

442 ) 

443 ok = _claim((_line_key(A), _line_key(B)), ok) 

444 if np.any(ok): 

445 out.append( 

446 Reflection( 

447 (A, B), 

448 SAB, 

449 ok, 

450 ( 

451 np.clip(wn1, 0.0, 1.0), 

452 np.clip(np.broadcast_to(wn, (P, T)), 0.0, 1.0), 

453 ), 

454 ) 

455 ) 

456 if src is None: 

457 out = [ 

458 Reflection(r.planes, r.image[0], r.mask[0], tuple(c[0] for c in r.cos_tis)) 

459 for r in out 

460 ] 

461 return out

469def transition(X): 

470 """The UTD Fresnel transition function 

471 ``F(X) = 2j sqrt(X) exp(jX) int_{sqrt(X)}^{inf} exp(-j t^2) dt``, 

472 ``F -> 1`` for large ``X`` and ``F ~ sqrt(pi X) exp(j pi/4)`` for small 

473 ``X`` — the factor that keeps the total field continuous through a 

474 shadow or reflection boundary.""" 

475 X = np.asarray(X, float) 

476 X = np.maximum(X, 1e-300) 

477 a = np.sqrt(X) 

478 z = a * math.sqrt(2.0 / math.pi) 

479 S, C = _fresnel(z) 

480 tail = (math.sqrt(math.pi) / 2.0) * np.exp(-1j * math.pi / 4.0) - math.sqrt( 

481 math.pi / 2.0 

482 ) * (C - 1j * S) 

483 return 2j * a * np.exp(1j * X) * tail

486def _N_pm(beta, n, sign): 

487 """``N^{±}``: the integer nearest a solution of ``2πnN − β = ±π``.""" 

488 return np.round((beta + sign * math.pi) / (2.0 * math.pi * n))

491def _a_pm(beta, n, sign): 

492 """``a^{±}(β) = 2 cos²((2πnN^{±} − β)/2)``.""" 

493 N = _N_pm(beta, n, sign) 

494 return 2.0 * np.cos((2.0 * math.pi * n * N - beta) / 2.0) ** 2

497def _cot_F(arg, X): 

498 """``cot(arg) · F(X)`` with the boundary limit handled: both factors 

499 are singular/zero together at a shadow or reflection boundary.""" 

500 arg = np.asarray(arg, float) 

501 s = np.sin(arg) 

502 small = np.abs(s) < 1e-7 

503 s_safe = np.where(small, 1.0, s) 

504 cot = np.cos(arg) / s_safe 

505 val = cot * transition(X) 

506 # limit at the boundary: n·[sqrt(2πkL) sgn(ε) − 2kLε e^{jπ/4}] e^{jπ/4} 

507 # is what the product tends to; the ε→0 part is the leading term, whose 

508 # sign flips across the boundary — the finite jump the neighbouring GO 

509 # term cancels. Evaluate it from a nudged argument instead of the 

510 # formula so the two sides are symmetric. 

511 if np.any(small): 

512 arg_n = np.where(small, np.where(arg >= 0, 1e-7, -1e-7), arg) 

513 cot_n = np.cos(arg_n) / np.sin(arg_n) 

514 val = np.where(small, cot_n * transition(X), val) 

515 return val

518def fresnel_grazing(theta, eps_c): 

519 """Reflection coefficients ``(R_h, R_v)`` of a lossy half-space for a 

520 ray at grazing angle ``theta`` (radians from the face plane), complex 

521 permittivity ``eps_c``. The same formulas as the composer's, written in 

522 the wedge's grazing angle; ``sin`` and ``cos²`` make them symmetric 

523 about ``π/2`` so a face angle past ``π/2`` reads correctly.""" 

524 st = np.sin(theta) 

525 Q = np.sqrt(eps_c - np.cos(theta) ** 2) 

526 R_h = (st - Q) / (st + Q) 

527 R_v = (eps_c * st - Q) / (eps_c * st + Q) 

528 return R_h, R_v

531def wedge_coefficients(k, L, n, phi, phi_p, eps0_c, epsn_c): 

532 """Heuristic UTD diffraction coefficients ``(D_soft, D_hard)`` for a 

533 wedge of exterior angle ``n·π`` with lossy faces of complex 

534 permittivity ``eps0_c`` (the 0-face) and ``epsn_c`` (the n-face), at 

535 normal incidence (``β0 = π/2``), distance parameter ``L`` (the 

536 source–edge distance for a far-field observer), observation angle 

537 ``phi`` and incidence angle ``phi_p`` measured from the 0-face through 

538 the air. Soft = E parallel to the edge (uses ``R_h``), hard = E in the 

539 plane of incidence (uses ``R_v``). 

540 

541 Luebbers' form puts the 0-face coefficient on the term whose boundary 

542 is the 0-face reflection and the n-face's on the n-face term. The two 

543 incident-field terms also carry boundaries when the wedge is concave 

544 (``n < 1``): the integers ``N^{±}`` are then non-zero exactly where the 

545 boundary belongs to a ray that has reflected off BOTH faces before 

546 leaving, so those terms carry ``(R_0 R_n)^{|N|}`` — the amplitude of 

547 the multiply reflected ray they switch — in the spirit of Holm's 

548 extension. With ``R = ∓1`` every factor is a sign and the perfectly 

549 conducting Kouyoumjian–Pathak coefficients come back.""" 

550 phi = np.asarray(phi, float) 

551 phi_p = np.asarray(phi_p, float) 

552 C = -np.exp(-1j * math.pi / 4.0) / (2.0 * n * math.sqrt(2.0 * math.pi * k)) 

553 dm = phi - phi_p 

554 dp = phi + phi_p 

555 kL = k * L 

556 D1 = _cot_F((math.pi + dm) / (2.0 * n), kL * _a_pm(dm, n, +1)) 

557 D2 = _cot_F((math.pi - dm) / (2.0 * n), kL * _a_pm(dm, n, -1)) 

558 D3 = _cot_F((math.pi + dp) / (2.0 * n), kL * _a_pm(dp, n, +1)) # n-face RB 

559 D4 = _cot_F((math.pi - dp) / (2.0 * n), kL * _a_pm(dp, n, -1)) # 0-face RB 

560 n1 = np.abs(_N_pm(dm, n, +1)) 

561 n2 = np.abs(_N_pm(dm, n, -1)) 

562 n3 = np.maximum(np.abs(_N_pm(dp, n, +1)) - 1.0, 0.0) 

563 n4 = np.abs(_N_pm(dp, n, -1)) 

564 R0_h, R0_v = fresnel_grazing(phi_p, eps0_c) 

565 Rn_h, Rn_v = fresnel_grazing(n * math.pi - phi, epsn_c) 

566 out = [] 

567 for R0, Rn in ((R0_h, Rn_h), (R0_v, Rn_v)): 

568 both = R0 * Rn 

569 out.append( 

570 C 

571 * (both**n1 * D1 + both**n2 * D2 + Rn * both**n3 * D3 + R0 * both**n4 * D4) 

572 ) 

573 return out[0], out[1]

576@dataclass(frozen=True) 

577class Diffraction: 

578 """One edge's diffracted contribution for source points ``src`` (P, 2) 

579 and elevations ``psi`` (T): the edge position ``p``, each source's 

580 distance ``s_p`` (P,) and unit direction ``d`` (P, 2) to the edge, the 

581 soft/hard coefficients (P, T) already carrying ``exp(-jk s')/sqrt(s')``, 

582 and the mask (P, T) of the source–edge legs that are clear and the 

583 diffracted rays that clear the rest of the profile.""" 

584 

585 wedge: Wedge 

586 s_p: np.ndarray 

587 d: np.ndarray 

588 D_soft: np.ndarray 

589 D_hard: np.ndarray 

590 mask: np.ndarray

593def diffractions( 

594 cut, 

595 psi: np.ndarray, 

596 h: float | None = None, 

597 k: float = None, 

598 omega: float = None, 

599 eps0=None, 

600 *, 

601 src=None, 

602): 

603 """Every non-collinear edge's UTD contribution toward elevations ``psi`` 

604 for the source points (``src`` (P, 2), or ``(0, h)``). Sources an edge 

605 cannot see (incident leg blocked) are masked, as are observation 

606 directions whose diffracted ray is blocked further out.""" 

607 from .engines.momwire import EPS0 as _EPS0 # noqa: PLC0415 — avoid a cycle at import 

608 

609 cut = _as_cut(cut) 

610 S = _sources(src, h) 

611 eps0 = _EPS0 if eps0 is None else eps0 

612 psi = np.asarray(psi, float) 

613 xb, zb = cut.xb, cut.zb 

614 w = np.stack([np.cos(psi), np.sin(psi)], axis=-1) 

615 out: list[Diffraction] = [] 

616 for wg in cut.wedges: 

617 v = wg.p[None, :] - S # (P, 2) 

618 s_p = np.hypot(v[:, 0], v[:, 1]) 

619 seen = s_p > 1e-9 

620 d = v / np.where(seen, s_p, 1.0)[:, None] 

621 seen &= segment_clear(S[:, 0], S[:, 1], wg.p[0], wg.p[1], xb, zb) 

622 if not np.any(seen): 

623 continue 

624 # angles from the 0-face through the air; the back side of the cut 

625 # is a mirror image, so its angles run the other way round 

626 phi_p = (wg.sense * (wg.a0 - np.arctan2(-d[:, 1], -d[:, 0]))) % ( 

627 2 * math.pi 

628 ) # (P,) 

629 phi = (wg.sense * (wg.a0 - psi)) % (2 * math.pi) # (T,) 

630 in_air = (phi[None, :] <= wg.n * math.pi + 1e-9) & ( 

631 phi_p[:, None] <= wg.n * math.pi + 1e-9 

632 ) 

633 e0 = wg.eps0 - 1j * wg.sig0 / (omega * eps0) 

634 en = wg.epsn - 1j * wg.sign / (omega * eps0) 

635 Ds, Dh = wedge_coefficients( 

636 k, s_p[:, None], wg.n, phi[None, :], phi_p[:, None], e0, en 

637 ) 

638 scale = (np.exp(-1j * k * s_p) / np.sqrt(np.where(seen, s_p, 1.0)))[:, None] 

639 clear_out = ray_clear(wg.p[0], wg.p[1], w[:, 0], w[:, 1], xb, zb)[None, :] 

640 mask = seen[:, None] & in_air & clear_out 

641 if src is None: 

642 out.append( 

643 Diffraction(wg, s_p[0], d[0], (Ds * scale)[0], (Dh * scale)[0], mask[0]) 

644 ) 

645 else: 

646 out.append(Diffraction(wg, s_p, d, Ds * scale, Dh * scale, mask)) 

647 return out

653 """Mask (P, T) of source points whose direct ray toward each elevation
