1"""ASCII for the free text a NEC deck carries on its comment cards. 

2 

3A NEC card deck is ASCII. The engines that read one are Fortran or C console 

4programs reading fixed-column records, and the decks they are handed are files 

5the workbench writes with an explicit encoding, never the platform default: 

6Windows' cp1252 has no Ω or λ, so a default-encoded write of such a deck 

7raised before the engine ever ran. The only free text a writer puts on a card 

8is a ``CM`` comment, chiefly a design's name, which for a dropped-in file is 

9its file name and can be anything the user typed. So every writer folds that 

10text here and the deck stays ASCII: the readable folds below first, then 

11accents dropped (é to e), then ``?`` for anything left. 

12""" 

13 

14from __future__ import annotations 

15 

16import unicodedata 

17 

18_FOLD = { 

19 "–": "-", # en dash 

20 "—": "-", # em dash 

21 "−": "-", # minus sign 

22 "µ": "u", # micro sign 

23 "μ": "u", # Greek mu 

24 "°": " deg", 

25 "Ω": "ohm", 

26 "Ω": "ohm", # ohm sign 

27 "λ": "lambda", 

28 "½": "1/2", 

29 "¼": "1/4", 

30 "¾": "3/4", 

31 "×": "x", 

32 "‘": "'", 

33 "’": "'", 

34 "“": '"', 

35 "”": '"', 

36 "…": "...", 

37} 

38 

39 

40def card_text(text: str) -> str: 

41 """``text`` as ASCII for a comment card.""" 

42 folded = "".join(_FOLD.get(c, c) for c in text) 

43 plain = "".join( 

44 c for c in unicodedata.normalize("NFKD", folded) if not unicodedata.combining(c) 

45 ) 

46 return plain.encode("ascii", errors="replace").decode("ascii")
