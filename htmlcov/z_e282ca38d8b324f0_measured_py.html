1"""Measured antenna data as a *reference trace* for the sweep plots (issue #595). 

2 

3A NanoVNA (or any VNA) exports a one-port Touchstone ``.s1p``: S11 versus 

4frequency for the antenna as it actually stands in the yard. This module turns 

5such a file into a :class:`MeasuredTrace` — a frequency-indexed complex 

6reflection coefficient that the CLI sweeps draw *alongside* the simulated 

7curve, closing the loop between a design and the bench. 

8 

9The distinction from ``touchstone.py``'s consumers is one of *use*, not format: 

10``network.TouchstoneLoad`` makes an ``.s1p`` a **circuit element** inside 

11``build_network()``; here the same parsed file is **reference data** on the 

12chart. One parser, two subsystems. 

13 

14Two pieces of arithmetic carry the feature: 

15 

16- **Reference renormalization.** The file declares its own ``R <z0>`` (a 

17 NanoVNA writes 50 Ω, but a 75 Ω or 200 Ω calibration is legal). The plot has 

18 its own ``--z0``. Comparing raw Γ across differing references is a silent 

19 error, so a trace is always renormalized through the impedance it represents: 

20 ``Z = z0_file·(1+Γ)/(1−Γ)`` then ``Γ' = (Z−z0)/(Z+z0)``. 

21- **Grid alignment.** The measured grid is the VNA's (often hundreds of 

22 points); the model grid is the sweep's (often ~21). :meth:`MeasuredTrace.align` 

23 interpolates the measurement onto the sweep grid *restricted to the overlap*, 

24 so a measurement covering only part of the swept span renders over its own 

25 band and nothing is extrapolated. No overlap at all is an error, not an empty 

26 chart. 

27 

28Comparison happens at whatever plane the chart already plots — normally the 

29antenna feedpoint, so the VNA should be calibrated at the feedpoint too (a 

30design whose ``build_network()`` includes a station chain plots the station 

31plane instead, and a shack-end measurement matches that). Issue #639 makes the 

32plane an explicit choice. 

33"""

55# Largest |Γ| the derived impedance/SWR views will evaluate at — see 

56# MeasuredTrace.impedance. 1 − 1e-9 puts a measured open at ~1e11 Ω: off the 

57# top of any chart, but finite. 

58RHO_MAX = 1.0 - 1e-9 

59 

60 

61class BandOverlapError(ValueError): 

62 """The measured band and the swept band do not overlap at all.""" 

63 

64 

65@dataclass(frozen=True, eq=False) 

66class MeasuredTrace: 

67 """A measured one-port response on its own frequency grid. 

68 

69 ``freqs`` is MHz ascending (the unit the builders and sweep charts use, not 

70 the Hz of the Touchstone file); ``gamma`` is the complex reflection 

71 coefficient referenced to ``z0``; ``label`` names the trace in legends. 

72 ``eq=False`` for the same reason as :class:`~antennaknobs.touchstone.Touchstone` 

73 — array fields make value-equality meaningless. 

74 """ 

75 

76 freqs: np.ndarray 

77 gamma: np.ndarray 

78 z0: float 

79 label: str = "measured" 

80 

81 @classmethod 

82 def from_touchstone( 

83 cls, ts: Touchstone, *, label: str = "measured" 

84 ) -> "MeasuredTrace": 

85 """Build from a parsed 1-port :class:`Touchstone`. 

86 

87 Y- and Z-parameter files convert through ``s_at`` (evaluated at the 

88 file's own grid points, where the interpolation is exact), so an 

89 ``.s1p`` written as R+jX overlays just like one written as S11. 

90 """ 

91 if ts.nports != 1: 

92 raise ValueError( 

93 f"a measured overlay needs a 1-port (.s1p) file; got {ts.nports} ports" 

94 ) 

95 if ts.ptype == "S": 

96 gamma = ts.params[:, 0, 0].copy() 

97 else: 

98 gamma = np.array([ts.s_at(float(f))[0, 0] for f in ts.freqs]) 

99 return cls(freqs=ts.freqs / 1e6, gamma=gamma, z0=float(ts.z0), label=label) 

100 

101 # -- derived views ------------------------------------------------------ 

102 @property 

103 def impedance(self) -> np.ndarray: 

104 """Measured impedance (Ω) — ``z0·(1+Γ)/(1−Γ)``. 

105 

106 A real measurement can land at or just outside |Γ| = 1 (noise on a 

107 near-open, an imperfect calibration), where that expression divides by 

108 zero. The magnitude is clamped a hair inside the unit circle so the 

109 result stays finite — huge, which is the honest reading of a measured 

110 open, and JSON-safe for the web overlay, where a bare ``inf`` would 

111 produce a body the browser refuses to parse. 

112 """ 

113 g = self.gamma 

114 rho = np.abs(g) 

115 over = rho > RHO_MAX 

116 if over.any(): 

117 g = np.where(over, g * (RHO_MAX / np.maximum(rho, 1e-300)), g) 

118 return self.z0 * (1.0 + g) / (1.0 - g) 

119 

120 @property 

121 def swr(self) -> np.ndarray: 

122 # Same clamp as `impedance`, for the same reason: |Γ| ≥ 1 would divide 

123 # by zero or go negative. The trace pins at a large SWR instead. 

124 rho = np.minimum(np.abs(self.gamma), RHO_MAX) 

125 return (1.0 + rho) / (1.0 - rho) 

126 

127 def renormalized(self, z0: float) -> "MeasuredTrace": 

128 """This trace re-referenced to ``z0`` (identity when already there).""" 

129 if abs(z0 - self.z0) < 1e-12: 

130 return self 

131 z = self.impedance 

132 return MeasuredTrace( 

133 freqs=self.freqs, 

134 gamma=(z - z0) / (z + z0), 

135 z0=float(z0), 

136 label=self.label, 

137 ) 

138 

139 # -- alignment onto a sweep grid --------------------------------------- 

140 def align(self, freqs: np.ndarray) -> tuple[np.ndarray, np.ndarray]: 

141 """Interpolate onto sweep grid ``freqs`` (MHz), clipped to the overlap. 

142 

143 Returns ``(xs, gamma)`` over the sweep points that fall inside the 

144 measured span. Raises :class:`BandOverlapError` when the bands are 

145 disjoint; logs an informational message when the measurement covers the 

146 swept band only partly (a legitimate and common case — a single-band 

147 measurement against a multi-band sweep). 

148 """ 

149 freqs = np.asarray(freqs, dtype=float) 

150 f0, f1 = float(self.freqs[0]), float(self.freqs[-1]) 

151 tol = 1e-9 * max(abs(f1), 1.0) 

152 inside = (freqs >= f0 - tol) & (freqs <= f1 + tol) 

153 if not inside.any(): 

154 raise BandOverlapError( 

155 f"measured band [{f0:.6g}, {f1:.6g}] MHz does not overlap the " 

156 f"swept band [{freqs.min():.6g}, {freqs.max():.6g}] MHz — " 

157 "sweep the measured band, or measure the swept one" 

158 ) 

159 if not inside.all(): 

160 logger.info( 

161 "measured trace %r covers %d of %d sweep points " 

162 "([%.6g, %.6g] MHz of [%.6g, %.6g] MHz); drawing the overlap", 

163 self.label, 

164 int(inside.sum()), 

165 freqs.size, 

166 f0, 

167 f1, 

168 freqs.min(), 

169 freqs.max(), 

170 ) 

171 xs = freqs[inside] 

172 gamma = np.interp(xs, self.freqs, self.gamma.real) + 1j * np.interp( 

173 xs, self.freqs, self.gamma.imag 

174 ) 

175 return xs, gamma

178def parse_measured( 

179 text: str, *, z0: float | None = None, label: str = "measured" 

180) -> MeasuredTrace: 

181 """Parse Touchstone ``text`` into a measured overlay trace. 

182 

183 The port count is *inferred from the data* rather than taken on trust, so a 

184 two-port file that reached here under a one-port name gets the "needs a 

185 1-port" error instead of being silently read as three times as many 

186 frequency points. This is the shared entry point for the web upload 

187 (``/measured``) and :func:`read_measured`. 

188 """ 

189 trace = MeasuredTrace.from_touchstone(parse_touchstone(text), label=label) 

190 return trace if z0 is None else trace.renormalized(z0)

193def read_measured( 

194 path, *, z0: float | None = None, label: str | None = None 

195) -> MeasuredTrace: 

196 """Read a measured ``.s1p`` from an arbitrary filesystem path. 

197 

198 This is the CLI/operator entry point: unlike 

199 :func:`~antennaknobs.touchstone.read_touchstone` — which is confined to a 

200 *design's* own folder because designs are untrusted code — a measured 

201 overlay is a file the person at the keyboard names on the command line, so 

202 an ordinary path is right. ``z0`` renormalizes the trace to the chart's 

203 reference; ``label`` defaults to the file's stem. 

204 """ 

205 p = Path(path) 

206 if _nports_from_name(p.name) != 1: # validate the extension before reading 

207 raise ValueError( 

208 f"{p.name}: a measured overlay needs a 1-port .s1p file " 

209 "(a .s2p is a two-port block — see TouchstoneTwoPort)" 

210 ) 

211 return parse_measured(p.read_text(), z0=z0, label=label or p.stem)
