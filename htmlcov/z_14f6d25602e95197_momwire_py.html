32def _parity_for_solver(solver, solver_kwargs): 

33 """The basis types have fixed parity expectations: 

34 - BSplineSolver degree=1 → tent basis, even (feed straddles 2 segs) 

35 - BSplineSolver degree=2 → quadratic → odd 

36 - SinusoidalSolver → odd 

37 - SinusoidalGalerkinSolver → odd (same basis, Galerkin testing) 

38 Anything else falls through as "any" (no coercion).""" 

39 name = getattr(solver, "__name__", "") 

40 if name in ("SinusoidalSolver", "SinusoidalGalerkinSolver"): 

41 # SinusoidalGalerkinSolver subclasses SinusoidalSolver — same three-term 

42 # basis and the same delta-gap feed, so it inherits the odd-segment 

43 # rule. Load-bearing for the whole point of the class: the basis × 

44 # testing instrument (momwire#182) only reads cleanly when the two 

45 # sinusoidal cells differ in exactly ONE thing (the testing), and a 

46 # parity fall-through to "any" would have them differ in the mesh too. 

47 return "odd" 

48 if name in ("BSplineSolver", "HMatrixSolver", "ArrayBlockSolver"): 

49 # HMatrixSolver and ArrayBlockSolver are BSplineSolver subclasses (same 

50 # basis), so they share the degree-driven parity. Getting this right 

51 # matters for cross-solver comparison: a mismatched parity would build 

52 # a *different* mesh and silently invalidate any A/B against the dense 

53 # bspline path. 

54 degree = (solver_kwargs or {}).get("degree", 2) 

55 return "even" if int(degree) == 1 else "odd" 

56 return "any"

156# NEC's extended thin-wire kernel — the `EK` card — on the momwire bases that 

157# implement it (momwire 0.26.0: BSpline + its HMatrix/ArrayBlock subclasses, 

158# and the point-matched SinusoidalSolver; issues momwire#249/#259/#269/#270). 

159# Two combinations REFUSE upstream, both at solver construction, both with a 

160# NotImplementedError. They are re-raised here — same type, same shape — for 

161# one reason: the engine can say no BEFORE a fill is attempted, so the caller's 

162# error path (the portal's `NEC ERROR` frame, the web solve's error envelope) 

163# reports a named refusal instead of a traceback out of the middle of a solve. 

164def _extended_kernel_refusal(solver, solver_kwargs): 

165 """Why this solver + kwargs pair cannot run the extended kernel, or None. 

166 

167 * ``SinusoidalGalerkinSolver`` — momwire#246/#233: NEC's ``EKSCX`` has no 

168 counterpart for the Galerkin fill's folded third source shape, so momwire 

169 ships EK on the point-matched sibling only and refuses rather than 

170 silently serving a reduced-kernel answer under an EK request. 

171 * ``use_singular_enrichment=True`` — momwire#271/#249 follow-up C: the 

172 enrichment DOFs carry their own singular quadrature and bypass the moment 

173 kernels entirely, and the O(a²) tube expansion was never derived for the 

174 s^(-1/2) shapes. 

175 

176 Everything else — including finite ground, which momwire#269 lifted — is 

177 served. 

178 """ 

179 name = getattr(solver, "__name__", str(solver)) 

180 if name == "SinusoidalGalerkinSolver": 

181 return ( 

182 "the extended thin-wire kernel is not available on the " 

183 "sinusoidal-galerkin basis: momwire implements NEC's extended " 

184 "kernel on the point-matched SinusoidalSolver only (momwire#246). " 

185 "Use the sinusoidal basis for an extended-kernel run, or drop the " 

186 "kernel request for the reduced-kernel Galerkin fill" 

187 ) 

188 if (solver_kwargs or {}).get("use_singular_enrichment"): 

189 return ( 

190 "the extended thin-wire kernel and singular enrichment cannot be " 

191 "used together (momwire#271): the enrichment DOFs bypass the " 

192 "moment kernels the extended kernel corrects. Turn one of the two " 

193 "off" 

194 ) 

195 return None

515 reduces to driven-port impedance. Validates the name/port contract in 

516 both directions: every PortOnWire resolves to a translated feed name, 

517 and every named wire is referenced by a PortOnWire (issue #578 — an

524 for name, port in net.ports.items(): 

525 if isinstance(port, PortOnWire): 

526 if name not in feed_name_to_idx: 

527 raise ValueError( 

528 f"network port {name!r} is a PortOnWire but no wire in " 

529 f"build_wires() carries that name; named wires: " 

530 f"{sorted(feed_name_to_idx)}" 

531 )

534 # End ports follow the gap feeds (matching momwire's solver-side 

535 # [feeds..., junction_ports...] Y ordering, issue #579), then the 

536 # virtual ports. 

537 next_idx = len(self._feeds) 

538 for name, _wire, _end in self._end_ports: 

539 port_to_idx[name] = next_idx 

540 next_idx += 1 

541 for name, port in net.ports.items(): 

542 if isinstance(port, PortVirtual): 

543 port_to_idx[name] = next_idx 

544 next_idx += 1 

545 

546 self._reducer = NetworkReducer(net, port_to_idx, next_idx) 

547 

548 # Finite-gap (distributed) ports — issue #477. A distributed port 

549 # is realised as one delta-gap sub-feed per SEGMENT of its named 

550 # wire, voltages split by length (constant E over the wire's fixed 

551 # physical extent), and the antenna Y contracted back to port 

552 # granularity by congruence: Y_port = Wᵀ·Y_sub·W, with W's column 

553 # for a port holding the length weights h_i/L (uniform segments → 

554 # 1/S). The weighted current readout Σ wᵢ·Iᵢ is the bilinear dual 

555 # of the voltage split, so port power Σ V*·I is preserved and the 

556 # reducer sees an ordinary port row. Everything downstream — 

557 # reducer, sweeps, excited state — keeps feed-granularity code 

558 # untouched; only the solver feed list and the Y contraction know. 

559 dist_idx = { 

560 i 

561 for i, n in enumerate(self._feed_names) 

562 if n 

563 and isinstance(net.ports.get(n), PortOnWire) 

564 and net.ports[n].distributed 

565 } 

566 if dist_idx: 

567 self._build_feed_expansion(dist_idx)

569 def _build_feed_expansion(self, dist_idx): 

570 """Expanded solver-feed positions + the (n_sub × n_feeds) weight 

571 matrix W for distributed ports. Non-distributed feeds keep a single 

572 sub-feed with weight 1 (W's column is a unit vector), so the 

573 contraction is exact for every port kind. 

574 

575 Every weight in a port's column carries that port's walk-direction 

576 sign d_k (issue #580): V_solver = W·V applies d_k to the applied 

577 voltage(s) and Y_port = Wᵀ·Y_sub·W applies d_k to the port's row AND 

578 column, so a single signed W normalizes both congruence sides to the 

579 authored p0→p1 convention (all sub-feeds of a distributed port share 

580 its d_k — they lie on the same walked edge).""" 

581 pos, cols = [], [] 

582 for k, ((pl, arc, _v), (epl, eidx)) in enumerate( 

583 zip(self._feeds, self._feed_edges) 

584 ): 

585 d = float(self._feed_dirs[k]) 

586 if k in dist_idx: 

587 poly = self._polylines[epl] 

588 lens = np.linalg.norm(np.diff(poly, axis=0), axis=1) 

589 arc0 = float(lens[:eidx].sum()) 

590 length = float(lens[eidx]) 

591 n_sub = int(self._edge_segments[epl][eidx]) 

592 for i in range(n_sub): 

593 pos.append((epl, arc0 + (i + 0.5) * length / n_sub)) 

594 cols.append((k, d / n_sub)) 

595 else: 

596 pos.append((pl, arc)) 

597 cols.append((k, d)) 

598 W = np.zeros((len(pos), len(self._feeds))) 

599 for row, (k, w) in enumerate(cols): 

600 W[row, k] = w 

601 self._exp_feed_pos = pos 

602 self._feed_W = W

604 def _solver_feeds(self, voltages=None): 

605 """The feed list handed to momwire solvers: per-feed (wire, arc, V), 

606 expanded to sub-feeds with length-split voltages when distributed 

607 ports exist. ``voltages`` (per ORIGINAL feed, in feed order) 

608 defaults to the translated feed voltages.""" 

609 if voltages is None: 

610 voltages = [v for *_, v in self._feeds] 

611 if self._feed_W is None: 

612 return [(w, a, complex(v)) for (w, a, _v), v in zip(self._feeds, voltages)] 

613 v_exp = self._feed_W @ np.asarray(voltages, dtype=np.complex128) 

614 return [(w, a, complex(v)) for (w, a), v in zip(self._exp_feed_pos, v_exp)]

616 def _contract_y(self, Y): 

617 """Contract a solver Y (sub-feed granularity) to port granularity; 

618 the signed W also normalizes each port's sign convention to the 

619 authored wire direction (issue #580). Identity when no distributed 

620 ports exist and every port edge was walked as authored. End-port 

621 rows (issue #579) trail the solver's feed rows and pass through 

622 unweighted — a junction-node port has no sub-feed expansion and no 

623 walk-sign ambiguity (a KCL row's outflow convention is geometric) — 

624 so W extends by an identity block. Works on one matrix or a swept 

625 (n_k, n, n) stack.""" 

626 if self._feed_W is None: 

627 return Y 

628 W = self._feed_W 

629 n_end = len(self._end_port_junctions) 

630 if n_end: 

631 W = np.block( 

632 [ 

633 [W, np.zeros((W.shape[0], n_end))], 

634 [np.zeros((n_end, W.shape[1])), np.eye(n_end)], 

635 ] 

636 ) 

637 if Y.ndim == 3: 

638 return np.einsum("ia,kij,jb->kab", W, Y, W) 

639 return W.T @ Y @ W

727 def _resolve_feed_voltages(self, Y_total): 

728 """Return the full per-feed voltage vector V with passive ports' 

729 voltages set so I_ext=0 there. The driven ports keep their applied V.""" 

730 n = Y_total.shape[0] 

731 driven = [i for i in range(n) if i not in self._tl_passive_feed_idx] 

732 passive = sorted(self._tl_passive_feed_idx) 

733 v_driven = np.array([self._feeds[i][2] for i in driven], dtype=np.complex128) 

734 V = np.empty(n, dtype=np.complex128) 

735 V[driven] = v_driven 

736 if passive: 

737 Y_pp = Y_total[np.ix_(passive, passive)] 

738 Y_pd = Y_total[np.ix_(passive, driven)] 

739 V[passive] = np.linalg.solve(Y_pp, -Y_pd @ v_driven) 

740 return V, driven

742 def _impedance_from_y(self, Y_total): 

743 """Driving-point Z at each driven port, with passive (TL-only) ports 

744 floating (I_ext=0). Matches PyNECEngine's per-driven-port semantics 

745 when all drivers are excited simultaneously.""" 

746 V, driven = self._resolve_feed_voltages(Y_total) 

747 I = Y_total @ V 

748 return [complex(V[i] / I[i]) for i in driven]

766 once per (wavelength, feed-voltage) tuple, caching on the engine

787 # the wire-loss amendment below is applied exactly once per call. 

788 self._amend_wire_loss(sim, coeffs, z)

791 def _amend_wire_loss(self, sim, coeffs, z): 

792 """Fold ohmic wire loss (momwire#131 distributed loading) into the 

793 power bookkeeping (issue #317): a "wire loss (I²R)" budget row and 

794 the matching efficiency drop. The loss already lives inside P_in 

795 (the loading raised the driving-point R), so gain = 4π·U/P_in needs 

796 no change — this keeps the *reported* efficiency and budget rows 

797 truthful about where those watts went. Insulation is reactive and 

798 contributes nothing; lossless designs are untouched.""" 

799 self._excited_p_wire = 0.0 

800 if not self._loading_kwargs: 

801 return 

802 p_wire, _per_wire = sim.wire_loss_power(coeffs) 

803 if p_wire <= 0.0: 

804 return 

805 self._excited_p_wire = float(p_wire) 

806 self._excited_power_budget = list(self._excited_power_budget) + [ 

807 ("wire loss (I²R)", float(p_wire)) 

808 ] 

809 p_in = self._excited_p_in 

810 if p_in is None: # plain path records no port-model power 

811 p_in = self._p_in_from_excited(sim, z) 

812 if p_in > 0.0: 

813 self._excited_efficiency = max( 

814 0.0, min(1.0, self._excited_efficiency - p_wire / p_in) 

815 )

817 @staticmethod 

818 def _p_in_from_excited(sim, z): 

819 """Source input power ½·Σ|V_f|²·Re(Z_f)/|Z_f|² from an excited 

820 solve's driving-point impedances (the plain-path formula shared by 

821 `input_power` and the wire-loss amendment).""" 

822 z_arr = np.atleast_1d(np.asarray(z, dtype=np.complex128)) 

823 volts = np.array([complex(v) for *_, v in sim.feeds], dtype=np.complex128) 

824 with np.errstate(divide="ignore", invalid="ignore"): 

825 terms = 0.5 * np.abs(volts) ** 2 * z_arr.real / np.abs(z_arr) ** 2 

826 return float(np.sum(terms[np.isfinite(terms)]))

863 def impedance_sweep(self, freqs): 

864 freqs = np.asarray(freqs, dtype=float) 

865 if freqs.ndim != 1 or freqs.size == 0: 

866 raise ValueError("freqs must be a 1-D non-empty array") 

867 s = self._make_solver(wavelength=self._wavelength_for(freqs[0])) 

868 k_array = 2.0 * np.pi * freqs * 1e6 / C_LIGHT 

869 if self._network is not None: 

870 Y_swept = self._contract_y( 

871 np.asarray(s.compute_y_matrix_swept(k_array), dtype=np.complex128) 

872 ) # (n_k, n_real, n_real) at port granularity 

873 zs = np.empty((freqs.size, self._reducer.n_driven), dtype=np.complex128) 

874 for ki, freq in enumerate(freqs): 

875 # A lossless line can sit on its pole at ONE sample of a sweep 

876 # (issue #647); poison that sample rather than the sweep. 

877 z = poison_singular_sample( 

878 self._reduce_at, 

879 Y_swept[ki], 

880 self._wavelength_for(freq), 

881 where=f" at {freq:.6g} MHz", 

882 ) 

883 zs[ki] = np.nan if z is None else z 

884 return zs 

885 if self._tls: 

886 # Batched Y at every frequency, then per-k TL stamping (βL is 

887 # frequency-dependent) and driven-port reduction. The Y assembly 

888 # is amortised across frequencies via the upstream swept solve; 

889 # the per-k post-processing is O(n_p³) and dwarfed by the solve. 

890 Y_swept = self._contract_y( 

891 np.asarray(s.compute_y_matrix_swept(k_array), dtype=np.complex128) 

892 ) # (n_k, n_p, n_p) — contraction applies the issue-#580 port 

893 # signs (legacy TLs have no distributed ports, so W is at most 

894 # the diagonal sign matrix; identity when W is None) 

895 n_driven = sum( 

896 1 for i in range(Y_swept.shape[1]) if i not in self._tl_passive_feed_idx 

897 ) 

898 zs = np.empty((freqs.size, n_driven), dtype=np.complex128) 

899 for ki, freq in enumerate(freqs): 

900 Y_total = self._apply_tls(Y_swept[ki], self._wavelength_for(freq)) 

901 zs[ki] = self._impedance_from_y(Y_total) 

902 return zs 

903 zs = s.compute_impedance_swept(k_array) 

904 # Single-feed: (n_k,); multi-feed: (n_k, n_feeds). Normalise to 

905 # (n_k, n_feeds) to match PyNECEngine. 

906 zs = np.asarray(zs) 

907 if zs.ndim == 1: 

908 zs = zs.reshape(-1, 1) 

909 return zs

911 def _make_excited_solver(self, *, wavelength): 

912 """Build a solver whose feed voltages match the actual excitation: 

913 for plain designs, just the build_wires() voltages; for TL or 

914 Network designs, the per-port voltages after the network reduction 

915 so basis coefficients reflect the branch-induced port voltages. 

916 Without this, network-spec designs (where every named feed carries 

917 a placeholder V=0) would solve with no excitation — every basis 

918 coefficient is zero and `compute_impedance`'s V/I returns NaN.""" 

919 if self._network is not None: 

920 # excited_state solves the same MNA system as the impedance 

921 # path, so resistive loads shape the current through their 

922 # physical series BC; it also returns the radiation efficiency 

923 # and the source input power the far field uses to normalise 

924 # gain. 

925 Y = self._compute_y_matrix(wavelength) 

926 ( 

927 V_full, 

928 self._excited_efficiency, 

929 self._excited_p_in, 

930 self._excited_power_budget, 

931 ) = self._reducer.excited_state(Y, wavelength) 

932 feeds_resolved = self._solver_feeds( 

933 [complex(V_full[i]) for i in range(len(self._feeds))] 

934 ) 

935 # End-port voltages (issue #579) follow the gap feeds in the 

936 # reducer's port ordering; force them in the excited solver so 

937 # the current distribution / far field see the branch-induced 

938 # node drive exactly like the gap ports do. 

939 n_f = len(self._feeds) 

940 end_port_voltages = [ 

941 complex(V_full[n_f + k]) for k in range(len(self._end_ports)) 

942 ] 

943 elif self._tls: 

944 self._excited_efficiency = 1.0 

945 self._excited_power_budget = [] 

946 Y = self._compute_y_matrix(wavelength) 

947 Y_total = self._apply_tls(Y, wavelength) 

948 V, driven = self._resolve_feed_voltages(Y_total) 

949 # Source input power at the driven ports of the TL-stamped port 

950 # model; the TLs are lossless so this equals the power entering 

951 # the wires. 

952 I_ports = Y_total @ V 

953 self._excited_p_in = 0.5 * float( 

954 sum((V[i] * np.conj(I_ports[i])).real for i in driven) 

955 ) 

956 # Through _solver_feeds so each port's resolved voltage picks up 

957 # its walk-direction sign (issue #580) exactly like the network 

958 # path; identity when every feed was walked as authored. 

959 feeds_resolved = self._solver_feeds([complex(v) for v in V]) 

960 else: 

961 self._excited_efficiency = 1.0 

962 self._excited_power_budget = [] 

963 # Plain path: no port model here; input_power() derives P_in from 

964 # the excited solve's driving-point impedance(s) on demand. 

965 self._excited_p_in = None 

966 return self._make_solver(wavelength=wavelength) 

967 kw = {} 

968 if self._end_port_junctions: 

969 # Legacy build_tls designs have no end ports, so this only 

970 # fires on the network path where end_port_voltages is set. 

971 kw["junction_ports"] = [ 

972 (j, v) for j, v in zip(self._end_port_junctions, end_port_voltages) 

973 ] 

974 return self._solver( 

975 wires=self._polylines, 

976 n_per_edge_per_wire=self._edge_segments, 

977 feeds=feeds_resolved, 

978 wavelength=wavelength, 

979 wire_radius=self._wire_radius, 

980 ground_z=self._ground_z, 

981 junctions=self._junctions or None, 

982 cancel=self._cancel, 

983 **kw, 

984 **self._loading_kwargs, 

985 **self._ground_solver_kwargs(), 

986 **self._kernel_solver_kwargs(), 

987 **self._solver_kwargs, 

988 )

990 def input_power(self): 

991 """Input power 1/2·Re(Σ V_f·I_f*) in watts over the SOURCE feeds of 

992 the excited solve — the gain normaliser (gain = 4π·U/P_in). Power 

993 burned in resistive loads and any ground absorption stay inside 

994 P_in (that is what makes 4π·U/P_in GAIN rather than directivity). 

995 

996 Network / TL designs record it while resolving the excitation in 

997 `_make_excited_solver` (a load port is excited too, but it is not a 

998 source, so its absorbed power must not be summed — the port model 

999 separates the two). The plain path derives it from the excited 

1000 driving-point impedance(s): I_f = V_f/Z_f, so each feed contributes 

1001 ½·|V_f|²·Re(Z_f)/|Z_f|². 

1002 """ 

1003 wavelength = self._wavelength_for(self.builder.freq) 

1004 sim, _coeffs, z = self._solved_excited(wavelength) 

1005 p_in = getattr(self, "_excited_p_in", None) 

1006 if p_in is not None: 

1007 return float(p_in) 

1008 return self._p_in_from_excited(sim, z)

1030 knot_currents = sim.currents_at_knots(coeffs) 

1031 out = []

1063 def _segment_dipoles(self, sim, coeffs): 

1064 """Returns (mid, dr, i_mid) — concatenated per-segment midpoints, 

1065 edge vectors, and midpoint currents from the MoM solution.""" 

1066 knot_currents = sim.currents_at_knots(coeffs) 

1067 mids, drs, i_mids = [], [], [] 

1068 for w_idx, polyline in enumerate(self._polylines): 

1069 knots = _polyline_knots(polyline, self._edge_segments[w_idx])

1071 drs.append(knots[1:] - knots[:-1])

1073 i_mids.append(0.5 * (cur[1:] + cur[:-1]))

1076 np.concatenate(drs, axis=0),

1080 def _evaluate_M_perp(self, mid, dr, i_mid, k, theta, phi, freq_hz): 

1081 """|M_perp(θ,φ)|² on the (theta, phi) grids (radians). 

1082 

1083 With ground enabled, adds the geometric-image contribution with PEC 

1084 polarity, then layers Fresnel coefficients on the reflected wave so 

1085 ρ_h=−1, ρ_v=+1 recovers the PEC limit exactly. Returns a real 

1086 (n_theta, n_phi) array.""" 

1087 sin_t, cos_t = np.sin(theta), np.cos(theta) 

1088 cos_p, sin_p = np.cos(phi), np.sin(phi) 

1089 

1090 rx = sin_t[:, None] * cos_p[None, :] 

1091 ry = sin_t[:, None] * sin_p[None, :] 

1092 rz = np.broadcast_to(cos_t[:, None], rx.shape) 

1093 rhat = np.stack([rx, ry, rz], axis=-1) 

1094 

1095 phase = k * np.einsum("ijc,nc->ijn", rhat, mid) 

1096 expp = np.exp(1j * phase) 

1097 weighted = i_mid[:, None] * dr 

1098 M = np.einsum("ijn,nc->ijc", expp, weighted) 

1099 m_dot_r = np.sum(M * rhat, axis=-1) 

1100 M_perp = M - m_dot_r[..., None] * rhat 

1101 

1102 if self._ground is None: 

1103 return np.sum(M_perp.real**2 + M_perp.imag**2, axis=-1) 

1104 

1105 # Geometric image — horizontal current flipped, vertical preserved, 

1106 # mirrored across z = ground_z. 

1107 z0 = self._ground_z 

1108 mid_img = mid.copy() 

1109 mid_img[:, 2] = 2 * z0 - mid[:, 2] 

1110 dr_img = dr * np.array([-1.0, -1.0, 1.0]) 

1111 weighted_img = i_mid[:, None] * dr_img 

1112 phase_img = k * np.einsum("ijc,nc->ijn", rhat, mid_img) 

1113 expp_img = np.exp(1j * phase_img) 

1114 M_img = np.einsum("ijn,nc->ijc", expp_img, weighted_img) 

1115 m_img_dot_r = np.sum(M_img * rhat, axis=-1) 

1116 M_img_perp = M_img - m_img_dot_r[..., None] * rhat 

1117 

1118 if self._ground[0] == "pec": 

1119 M_perp = M_perp + M_img_perp 

1120 return np.sum(M_perp.real**2 + M_perp.imag**2, axis=-1) 

1121 

1122 # ("finite", eps_r, sigma) / ("terrain", Terrain): polarisation basis 

1123 # at each ray and Fresnel reflection on the image wave. 

1124 # Vertical (TM, in-plane) and horizontal (TE, out-of-plane) 

1125 # polarisation unit vectors for the reflected ray, written straight 

1126 # from the azimuth/zenith angles rather than from the horizontal 

1127 # projection s = sin(theta). Dividing by s degenerates at the zenith 

1128 # (s -> 0, the plane of incidence is undefined): the old s->1.0 guard 

1129 # collapsed BOTH vectors to zero there, dropping the reflected wave and 

1130 # reading ~3 dB low. The phi-limit h_hat=(-sin phi, cos phi, 0), 

1131 # v_hat=(-cos phi, -sin phi, 0) is exact, unit, orthonormal to rhat, 

1132 # and recovers the PEC limit (rho_h=-1, rho_v=+1) at theta=0. 

1133 s = np.sqrt(rx * rx + ry * ry) # = sin(theta); feeds sin2_ti below 

1134 cos_p_g = np.broadcast_to(cos_p[None, :], rx.shape) 

1135 sin_p_g = np.broadcast_to(sin_p[None, :], rx.shape) 

1136 cos_t_g = np.broadcast_to(cos_t[:, None], rx.shape) 

1137 sin_t_g = np.broadcast_to(sin_t[:, None], rx.shape) 

1138 h_hat = np.stack([-sin_p_g, cos_p_g, np.zeros_like(rx)], axis=-1) 

1139 v_hat = np.stack([-cos_p_g * cos_t_g, -sin_p_g * cos_t_g, sin_t_g], axis=-1) 

1140 M_img_h = np.sum(M_img_perp * h_hat, axis=-1) 

1141 M_img_v = np.sum(M_img_perp * v_hat, axis=-1) 

1142 

1143 omega = 2 * np.pi * freq_hz 

1144 if self._ground[0] == "terrain": 

1145 # Faceted terrain (issue #534): per direction, find the facet 

1146 # the specular point lands on, shift the image plane to the 

1147 # facet surface (an extra 2·k·z_f·cosθ path phase — z_f ≤ 0 

1148 # below the crest, so the reflected wave rides the full 

1149 # effective height), tilt the incidence angle by the facet 

1150 # slope, and evaluate Fresnel with the facet's medium. The 

1151 # image itself stays the z=ground_z mirror built above; the 

1152 # plane shift factors out as a per-direction scalar phase. 

1153 terrain = self._ground[1] 

1154 w = np.abs(i_mid) * np.linalg.norm(dr, axis=1) 

1155 wsum = float(np.sum(w)) 

1156 h_ref = ( 

1157 float( 

1158 (np.sum(w * mid[:, 2]) / wsum) if wsum > 0 else np.mean(mid[:, 2]) 

1159 ) 

1160 - z0 

1161 ) 

1162 sec_idx = terrain.sector_for(np.degrees(phi)) 

1163 z_f = np.empty(rx.shape) 

1164 beta_g = np.empty(rx.shape) 

1165 eps_g = np.empty(rx.shape) 

1166 sig_g = np.empty(rx.shape) 

1167 for i, sector in enumerate(terrain.sectors): 

1168 cols = sec_idx == i 

1169 if not np.any(cols): 

1170 continue 

1171 zf_t, b_t, e_t, s_t = specular_cut(sector, theta, h_ref) 

1172 z_f[:, cols] = zf_t[:, None] 

1173 beta_g[:, cols] = b_t[:, None] 

1174 eps_g[:, cols] = e_t[:, None] 

1175 sig_g[:, cols] = s_t[:, None] 

1176 pf = np.exp(2j * k * z_f * cos_t_g) 

1177 M_img_h = M_img_h * pf 

1178 M_img_v = M_img_v * pf 

1179 th_loc = np.clip(theta[:, None] - beta_g, 0.0, np.pi / 2) 

1180 # On untilted facets keep the exact same trig route as the flat 

1181 # branch (rz / s·s) so the single-flat-facet terrain reproduces 

1182 # ("finite", eps, sigma) bit-for-bit — issue #534 gate 1. 

1183 cos_ti = np.where(beta_g == 0.0, rz, np.cos(th_loc)) 

1184 sin2_ti = np.where(beta_g == 0.0, s * s, np.sin(th_loc) ** 2) 

1185 eps_c = eps_g - 1j * sig_g / (omega * EPS0) 

1186 else: 

1187 _, eps_r, sigma = self._ground 

1188 eps_c = eps_r - 1j * sigma / (omega * EPS0) 

1189 cos_ti = rz 

1190 sin2_ti = s * s 

1191 Q = np.sqrt(eps_c - sin2_ti) 

1192 rho_h = (cos_ti - Q) / (cos_ti + Q) 

1193 rho_v = (eps_c * cos_ti - Q) / (eps_c * cos_ti + Q) 

1194 

1195 # PEC reflection corresponds to ρ_h=−1, ρ_v=+1. The image we built 

1196 # already has the PEC sign convention baked in, so we need the 

1197 # Fresnel-vs-PEC ratio per polarisation: 

1198 # reflected_h = (−ρ_h) · M_img_h # PEC was +1·M_img_h 

1199 # reflected_v = (+ρ_v) · M_img_v # PEC was +1·M_img_v 

1200 M_refl = (rho_v * M_img_v)[..., None] * v_hat - (rho_h * M_img_h)[ 

1201 ..., None 

1202 ] * h_hat 

1203 M_perp = M_perp + M_refl 

1204 return np.sum(M_perp.real**2 + M_perp.imag**2, axis=-1)

1206 def far_field(self, *, n_theta=90, n_phi=360, del_theta=1, del_phi=1): 

1207 self._raise_if_cancelled() 

1208 assert 90 % n_theta == 0 and 90 == del_theta * n_theta 

1209 assert 360 % n_phi == 0 and 360 == del_phi * n_phi 

1210 

1211 wavelength = self._wavelength_for(self.builder.freq) 

1212 k = 2.0 * np.pi / wavelength 

1213 freq_hz = self.builder.freq * 1e6 

1214 

1215 sim, coeffs, _z = self._solved_excited(wavelength) 

1216 mid, dr, i_mid = self._segment_dipoles(sim, coeffs) 

1217 

1218 # Gain normaliser from the source input power (same convention as the 

1219 # web solve path): gain = 4π·U/P_in = η₀k²/(8π·P_in)·|M_perp|². Load 

1220 # loss lives inside P_in, so terminated antennas come out as GAIN with 

1221 # no efficiency multiply. 

1222 p_in = self.input_power() 

1223 if p_in > 0: 

1224 directivity_norm = ETA0 * k * k / (8.0 * np.pi * p_in) 

1225 else: 

1226 # Defensive fallback (a pathological R_in ≤ 0): normalise by the 

1227 # integrated pattern instead. Cell-centred grid over the sphere, 

1228 # upper hemisphere only with ground (the image contribution is 

1229 # already folded into |M_perp|² there). 

1230 n_th_int, n_ph_int = 90, 180 

1231 if self._ground is not None: 

1232 theta_int = (np.arange(n_th_int) + 0.5) * (np.pi / 2 / n_th_int) 

1233 dtheta = np.pi / 2 / n_th_int 

1234 else: 

1235 theta_int = (np.arange(n_th_int) + 0.5) * (np.pi / n_th_int) 

1236 dtheta = np.pi / n_th_int 

1237 phi_int = np.arange(n_ph_int) * (2 * np.pi / n_ph_int) 

1238 dphi = 2 * np.pi / n_ph_int 

1239 mag2_int = self._evaluate_M_perp( 

1240 mid, dr, i_mid, k, theta_int, phi_int, freq_hz 

1241 ) 

1242 p_rad = float(np.sum(mag2_int * np.sin(theta_int)[:, None]) * dtheta * dphi) 

1243 if p_rad <= 0: 

1244 raise RuntimeError("computed zero radiated power") 

1245 efficiency = getattr(self, "_excited_efficiency", 1.0) 

1246 directivity_norm = 4 * np.pi / p_rad * efficiency 

1247 

1248 # Evaluate on the user grid (NEC convention: θ from 0 to 90−Δθ). 

1249 theta_deg = np.linspace(0, 90 - del_theta, n_theta) 

1250 phi_deg = np.linspace(0, 360, n_phi + 1) 

1251 theta_user = np.deg2rad(theta_deg) 

1252 phi_user = np.deg2rad(phi_deg) 

1253 

1254 mag2_user = self._evaluate_M_perp( 

1255 mid, dr, i_mid, k, theta_user, phi_user, freq_hz 

1256 ) 

1257 D = directivity_norm * mag2_user 

1258 # Floor before log so points where M_perp is exactly zero (poles, 

1259 # nulls below quantisation) don't produce −inf. 

1260 dBi = 10.0 * np.log10(np.maximum(D, 1e-30)) 

1261 

1262 rings = dBi.tolist() 

1263 return FarField( 

1264 rings=rings, 

1265 max_gain=float(np.max(dBi)), 

1266 min_gain=float(np.min(dBi)), 

1267 thetas=theta_deg, 

1268 phis=phi_deg, 

1269 )
