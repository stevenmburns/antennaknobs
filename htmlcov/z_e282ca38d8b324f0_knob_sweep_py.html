1"""How a design says its knob should be swept (AK#1975). 

2 

3A knob sweep (the workbench's "Sweep this knob…", `antennaknobs sweep 

4--param`) defaults to the knob's whole slider range in a few linear points. 

5For most knobs that is right. For a tuner's coil or capacitor it is not: the 

6MFJ-926B capacitor runs 0–3961 pF, so 11 linear points step 396 pF, and the 

720 m answer (39 pF) falls between the first two (Steve, 2026-10-10, tuning 

8`wire.doublet_remote_tuner` by hand on the Smith chart). 

9 

10So a design may add a ``sweep`` entry to a knob's ``ui_params``: 

11 

12 "tuner_c_pF": {"min": 0.0, "max": 3961.0, 

13 "sweep": {"lo": 1.0, "log": True, "points": 81}}, 

14 

15any subset of ``lo``, ``hi`` (default: the knob's min / max), ``points`` and 

16``log``. The workbench's knob sweep and the command line's ``--param`` sweep 

17both start from it; an explicit range, point count or ``--log`` still wins. 

18A hint is distrusted like any authored value: one that cannot be swept (a 

19log range not above zero, a point count out of range, lo not below hi) is 

20dropped whole, never half-applied. 

21""" 

22 

23from __future__ import annotations 

24 

25from collections.abc import Mapping 

26 

27# The workbench's own bounds on a sweep's points (lib/paramSweep.ts 

28# MIN_POINTS / MAX_POINTS). 

29MIN_POINTS = 2 

30MAX_POINTS = 201 

31 

32# A matching network's part (a tuner's coil or capacitor): log-spaced, finely 

33# enough that one sweep traces the part's whole path on the Smith chart. A 

34# part's reactance runs over decades of its value, so even steps in the value 

35# would crowd one end; 81 points put about 40 a decade on a 1-100 pF range. 

36TUNER_PART_SWEEP = {"log": True, "points": 81} 

37 

38 

39def sanitize_sweep_hint(raw, lo_default=None, hi_default=None) -> dict | None: 

40 """The usable part of a ``ui_params`` ``sweep`` entry, or None. 

41 

42 Returns only the keys the design gave, so a front end can tell "the 

43 design says linear" from "the design says nothing". ``lo_default`` / 

44 ``hi_default`` (the knob's min / max) are only used to check that the 

45 range it implies can be swept.""" 

46 if not isinstance(raw, Mapping): 

47 return None 

48 out: dict = {} 

49 for key in ("lo", "hi"): 

50 if key in raw: 

51 v = raw[key] 

52 if isinstance(v, bool) or not isinstance(v, (int, float)): 

53 return None 

54 out[key] = float(v) 

55 if "points" in raw: 

56 n = raw["points"] 

57 if isinstance(n, bool) or not isinstance(n, int): 

58 return None 

59 if not MIN_POINTS <= n <= MAX_POINTS: 

60 return None 

61 out["points"] = n 

62 if "log" in raw: 

63 if not isinstance(raw["log"], bool): 

64 return None 

65 out["log"] = raw["log"] 

66 lo = out.get("lo", lo_default) 

67 hi = out.get("hi", hi_default) 

68 if lo is not None and hi is not None: 

69 if not float(lo) < float(hi): 

70 return None 

71 if out.get("log") and not float(lo) > 0.0: 

72 return None 

73 return out or None 

74

76def knob_sweep_hint(builder, name: str) -> dict | None: 

77 """A knob's sanitised sweep hint, read off its builder's ``ui_params``.""" 

78 # Lazy: band_opt pulls scipy (via opt), and the web adapter imports this 

79 # module at startup for `sanitize_sweep_hint` alone. 

80 from .band_opt import _ui_meta 

81 

82 meta = _ui_meta(builder, name) 

83 if meta is None: 

84 return None 

85 return sanitize_sweep_hint(meta.get("sweep"), meta.get("min"), meta.get("max"))
