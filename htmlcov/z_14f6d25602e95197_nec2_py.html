1"""NEC-2 as an external-oracle engine over a text coupling — issue #1354. 

2 

3antennaknobs already has NEC-2 physics in process, through `PyNECEngine` and 

4the `pynec-accel` wheel. What it cannot do is SHIP it: pynec-accel wraps 

5nec2++, which is GPLv2, so putting it in the frozen Windows workbench (#1349) 

6would make that zip a combined work with the source-offer and notice 

7obligations that carries — and would reverse the stance the published Docker 

8image already takes (``INCLUDE_PYNEC=0``, MIT/BSD only). Decision 2026-09-10 

9(Steve): leave it out. 

10 

11This is the way round it, and it is the shape `NEC5Engine` already proved: drive 

12a **user-supplied** console binary over text — a deck written to a temp dir, the 

13binary run as a subprocess, its printout parsed — and never link to it. 

14antennaknobs then distributes nothing GPL, the in-process PyNEC lane stays for 

15people who install it, and the frozen bundle gains NEC-2 the moment a user 

16points it at a binary they very likely already have: 4nec2 ships one. 

17 

18The deck is written by `nec_export.export_nec`, NOT by a second writer. That 

19module was built for the download button and emits the same NEC-2 dialect a 

20`PyNECEngine` solve resolves to, wire tuple for wire tuple, so a deck this 

21engine runs is a text twin of what PyNEC would have solved in process. A second 

22writer here is how the two would drift apart, and the drift would look like an 

23engine disagreement. 

24 

25Two invocation forms, chosen by PROBING 

26--------------------------------------- 

27There is no one NEC-2 command line, and the difference is not something a file 

28name can be trusted to say (a renamed binary, a wrapper script, a symlink): 

29 

30* the **argument** form, ``nec2c -i deck.nec -o deck.out``, which nec2c and 

31 nec2++ take; 

32* the **stdin** form, the input and output file names on standard input — what 

33 4nec2's ``nec2dxs*.exe`` reads, and what NEC5CL reads too, so 

34 `nec5.run_deck`'s shape carries over. 

35 

36So `run_deck` tries the argument form, then the stdin form, and remembers which 

37one answered for that binary (keyed on its identity, not its name, so replacing 

38it in place re-probes). Two details are load-bearing. Stdin is closed on the 

39argument form — a binary that accepts the flags AND would also read a file name 

40from stdin otherwise hangs on a pipe nobody writes to. And a printout is taken 

41from the output FILE if one appeared, else from stdout: some nec2c builds write 

42the report to standard output and leave no file, which is a working binary and 

43must not read as a failed run. 

44 

45Exit codes are not trusted, for the reason NEC-5's wrapper does not trust them 

46either: these are Fortran (or Fortran-descended) programs that ``STOP`` on input 

47errors. The health signal is a parseable printout. 

48 

49Printout format 

50--------------- 

51Pinned against momwire's portal, which writes a column-exact NEC-2 printout 

52(``momwire/tests/fixtures/nec_portal/*.out``) because SimNEC reads it as nec2c. 

53The three blocks this engine parses, and how they differ from NEC-5's — which is 

54why the parsers are not shared: 

55 

56* ``ANTENNA INPUT PARAMETERS``: ``TAG SEG Vre Vim Ire Iim Zre Zim Yre Yim P`` — 

57 **11** tokens with the impedance at 6/7. NEC-5's row has a third leading 

58 index and 12 tokens with the impedance at 7/8, so a shared parser would read 

59 the admittance as the impedance on one of the two. 

60* ``CURRENTS AND LOCATION``: ``SEG TAG X Y Z len Ire Iim mag phase`` — 10 

61 tokens, the same layout NEC-5 prints under a different heading. 

62* ``RADIATION PATTERNS``: angles then gains, ``TOTAL`` at index 4, and the 

63 ``SENSE`` word absent on a true null row, so 11 or 12 tokens. Same shape as 

64 NEC-5's. 

65"""

137def _run_form(exe: str, deck: str, form: str, timeout: float) -> str | None: 

138 with tempfile.TemporaryDirectory(prefix="nec2_") as td: 

139 tdp = Path(td) 

140 (tdp / "model.nec").write_text(deck) 

141 argv = [exe] 

142 stdin_text = "" 

143 if form == FORM_ARGS: 

144 argv += ["-i", "model.nec", "-o", "model.out"] 

145 else: 

146 # NEC5CL's convention: input name, output name, then a blank line. 

147 stdin_text = "model.nec\nmodel.out\n\n" 

148 try: 

149 proc = subprocess.run( 

150 argv, 

151 input=stdin_text, 

152 text=True, 

153 capture_output=True, 

154 cwd=td, 

155 timeout=timeout, 

156 ) 

157 except subprocess.TimeoutExpired as e: 

158 raise NEC2Error(f"NEC-2 timed out after {timeout:.0f}s") from e 

159 return _printout(tdp, "model.out", proc)

175def run_deck(exe: str, deck: str, *, timeout: float, form: str | None = None) -> str: 

176 """One deck through the binary, returning the printout text. 

177 

178 Module-level for the reason `nec5.run_deck` is: the availability probe must 

179 run a deck exactly the way a solve does, because "is this the right binary" 

180 is not a question the filesystem can answer. 

181 

182 `form` forces one invocation form; the default tries the argument form and 

183 then the stdin form, and caches whichever answered. 

184 """ 

185 ident = _identity(exe) 

186 forms: tuple[str, ...] 

187 if form is not None: 

188 forms = (form,) 

189 else: 

190 cached = _FORM_CACHE.get(exe) 

191 if cached is not None and ident is not None and cached[:2] == ident: 

192 forms = (cached[2],) 

193 else: 

194 forms = (FORM_ARGS, FORM_STDIN) 

195 

196 errors = [] 

197 for f in forms: 

198 try: 

199 text = _run_form(exe, deck, f, timeout) 

200 except OSError as e: 

201 raise NEC2Error(f"{exe} could not be executed: {e}") from e 

202 if text is not None: 

203 if ident is not None: 

204 _FORM_CACHE[exe] = (ident[0], ident[1], f) 

205 return text 

206 errors.append(f) 

207 raise NEC2Error( 

208 f"NEC-2 produced no printout from {exe} (tried the " 

209 + " and the ".join(errors) 

210 + " form)" 

211 )

214# A one-wire deck the probe runs to prove the binary IS a NEC-2. Half a 

215# wavelength at 300 MHz, three segments, centre-fed on segment 2 — the smallest 

216# input that prints an ANTENNA INPUT PARAMETERS block, which is what every 

217# solve path here parses. 

218# 

219# `XQ` is load-bearing, the same way it is in NEC-5's probe: without an 

220# execution request NEC reads every card, echoes the geometry and computes 

221# nothing, so a probe on an EN-terminated deck rejects a GENUINE binary. 

222_PROBE_DECK = """CM antennaknobs NEC2_EXE probe 

223CE 

224GW 1 3 0. -.25 0. 0. .25 0. .001 

225GE 0 

226EX 0 1 2 0 1. 0. 

227FR 0 1 0 0 300. 0. 

228XQ 

229EN 

230""" 

231 

232# path -> (mtime, size, verdict, reason), keyed on the file's identity rather 

233# than its name so replacing the binary in place re-probes. 

234_PROBE_CACHE: dict[str, tuple[float, int, bool, str]] = {}

237def probe_nec2(explicit: str | None = None, *, timeout: float = 20.0) -> str | None: 

238 """The resolved NEC-2 executable, or None with the reason logged. 

239 

240 `find_nec2` answers "is there an executable file at $NEC2_EXE", which is a 

241 question about the filesystem and not about NEC-2 — and the NEC-5 lane 

242 already paid for that distinction: on 2026-09-09 `NEC5_EXE` pointed at an 

243 18 KB spy shim, the roster offered a NEC-5 tab, and the failure arrived at 

244 the first solve as the shim's own error text (#1339). Any wrong file does 

245 that. 

246 

247 So this RUNS a one-wire deck once per (path, mtime, size) and requires a 

248 parseable printout. It doubles as the invocation-form discovery: whichever 

249 form answered here is cached for the solves that follow. 

250 """ 

251 exe = find_nec2(explicit) 

252 if exe is None: 

253 return None 

254 ident = _identity(exe) 

255 if ident is None: 

256 return None 

257 cached = _PROBE_CACHE.get(exe) 

258 if cached is not None and cached[:2] == ident: 

259 return exe if cached[2] else None 

260 

261 reason = "" 

262 try: 

263 text = run_deck(exe, _PROBE_DECK, timeout=timeout) 

264 ok = _AIP_HEADER in text 

265 if not ok: 

266 reason = ( 

267 f"{exe} ran but produced no {_AIP_HEADER} block; " 

268 f"printout head: {text[:300]!r}" 

269 ) 

270 except NEC2Error as e: 

271 ok, reason = False, f"{exe} is not a working NEC-2 binary: {e}" 

272 except OSError as e: # not executable on this platform, bad format, ... 

273 ok, reason = False, f"{exe} could not be executed: {e}" 

274 

275 _PROBE_CACHE[exe] = (ident[0], ident[1], ok, reason) 

276 if not ok: 

277 _log.warning("NEC-2 unavailable: %s", reason) 

278 return exe if ok else None

281def refuse_nec2_geometry(tups, ground, *, suggest_download: bool = False) -> None: 

282 """Refuse geometry a NEC-2 deck cannot carry, by name, rather than let the 

283 binary answer. 

284 

285 NEC-2's ground is a half-space boundary condition on the fields ABOVE it; the 

286 formulation has no below-ground medium at all. A buried wire is not rejected 

287 by nec2++ — it is solved **as if it were in air** and a number is printed with 

288 no warning, the worst failure available to an oracle. So the wrapper refuses, 

289 and so does the deck WRITER: handing a user a .nec of a buried antenna is the 

290 same wrong answer one step further away from anyone who could catch it 

291 (antennaknobs#1389). 

292 

293 One function for both, so the app cannot offer a download of a deck its own 

294 NEC-2 lane would refuse. A wire lying IN the plane is refused for a related 

295 reason — its image coincides with itself. A wire with ONE end at z=0 is 

296 ordinary NEC-2 (every ground-mounted vertical) and is not refused. 

297 

298 With no ground there is no plane and nothing to refuse: a free-space model may 

299 sit anywhere, below z=0 included. 

300 

301 `suggest_download` adds the sentence the gear menu needs — the NEC-5 deck of 

302 the same design, which is the one the user actually wanted. 

303 """ 

304 if ground is None or ground == "free": 

305 return 

306 instead = ( 

307 " — download the NEC-5 deck instead, whose ground serves buried conductors" 

308 if suggest_download 

309 else "" 

310 ) 

311 for i, t in enumerate(tups): 

312 w = as_wire(t) 

313 z0, z1 = float(w.p0[2]), float(w.p1[2]) 

314 if z0 == 0.0 and z1 == 0.0: 

315 raise ValueError( 

316 f"wire {i + 1} lies in the ground plane (z=0), where its own " 

317 "image coincides with it — NEC-2's ground forbids it" 

318 ) 

319 if (z0 < 0.0 < z1) or (z1 < 0.0 < z0): 

320 raise NotImplementedError( 

321 f"wire {i + 1} crosses the ground plane mid-span: NEC-2 has no " 

322 "below-ground medium and solves the buried part as if it were in " 

323 "air, so the wrapper refuses rather than report a number nothing " 

324 "warns about — split the wire at z=0 and run the buried part on " 

325 "momwire (or NEC-5)" + instead 

326 ) 

327 if z0 < 0.0 or z1 < 0.0: 

328 raise NotImplementedError( 

329 f"wire {i + 1} dips below z=0: NEC-2 has no below-ground medium " 

330 "and solves a buried conductor as if it were in air — use the " 

331 "momwire engine, whose buried serve is certified, or NEC-5, whose " 

332 "Sommerfeld path serves it" + instead 

333 )

336class NEC2Engine(SimulationEngine): 

337 """A user-supplied NEC-2 console binary, driven over text. 

338 

339 Same physics as `PyNECEngine` — it is the same code family — reached 

340 without linking to it. Where the two differ is worth knowing before 

341 comparing numbers: PyNEC solves TL / virtual-driver networks by a 

342 multiport-Y reduction outside the field solve, and there is no faithful 

343 single-deck spelling of that, so `nec_export.export_nec` refuses those and 

344 so does this engine. A design PyNEC serves and this refuses is that. 

345 """ 

346 

347 supports_far_field = True 

348 # NEC-2 puts a voltage source at a segment CENTRE, so a centre feed wants 

349 # an odd count with the source on the middle segment — PyNECEngine's parity, 

350 # not NEC-5's "even" (whose source sits at a segment end). 

351 segment_parity = "odd" 

352 

353 def __init__( 

354 self, 

355 builder, 

356 *, 

357 ground=None, 

358 nec2_exe: str | None = None, 

359 timeout: float = 120.0, 

360 capture_dir=None, 

361 ): 

362 super().__init__(builder) 

363 from .pynec import DEFAULT_GROUND 

364 

365 self.ground = DEFAULT_GROUND if ground is None else ground 

366 self.timeout = float(timeout) 

367 # AK#1428: with a capture dir (or `ANTENNAKNOBS_CAPTURE_DIR/nec2` from 

368 # the environment) every run writes `<hash>.nec` and `<hash>.out` 

369 # beside each other. Write-only — unlike NEC-5's #872 cache, a 

370 # captured NEC-2 printout is never served back. 

371 if capture_dir is None: 

372 from ..engine_capture import capture_dir_from_env 

373 

374 capture_dir = capture_dir_from_env("nec2") 

375 self._capture_dir = Path(capture_dir).expanduser() if capture_dir else None 

376 if self._capture_dir is not None: 

377 self._capture_dir.mkdir(parents=True, exist_ok=True) 

378 exe = find_nec2(nec2_exe) 

379 if exe is None: 

380 raise NEC2Error( 

381 "no NEC-2 binary: set $NEC2_EXE (or pass nec2_exe=) to a NEC-2 " 

382 "console executable — nec2c, nec2++, or 4nec2's nec2dxs*.exe" 

383 ) 

384 self.exe = exe 

385 # COERCED, not raw: `export_nec` builds its deck through a 

386 # `PyNECEngine`, which applies `segment_parity` to every fed or named 

387 # wire, so a raw `build_wires()` here would disagree with the deck the 

388 # binary actually ran and mis-map its currents wire by wire. 

389 self.tups = self._coerce_wire_tuples(builder.build_wires()) 

390 refuse_graded_wires(self.tups, "NEC-2") 

391 self._check_geometry_against_ground() 

392

394 def _check_geometry_against_ground(self) -> None: 

395 return refuse_nec2_geometry(self.tups, self.ground)

398 def deck(self, freq: float, *, npoints: int = 1, df: float = 0.0, rp=None) -> str: 

399 """The deck for one run, from `nec_export.export_nec`. 

400 

401 `rp` is (n_theta, n_phi, del_theta, del_phi) for a pattern run; None 

402 asks for impedance only, which `export_nec` spells as `XQ`. 

403 """ 

404 from ..nec_export import export_nec 

405 

406 text = export_nec( 

407 self.builder, 

408 ground=self.ground, 

409 freq=freq, 

410 df=df, 

411 npoints=npoints, 

412 include_rp=False, 

413 ) 

414 if rp is None: 

415 return text 

416 n_theta, n_phi, del_theta, del_phi = rp 

417 card = f"RP 0 {n_theta} {n_phi} 1000 0 0 {del_theta:g} {del_phi:g}" 

418 # export_nec closes with XQ then EN; the pattern card replaces the XQ 

419 # (an RP triggers the solve itself) so the deck runs once, not twice. 

420 lines = [ln for ln in text.splitlines() if ln.split()[:1] != ["XQ"]] 

421 assert lines[-1] == "EN", lines[-1] 

422 lines[-1:] = [card, "EN"] 

423 return "\n".join(lines) + "\n"

425 def _run(self, deck: str) -> str: 

426 h = hashlib.sha256(deck.encode()).hexdigest()[:16] 

427 _log.debug( 

428 "NEC-2 %s: deck for %s (%d lines)\n%s", h, self.exe, deck.count("\n"), deck 

429 ) 

430 t0 = time.perf_counter() 

431 text = run_deck(self.exe, deck, timeout=self.timeout) 

432 _log.info( 

433 "NEC-2 %s: %.2f s, printout %d lines", 

434 h, 

435 time.perf_counter() - t0, 

436 text.count("\n"), 

437 ) 

438 _log.debug("NEC-2 %s: printout\n%s", h, text) 

439 if self._capture_dir is not None: 

440 (self._capture_dir / f"{h}.nec").write_text(deck) 

441 (self._capture_dir / f"{h}.out").write_text(text) 

442 _log.info( 

443 "NEC-2 %s: deck and printout captured under %s", h, self._capture_dir 

444 ) 

445 if _AIP_HEADER not in text: 

446 # The binary ran and reported a fault. Say WHAT it said: without 

447 # this the first parser to look raises "no ANTENNA INPUT 

448 # PARAMETERS" or "no POWER BUDGET", which is true and useless — 

449 # it names the block that is missing rather than the reason. 

450 # Found on a design carrying an `LD 2` card, where the answer was 

451 # "LD type 2 is not supported by this engine". 

452 for line in text.splitlines(): 

453 if _ENGINE_ERROR_RE.search(line): 

454 raise NEC2Error(f"{Path(self.exe).name}: {line.strip()}") 

455 return text

458 @staticmethod 

459 def _parse_input_parameters(text: str) -> list[list[tuple[int, int, complex]]]: 

460 """Every ANTENNA INPUT PARAMETERS section, one list per frequency, each 

461 row (tag, seg, Z). Row layout: TAG SEG Vre Vim Ire Iim Zre Zim Yre Yim 

462 POWER — 11 tokens, impedance at 6/7 (see the module docstring on why 

463 this is not NEC-5's parser).""" 

464 chunks = text.split(_AIP_HEADER)[1:] 

465 if not chunks: 

466 raise NEC2Error(f"no {_AIP_HEADER} in NEC-2 printout; tail: " + text[-500:]) 

467 out = [] 

468 for chunk in chunks: 

469 rows: list[tuple[int, int, complex]] = [] 

470 for line in chunk.splitlines(): 

471 toks = line.split() 

472 if len(toks) != 11: 

473 if rows: 

474 break 

475 continue 

476 try: 

477 tag, seg = int(toks[0]), int(toks[1]) 

478 z = complex(float(toks[6]), float(toks[7])) 

479 except ValueError: 

480 if rows: 

481 break 

482 continue 

483 rows.append((tag, seg, z)) 

484 if not rows: 

485 raise NEC2Error(f"unparseable {_AIP_HEADER} section") 

486 out.append(rows) 

487 return out

489 @staticmethod 

490 def _parse_currents(text: str) -> list[dict[int, list[complex]]]: 

491 """Every CURRENTS AND LOCATION section, one dict per frequency mapping 

492 tag -> segment-centre currents in segment order. Row layout: SEG TAG X 

493 Y Z len Ire Iim mag phase — 10 tokens.""" 

494 chunks = text.split(_CURRENTS_HEADER)[1:] 

495 out = [] 

496 for chunk in chunks: 

497 per_tag: dict[int, list[complex]] = {} 

498 started = False 

499 for line in chunk.splitlines(): 

500 toks = line.split() 

501 if len(toks) != 10: 

502 if started: 

503 break 

504 continue 

505 try: 

506 tag = int(toks[1]) 

507 cur = complex(float(toks[6]), float(toks[7])) 

508 except ValueError: 

509 if started: 

510 break 

511 continue 

512 started = True 

513 per_tag.setdefault(tag, []).append(cur) 

514 if per_tag: 

515 out.append(per_tag) 

516 if not out: 

517 raise NEC2Error(f"no {_CURRENTS_HEADER} section in NEC-2 printout") 

518 return out

520 @staticmethod 

521 def _parse_radiation_patterns(text: str) -> dict[tuple[float, float], float]: 

522 """The RADIATION PATTERNS section as {(theta, phi): total gain dB}. 

523 

524 Row layout: THETA PHI VERT HORIZ TOTAL AXIAL TILT [SENSE] E(TH)mag 

525 phase E(PHI)mag phase — 12 tokens with the SENSE word, 11 when it is 

526 blank on a true null row. The angles and TOTAL sit at fixed leading 

527 positions either way, which is what makes the count tolerance safe. 

528 """ 

529 try: 

530 chunk = text.split(_PATTERN_HEADER, 1)[1] 

531 except IndexError: 

532 raise NEC2Error( 

533 f"no {_PATTERN_HEADER} in NEC-2 printout; tail: " + text[-500:] 

534 ) from None 

535 gains: dict[tuple[float, float], float] = {} 

536 started = False 

537 for line in chunk.splitlines(): 

538 toks = line.split() 

539 if len(toks) not in (11, 12): 

540 if started: 

541 break 

542 continue 

543 try: 

544 theta, phi = float(toks[0]), float(toks[1]) 

545 total = float(toks[4]) 

546 except ValueError: 

547 if started: 

548 break 

549 continue 

550 started = True 

551 gains[(round(theta, 2), round(phi, 2))] = total 

552 if not gains: 

553 raise NEC2Error(f"unparseable {_PATTERN_HEADER} section") 

554 return gains

556 @staticmethod 

557 def _parse_feed_voltages(text: str) -> list[complex]: 

558 """The DRIVE VALUE of each feed, from the printout's own VOLTAGE columns. 

559 

560 Taken from the report rather than reconstructed from the deck, because 

561 the report is what the binary was actually driven with — and because 

562 this engine has no resolved-feed list of its own to reconstruct from: 

563 `nec_export.export_nec` builds and discards the `PyNECEngine` that 

564 resolves them. Row layout as `_parse_input_parameters`: V at 2/3. 

565 """ 

566 chunks = text.split(_AIP_HEADER)[1:] 

567 if not chunks: 

568 raise NEC2Error(f"no {_AIP_HEADER} in NEC-2 printout") 

569 out: list[complex] = [] 

570 for line in chunks[0].splitlines(): 

571 toks = line.split() 

572 if len(toks) != 11: 

573 if out: 

574 break 

575 continue 

576 try: 

577 int(toks[0]), int(toks[1]) 

578 out.append(complex(float(toks[2]), float(toks[3]))) 

579 except ValueError: 

580 if out: 

581 break 

582 continue 

583 return out

585 @staticmethod 

586 def _parse_power_budget(text: str) -> dict: 

587 """The POWER BUDGET block, in the same keys `NEC5Engine` returns. 

588 

589 NEC-2's block, and the spellings are its own:: 

590 

591 INPUT POWER = 1.1068E-02 Watts 

592 RADIATED POWER= 1.1068E-02 Watts 

593 STRUCTURE LOSS= 0.0000E+00 Watts 

594 NETWORK LOSS = 0.0000E+00 Watts 

595 EFFICIENCY = 100.00 Percent 

596 

597 Two differences from NEC-5's that a shared parser would get wrong: the 

598 conductor loss is ``STRUCTURE LOSS`` rather than ``WIRE LOSS``, and 

599 there is a ``NETWORK LOSS`` line NEC-5 has no counterpart for. The 

600 label match is on the label text and not on a column, because 

601 ``RADIATED POWER=`` carries no space before its ``=``. 

602 

603 Raises rather than returning partial: a budget with a missing line is 

604 how an efficiency of 1.0 gets shipped as if it were measured. 

605 """ 

606 try: 

607 chunk = text.split("POWER BUDGET", 1)[1] 

608 except IndexError: 

609 raise NEC2Error("no POWER BUDGET in NEC-2 printout") from None 

610 keys = { 

611 "INPUT POWER": "input_w", 

612 "RADIATED POWER": "radiated_w", 

613 "STRUCTURE LOSS": "wire_loss_w", 

614 "NETWORK LOSS": "network_loss_w", 

615 "EFFICIENCY": "efficiency_pct", 

616 } 

617 out: dict = {} 

618 for line in chunk.splitlines()[:12]: 

619 for label, key in keys.items(): 

620 if label in line and "=" in line: 

621 out[key] = float(line.split("=", 1)[1].split()[0]) 

622 missing = set(keys.values()) - set(out) 

623 if missing: 

624 raise NEC2Error( 

625 f"incomplete POWER BUDGET section: missing {sorted(missing)}, " 

626 f"parsed {out}" 

627 ) 

628 return out

630 def solve_snapshot(self): 

631 """One binary run serving the whole web-solve contract: impedances, 

632 knot currents and the power budget from ONE printout. 

633 

634 `impedance()` and `current_distribution()` each spawn a process, and a 

635 web solve must not pay twice. Also stamps the duck-typed 

636 ``_excited_efficiency`` / ``_excited_p_in`` / ``_excited_power_budget`` 

637 the web adapter's efficiency and budget helpers read on every engine. 

638 

639 **ONLY this method stamps them** — not `impedance()`, not 

640 `current_distribution()`. `PyNECEngine` differs here: its 

641 `current_distribution()` stamps them as a side effect of the in-process 

642 solve it already has, so a caller that works on the PyNEC engine can 

643 read nothing on this one. The web lane calls `solve_snapshot`, so it is 

644 served; `cli.py`'s ``--power`` schematic annotation and the pattern 

645 command call `current_distribution()` and then read the attributes, so 

646 on this engine they find None and omit the power table. That is a 

647 missing annotation and not a wrong number — both sites guard with 

648 ``if budget and p_in`` — but it is why a NEC-2 or NEC-5 run prints no 

649 budget there where a PyNEC run does. 

650 

651 THE BUDGET IS NOT OPTIONAL. Those attributes have plausible fallbacks — 

652 `getattr(eng, "_excited_efficiency", 1.0)` — and a NEC-2 tab reporting 

653 100 % efficiency because nothing was parsed is the same shape of wrong 

654 answer as the buried wire this engine refuses: confident, unwarned, and 

655 indistinguishable from a real result. So a printout with no budget 

656 raises, and the sentence names what was missing. 

657 

658 A plain ``XQ`` deck already carries the block — measured through 

659 momwire's portal, which writes nec2c's printout column for column, on a 

660 43-segment and a 1376-segment deck. A build that needs a pattern 

661 request to print one gets ONE retry with the smallest possible ``RP`` 

662 (1x1, measured at ~10 ms of the 0.45 s / 1.10 s those decks cost, i.e. 

663 inside the process-startup noise); if that printout has no budget 

664 either, the solve refuses. 

665 """ 

666 text = self._run(self.deck(self.builder.freq)) 

667 try: 

668 budget = self._parse_power_budget(text) 

669 except NEC2Error: 

670 text = self._run(self.deck(self.builder.freq, rp=(1, 1, 0, 0))) 

671 budget = self._parse_power_budget(text) 

672 zs = self._impedances(self._parse_input_parameters(text)[0]) 

673 currents = self._currents_from(self._parse_currents(text)[0]) 

674 self._excited_efficiency = budget["efficiency_pct"] / 100.0 

675 self._excited_p_in = budget["input_w"] 

676 # LOSSES ONLY — see the power-budget protocol comment in 

677 # `antennaknobs.engine`. The radiated power is NOT a budget row: every 

678 # consumer subtracts the rows from the input to get what reaches the 

679 # antenna, so listing P_rad there printed "antenna (accepted): 0 %" on a 

680 # lossless design where momwire printed 100 % (issue #1354). 

681 self._excited_power_budget = [("Wire loss", budget["wire_loss_w"])] 

682 if budget["network_loss_w"]: 

683 # NEC-5's block has no counterpart, so this row only appears here. 

684 self._excited_power_budget.append( 

685 ("Network loss", budget["network_loss_w"]) 

686 ) 

687 self._excited_p_radiated = budget["radiated_w"] 

688 # The per-feed drive values, for the web lane's multi-feed response. 

689 self._excited_feed_values = self._parse_feed_voltages(text) 

690 return zs, currents, budget

696 def impedance(self): 

697 """One impedance per driven port, in EX-card order. 

698 

699 A LIST even for a single feed, which is `PyNECEngine`'s and 

700 `NEC5Engine`'s contract: a caller that unpacks the sequence works on 

701 every engine, where a scalar-for-one special case makes the shape 

702 depend on the design. 

703 """ 

704 rows = self._parse_input_parameters(self._run(self.deck(self.builder.freq))) 

705 zs = self._impedances(rows[0]) 

706 if not zs: 

707 raise NEC2Error("NEC-2 printed no driving-point impedance") 

708 return zs

710 def impedance_sweep(self, freqs): 

711 """(n_freqs, n_ports) impedances — one binary run per frequency. 

712 

713 NEC-2's FR card can step a uniform grid in one run, and this 

714 deliberately does not: the sweep endpoints ask for arbitrary frequency 

715 lists, and a per-point run is the shape that always answers. Batching a 

716 uniform grid is an optimisation, not a correctness change. 

717 """ 

718 freqs = np.asarray(freqs, dtype=float) 

719 if freqs.ndim != 1 or freqs.size == 0: 

720 raise ValueError("freqs must be a 1-D non-empty array") 

721 out = [] 

722 for f in freqs: 

723 rows = self._parse_input_parameters(self._run(self.deck(float(f)))) 

724 out.append(self._impedances(rows[0])) 

725 return np.array(out)

729 per_tag = self._parse_currents(text)[0]

731 

732 def _currents_from(self, per_tag) -> list[WireCurrents]: 

733 """One `WireCurrents` per exported wire: n_seg+1 KNOT positions, with 

734 interior knots the average of the two adjacent segment-centre currents. 

735 

736 `PyNECEngine.current_distribution`'s rule exactly, because this is the 

737 same physics reached a different way and the web UI renders both with 

738 one renderer: a boundary knot goes to zero at a genuine free end (the 

739 open-wire boundary condition) but carries the adjacent centre current at 

740 a JUNCTION, where the current is continuous through the shared point. 

741 Without that, a one-segment feed stub — both ends junctions — would 

742 render zero current along its whole length while sitting at a current 

743 maximum, opening a visible gap right at the feed. 

744 

745 `export_nec` writes one GW per wire tuple with tag = index + 1, so the 

746 tag IS the wire. 

747 """ 

748 

749 def _key(p): 

750 return tuple(np.round(np.asarray(p, dtype=float), 6)) 

751 

752 endpoint_count: dict = {} 

753 for t in self.tups: 

754 for p in (as_wire(t).p0, as_wire(t).p1): 

755 endpoint_count[_key(p)] = endpoint_count.get(_key(p), 0) + 1 

756 

757 out = [] 

758 for i, t in enumerate(self.tups): 

759 w = as_wire(t) 

760 n_seg = int(w.n_seg) 

761 cur_per_seg = np.asarray(per_tag.get(i + 1, []), dtype=np.complex128) 

762 if cur_per_seg.shape[0] != n_seg: 

763 raise NEC2Error( 

764 f"tag {i + 1}: expected {n_seg} segment currents, got " 

765 f"{cur_per_seg.shape[0]}" 

766 ) 

767 knots = np.linspace( 

768 np.asarray(w.p0, dtype=float), np.asarray(w.p1, dtype=float), n_seg + 1 

769 ) 

770 knot_cur = np.zeros(n_seg + 1, dtype=np.complex128) 

771 if n_seg >= 2: 

772 knot_cur[1:-1] = 0.5 * (cur_per_seg[:-1] + cur_per_seg[1:]) 

773 if n_seg >= 1: 

774 if endpoint_count.get(_key(w.p0), 0) >= 2: 

775 knot_cur[0] = cur_per_seg[0] 

776 if endpoint_count.get(_key(w.p1), 0) >= 2: 

777 knot_cur[-1] = cur_per_seg[-1] 

778 out.append(WireCurrents(knot_positions=knots, knot_currents=knot_cur)) 

779 return out

781 def far_field(self, *, n_theta=90, n_phi=360, del_theta=1, del_phi=1): 

782 # Same grid contract as PyNECEngine._collect_pattern: the upper 

783 # hemisphere in theta (0 .. 90-del from zenith), full circle in phi 

784 # with the 360-degree seam duplicated. 

785 assert 90 % n_theta == 0 and 90 == del_theta * n_theta 

786 assert 360 % n_phi == 0 and 360 == del_phi * n_phi 

787 text = self._run( 

788 self.deck(self.builder.freq, rp=(n_theta, n_phi, del_theta, del_phi)) 

789 ) 

790 gains = self._parse_radiation_patterns(text) 

791 thetas = np.arange(n_theta) * del_theta 

792 phis = np.arange(n_phi + 1) * del_phi 

793 rings = [] 

794 for th in thetas: 

795 ring = [ 

796 gains[(round(float(th), 2), round(float(ph % 360.0), 2))] for ph in phis 

797 ] 

798 rings.append(ring) 

799 flat = [g for ring in rings for g in ring if g > NULL_GAIN_DB] 

800 return FarField( 

801 rings=rings, 

802 max_gain=max(flat) if flat else NULL_GAIN_DB, 

803 min_gain=min(flat) if flat else NULL_GAIN_DB, 

804 thetas=thetas.astype(float), 

805 phis=phis.astype(float), 

806 )
